    29→  /**
    30→   * Main initialization function - called when DOM is ready
    31→   */
    32→  async function initialize() {
    33→    if (AppState.initialized || AppState.initializing) return;
    34→    AppState.initializing = true;
    35→    
    36→    console.log('🚀 Initializing SBIDE...');
    37→    
    38→    try {
    39→      // Show loading state or hide error
    40→      const errorEl = document.getElementById('init-error-message');
    41→      if (errorEl) errorEl.classList.add('hidden');
    42→      
    43→      // Step 1: Initialize storage (IndexedDB)
    44→      console.log('📦 Initializing storage...');
    45→      await IDEStorage.init();
    46→      /* storage fallback warning — non-blocking */
    47→      if (window.IDEStorage && IDEStorage.isFallbackMode && IDEStorage.isFallbackMode()) {
    48→        console.warn('⚠️ Storage running in fallback (in-memory) mode — data will not persist across reloads.');
    49→        if (window.IDEUtils && IDEUtils.showToast) {
    50→          IDEUtils.showToast('Storage unavailable — running in temporary mode. Data won\'t persist.', 'warning', 6000);
    51→        }
    52→      }
    53→      console.log('✅ Storage initialized');
    54→      
    55→      // Step 2: Initialize state from persistence (MUST be before themes!)
    56→      console.log('💾 Hydrating state...');
    57→      const savedState = IDEState.getState();
    58→      console.log('✅ State hydrated', { 
    59→        hasProject: !!savedState.currentProject, 
    60→        theme: savedState.settings?.theme 
    61→      });
    62→      
    63→      // Step 3: Initialize themes (AFTER state is loaded)
    64→      console.log('🎨 Initializing themes...');
    65→      if (window.IDEThemes) {
    66→        IDEThemes.init();
    67→      }
    68→      console.log('✅ Themes initialized');
    69→      
    70→      // Step 4: Initialize components in order
    71→      await initializeComponents();
    72→      
    73→      // Step 5: Setup global event handlers
    74→      setupGlobalHandlers();
    75→      
    76→      // Step 6: Load initial data
    77→      await loadInitialData();
    78→      
    79→      // Mark as initialized
    80→      AppState.initialized = true;
    81→      AppState.initializing = false;
    82→      
    83→      console.log('✨ SBIDE ready!');
    84→      
    85→      // Dispatch ready event
    86→      window.dispatchEvent(new CustomEvent('ide-ready'));
    87→      
    88→    } catch (error) {
    89→      console.error('❌ Initialization failed:', error);
    90→      AppState.error = error;
    91→      AppState.initializing = false;
    92→      
    93→      showInitError(error.message || 'Failed to initialize application');
    94→    }
    95→  }

   223→    // ============================================
   224→    // Mobile bottom-nav buttons (Files / Chat / Editor / Settings)
   225→    // Wire up click handlers so the buttons actually do something.
   226→    // On mobile (<768px), only one panel can comfortably fit at a time,
   227→    // so each button hides the others and shows its target.
   228→    // ============================================
   229→    const mobileNav = document.getElementById('mobile-nav');
   230→    
   231→    // Create backdrop overlay for sidebar mode (tapping outside closes it)
   232→    let mobileBackdrop = null;
   233→    function getMobileBackdrop() {
   234→      if (!mobileBackdrop) {
   235→        mobileBackdrop = document.createElement('div');
   236→        mobileBackdrop.className = 'mobile-overlay';
   237→        mobileBackdrop.setAttribute('aria-hidden', 'true');
   238→        mobileBackdrop.addEventListener('click', () => {
   239→          // When backdrop is tapped, switch back to chat view
   240→          const chatBtn = mobileNav?.querySelector('.mobile-nav-btn[data-panel="chat"]');
   241→          if (chatBtn) chatBtn.click();
   242→        });
   243→        document.body.appendChild(mobileBackdrop);
   244→      }
   245→      return mobileBackdrop;
   246→    }
   247→
   248→    function showMobileBackdrop() {
   249→      const backdrop = getMobileBackdrop();
   250→      backdrop.classList.remove('hidden');
   251→      // Force reflow then add visible
   252→      void backdrop.offsetWidth;
   253→    }
   254→
   255→    function hideMobileBackdrop() {
   256→      if (mobileBackdrop) {
   257→        mobileBackdrop.classList.add('hidden');
   258→      }
   259→    }
   260→
   261→    // Debounce tracker to prevent rapid-fire button presses
   262→    let lastMobileNavClick = 0;
   263→    const MOBILE_NAV_DEBOUNCE_MS = 150;
   264→
   265→    if (mobileNav) {
   266→      const mobileBtns = mobileNav.querySelectorAll('.mobile-nav-btn');
   267→      
   268→      mobileBtns.forEach(btn => {
   269→        btn.addEventListener('click', (e) => {
   270→          // Prevent default for touch devices to avoid any ghost clicks
   271→          // e.preventDefault();
   272→          
   273→          if (btn.disabled) return;
   274→          
   275→          // Debounce: ignore rapid repeated clicks
   276→          const now = Date.now();
   277→          if (now - lastMobileNavClick < MOBILE_NAV_DEBOUNCE_MS) return;
   278→          lastMobileNavClick = now;
   279→
   280→          const panel = btn.dataset.panel;
   281→          if (!panel) return;
   282→
   283→          // Update active state on the nav buttons
   284→          // Special case: settings is a momentary action — don't keep it active
   285→          const isMomentaryAction = panel === 'settings';
   286→          mobileBtns.forEach(b => {
   287→            if (isMomentaryAction) {
   288→              b.classList.remove('active');
   289→              // Restore previous active state after a delay
   290→              requestAnimationFrame(() => {
   291→                const chatBtn = mobileNav.querySelector('.mobile-nav-btn[data-panel="chat"]');
   292→                if (chatBtn) chatBtn.classList.add('active');
   293→              });
   294→            } else {
   295→              b.classList.toggle('active', b === btn);
   296→            }
   297→          });
   298→
   299→          const sidebarEl = document.getElementById('sidebar');
   300→          const chatEl = document.getElementById('chat-area');
   301→          const editorEl = document.getElementById('editor-panel');
   302→
   303→          switch (panel) {
   304→            case 'sidebar': {
   305→              // Show sidebar, hide chat + editor on mobile
   306→              if (sidebarEl) sidebarEl.classList.remove('hidden');
   307→              if (chatEl) chatEl.classList.add('hidden');
   308→              if (editorEl) editorEl.classList.add('hidden');
   309→              if (AppState?.sidebar) AppState.sidebar.showSidebar();
   310→              if (window.IDEState) IDEState.setSidebarOpen(true);
   311→              // Show backdrop so user can tap to close sidebar
   312→              showMobileBackdrop();
   313→              break;
   314→            }
   315→            case 'chat': {
   316→              // Show chat, hide sidebar + editor on mobile
   317→              if (sidebarEl) {
   318→                sidebarEl.classList.add('hidden');
   319→                // Also close the sidebar slide-out
   320→                if (AppState?.sidebar) AppState.sidebar.hideSidebar?.();
   321→              }
   322→              if (chatEl) chatEl.classList.remove('hidden');
   323→              if (editorEl) editorEl.classList.add('hidden');
   324→              if (window.IDEState) IDEState.setEditorPanelOpen(false);
   325→              hideMobileBackdrop();
   326→              break;
   327→            }
   328→            case 'editor': {
   329→              // Show editor, hide sidebar + chat on mobile
   330→              if (sidebarEl) {
   331→                sidebarEl.classList.add('hidden');
   332→                if (AppState?.sidebar) AppState.sidebar.hideSidebar?.();
   333→              }
   334→              if (chatEl) chatEl.classList.add('hidden');
   335→              if (editorEl) editorEl.classList.remove('hidden');
   336→              if (window.IDEState) IDEState.setEditorPanelOpen(true);
   337→              hideMobileBackdrop();
   338→              break;
   339→            }
   340→            case 'settings': {
   341→              // Open the settings modal — this is momentary, don't change panels
   342→              if (AppState?.sidebar && typeof AppState.sidebar.toggleSettings === 'function') {
   343→                AppState.sidebar.toggleSettings(true);
   344→                
   345→                // Watch for settings modal close to restore proper active state
   346→                const settingsModal = document.getElementById('settings-modal');
   347→                if (settingsModal) {
   348→                  const observer = new MutationObserver((mutations) => {
   349→                    mutations.forEach((mutation) => {
   350→                      if (mutation.attributeName === 'class') {
   351→                        if (settingsModal.classList.contains('hidden')) {
   352→                          // Modal closed — ensure chat button is active again
   353→                          const chatBtn = mobileNav.querySelector('.mobile-nav-btn[data-panel="chat"]');
   354→                          mobileBtns.forEach(b => b.classList.remove('active'));
   355→                          if (chatBtn) chatBtn.classList.add('active');
   356→                          observer.disconnect();
   357→                        }
   358→                      }
   359→                    });
   360→                  });
   361→                  observer.observe(settingsModal, { attributes: true });
   362→                  
   363→                  // Auto-disconnect after 10 seconds (safety net)
   364→                  setTimeout(() => observer.disconnect(), 10000);
   365→                }
   366→              }
   367→              break;
   368→            }
   369→          }
   370→        });
   371→
   372→        // Prevent touch-hold context menu on nav buttons
   373→        btn.addEventListener('contextmenu', (e) => e.preventDefault());
   374→      });
   375→
   376→      // Keep mobile-nav button active state in sync with editor panel visibility
   377→      if (window.IDEState) {
   378→        IDEState.subscribe((state, prevState) => {
   379→          if (state.editorPanelOpen !== prevState.editorPanelOpen) {
   380→            const editorBtn = mobileNav.querySelector('.mobile-nav-btn[data-panel="editor"]');
   381→            const chatBtn = mobileNav.querySelector('.mobile-nav-btn[data-panel="chat"]');
   382→            // Only update active state if we're on mobile (mobile-nav is visible)
   383→            const isMobile = window.matchMedia('(max-width: 767px)').matches;
   384→            if (!isMobile) return;
   385→            if (editorBtn && chatBtn) {
   386→              editorBtn.classList.toggle('active', state.editorPanelOpen);
   387→              chatBtn.classList.toggle('active', !state.editorPanelOpen);
   388→            }
   389→          }
   390→        });
   391→      }
   392→
   393→      // Handle sidebar close from within sidebar component (e.g., X button)
   394→      // and sync mobile nav state
   395→      const sidebarObserver = new MutationObserver((mutations) => {
   396→        const sidebarEl = document.getElementById('sidebar');
   397→        if (!sidebarEl) return;
   398→        
   399→        mutations.forEach((mutation) => {
   400→          if (mutation.attributeName === 'class') {
   401→            const isOpen = sidebarEl.classList.contains('open') && !sidebarEl.classList.contains('hidden');
   402→            const sidebarBtn = mobileNav.querySelector('.mobile-nav-btn[data-panel="sidebar"]');
   403→            
   404→            if (!isOpen && sidebarBtn?.classList.contains('active')) {
   405→              // Sidebar was closed externally — switch to chat
   406→              const chatBtn = mobileNav.querySelector('.mobile-nav-btn[data-panel="chat"]');
   407→              mobileBtns.forEach(b => b.classList.remove('active'));
   408→              if (chatBtn) chatBtn.classList.add('active');
   409→              hideMobileBackdrop();
   410→            }
   411→          }
   412→        });
   413→      });
   414→
   415→      const sidebarEl = document.getElementById('sidebar');
   416→      if (sidebarEl) {
   417→        sidebarObserver.observe(sidebarEl, { attributes: true });
   418→      }
   419→    }

   564→    // --- Independent visibility toggles (Issue #3 fix) ---
   565→    // Chat and Editor buttons toggle their respective panels on/off
   566→    // independently. Both can be visible at once (split view), and
   567→    // either can be the sole visible panel (filling the full width).

   571→    function syncViewToggleStates() {
   572→      const chat = document.getElementById('chat-area');
   573→      const editor = document.getElementById('editor-panel');
   574→      const chatVisible = chat ? !chat.classList.contains('hidden') : false;
   575→      const editorVisible = editor ? !editor.classList.contains('hidden') : false;
   576→
   577→      if (viewChatBtn) {
   578→        viewChatBtn.classList.toggle('active', chatVisible);
   579→        viewChatBtn.setAttribute('aria-selected', String(chatVisible));
   580→        viewChatBtn.setAttribute('aria-pressed', String(chatVisible));
   581→      }
   582→      if (viewEditorBtn) {
   583→        viewEditorBtn.classList.toggle('active', editorVisible);
   584→        viewEditorBtn.setAttribute('aria-selected', String(editorVisible));
   585→        viewEditorBtn.setAttribute('aria-pressed', String(editorVisible));
   586→      }
   587→      // Preview button active state is handled separately when entering
   588→      // preview mode; default to inactive on sync.
   589→      if (viewPreviewBtn && !window.__idePreviewActive) {
   590→        viewPreviewBtn.classList.remove('active');
   591→        viewPreviewBtn.setAttribute('aria-selected', 'false');
   592→        viewPreviewBtn.setAttribute('aria-pressed', 'false');
   593→      }
   594→    }

   596→    function ensureAtLeastOneVisible() {

  1090→  /**
  1091→   * Download the currently active file in the editor as a single-file download
  1092→   * (not a ZIP). Reads the latest content from storage so unsaved edits are
  1093→   * not lost. If no file is active, falls back to a toast prompt.
  1094→   */
  1095→  async function handleDownloadCurrentFile() {
  1096→    const activeFile = IDEState?.get('activeFile');
  1097→    if (!activeFile || !activeFile.path) {
  1098→      if (IDEUtils) IDEUtils.showToast('Open a file first to download it', 'info');
  1099→      return;
  1100→    }
  1101→    
  1102→    const project = IDEState?.get('currentProject');
  1103→    if (!project) {
  1104→      if (IDEUtils) IDEUtils.showToast('No project selected', 'warning');
  1105→      return;
  1106→    }
  1107→    
  1108→    try {
  1109→      // Read the latest content from storage (respects any saved edits).
  1110→      // Fall back to activeFile.content if storage read fails.
  1111→      let content = activeFile.content;
  1112→      try {
  1113→        const stored = await IDEStorage.Files.readContent(project.name, activeFile.path);
  1114→        if (stored != null) content = stored;
  1115→      } catch (e) {
  1116→        console.warn('Could not read file from storage, using in-memory copy:', e);
  1117→      }
  1118→      
  1119→      // Pick a sensible MIME type from the file extension
  1120→      const ext = (activeFile.name.split('.').pop() || '').toLowerCase();
  1121→      const mimeMap = {
  1122→        html: 'text/html', htm: 'text/html',
  1123→        css: 'text/css', js: 'text/javascript', mjs: 'text/javascript',
  1124→        ts: 'text/typescript', tsx: 'text/typescript', jsx: 'text/javascript',
  1125→        json: 'application/json', md: 'text/markdown',
  1126→        txt: 'text/plain', svg: 'image/svg+xml', xml: 'application/xml',
  1127→        yaml: 'text/yaml', yml: 'text/yaml'
  1128→      };
  1129→      const mime = mimeMap[ext] || 'text/plain';
  1130→      
  1131→      const blob = new Blob([content || ''], { type: mime + ';charset=utf-8' });
  1132→      const url = URL.createObjectURL(blob);
  1133→      
  1134→      const link = document.createElement('a');
  1135→      link.href = url;
  1136→      link.download = activeFile.name;
  1137→      document.body.appendChild(link);
  1138→      link.click();
  1139→      document.body.removeChild(link);
  1140→      
  1141→      // Revoke on next tick so the download has time to start
  1142→      setTimeout(() => URL.revokeObjectURL(url), 1000);
  1143→      
  1144→      if (IDEUtils) IDEUtils.showToast(`Downloaded ${activeFile.name}`, 'success');
  1145→    } catch (error) {
  1146→      console.error('File download failed:', error);
  1147→      if (IDEUtils) IDEUtils.showToast('Download failed: ' + (error.message || 'unknown error'), 'error');
  1148→    }
  1149→  }
  1150→
  1151→  /**
  1152→   * Export the entire current project as a ZIP archive (all files).
  1153→   * Used by the project-level export action — NOT by the per-file
  1154→   * download-file-btn in the editor header (which calls
  1155→   * handleDownloadCurrentFile above).
  1156→   */
  1157→  async function handleExportProject() {
  1158→    const project = IDEState?.get('currentProject');
  1159→    if (!project) {
  1160→      if (IDEUtils) IDEUtils.showToast('No project selected', 'warning');
  1161→      return;
  1162→    }
  1163→    
  1164→    try {
  1165→      if (IDEUtils) IDEUtils.showToast('Preparing export...', 'info');
  1166→      
  1167→      const files = await IDEStorage.Files.exportAsObject(project.name);
  1168→      
  1169→      if (window.JSZip) {
  1170→        const zip = new JSZip();
  1171→        
  1172→        Object.entries(files).forEach(([path, content]) => {
  1173→          zip.file(path, content);
  1174→        });
  1175→        
  1176→        const blob = await zip.generateAsync({ type: 'blob' });
  1177→        const url = URL.createObjectURL(blob);
  1178→        
  1179→        const link = document.createElement('a');
  1180→        link.href = url;

  1334→          IDEUtils.copyToClipboard(item.path);
  1335→        }
  1336→        break;
  1337→        
  1338→      case 'delete':
  1339→        deleteItem(item);
  1340→        break;
  1341→        
  1342→      case 'new-file':
  1343→        handleNewFile();
  1344→        break;
  1345→

  1415→  /**
  1416→   * Update context window indicator in toolbar
  1417→   */
  1418→  function updateContextWindowIndicator() {
  1419→    if (!window.LLMManagerComponent) return;
  1420→    
  1421→    let usage;
  1422→    try {
  1423→      usage = LLMManagerComponent.getContextUsage();
  1424→    } catch (e) {
  1425→      usage = null;
  1426→    }
  1427→    
  1428→    // Defensive: ensure usage is a valid object with numeric percentage
  1429→    const safePct = (usage && typeof usage.percentage === 'number' && !isNaN(usage.percentage)) 
  1430→      ? Math.min(100, Math.max(0, usage.percentage)) : 0;
  1431→    const safeTotal = (usage && typeof usage.total === 'number' && !isNaN(usage.total)) 
  1432→      ? usage.total : 0;
  1433→    const safeUsed = (usage && typeof usage.used === 'number' && !isNaN(usage.used)) 
  1434→      ? usage.used : 0;
  1435→    const safeMsgCount = (usage && typeof usage.messageCount === 'number') 
  1436→      ? usage.messageCount : 0;
  1437→    
  1438→    // Update fill bar
  1439→    const fill = document.getElementById('context-window-fill');
  1440→    if (fill) {
  1441→      fill.style.width = `${safePct}%`;
  1442→      
  1443→      // Color based on usage level
  1444→      fill.classList.remove('warning', 'danger');
  1445→      if (safePct > 80) {
  1446→        fill.classList.add('danger');
  1447→      } else if (safePct > 60) {
  1448→        fill.classList.add('warning');
  1449→      }
  1450→    }
  1451→    
  1452→    // Update text - show informative label when no/low usage
  1453→    const text = document.getElementById('context-window-text');
  1454→    if (text) {
  1455→      const provider = LLMManagerComponent.getCurrentProvider();
  1456→      
  1457→      if (!provider) {
  1458→        text.textContent = '—';
  1459→        text.style.color = 'var(--color-text-muted, #9ca3af)';
  1460→      } else if (safePct === 0 && safeMsgCount === 0) {
  1461→        text.textContent = 'Ready';
  1462→        text.style.color = 'var(--color-text-tertiary, #9ca3af)';
  1463→      } else {
  1464→        text.textContent = `${safePct}%`;
  1465→        text.style.color = '';
  1466→      }
  1467→    }
  1468→    
  1469→    // Update title with more info
  1470→    const indicator = document.getElementById('context-window-indicator');
  1471→    if (indicator) {
  1472→      const provider = LLMManagerComponent.getCurrentProvider();
  1473→      if (!provider) {
  1474→        indicator.title = 'No LLM provider selected\nClick to choose a provider';
  1475→      } else if (safePct === 0 && safeMsgCount === 0) {
  1476→        indicator.title = `Context window ready (${formatNumber(safeTotal)} tokens available)\nClick to switch provider`;
  1477→      } else {
  1478→        indicator.title = `Context: ${formatNumber(safeUsed)} / ${formatNumber(safeTotal)} tokens (${safeMsgCount} messages)\nClick to switch provider`;
  1479→      }
  1480→    }
  1481→  }

  1569→  // ============================================

  1571→  /**
  1572→   * Setup listeners for progressive section loading
  1573→   * Hides skeletons as each section's module loads (from index.js events)
  1574→   * This provides visual feedback that sections are becoming ready
  1575→   */
  1576→  function setupProgressiveReveal() {
  1577→    // Track which sections have been revealed
  1578→    const revealedSections = new Set();
  1579→    
  1580→    /**
  1581→     * Hide a specific skeleton with smooth transition
  1582→     * @param {string} sectionId - The ID of the skeleton element
  1583→     */
  1584→    function revealSection(sectionId) {
  1585→      if (revealedSections.has(sectionId)) return;
  1586→      revealedSections.add(sectionId);
  1587→      
  1588→      const skeleton = document.getElementById(sectionId);
  1589→      if (skeleton) {
  1590→        // Add fade-out class (defined in CSS)
  1591→        skeleton.classList.add('fade-out');
  1592→        
  1593→        // Remove from DOM after transition completes
  1594→        setTimeout(() => {
  1595→          if (skeleton.parentNode) {
  1596→            skeleton.style.display = 'none';
  1597→          }
  1598→        }, 320);
  1599→        
  1600→        console.log(`[SBIDE] 🎬 Revealed section: ${sectionId}`);
  1601→      }
  1602→    }
  1603→    
  1604→    // Listen for section-ready events from barrel loader (index.js)
  1605→    window.addEventListener('ide:section-ready', (event) => {
  1606→      const { section, module, elapsed } = event.detail || {};
  1607→      
  1608→      // Map section names to their skeleton IDs
  1609→      const sectionToSkeleton = {
  1610→        'sidebar': 'sidebar-skeleton',
  1611→        'chat': 'chat-skeleton',
  1612→        'editor': 'editor-skeleton'
  1613→      };
  1614→      
  1615→      const skeletonId = sectionToSkeleton[section];
  1616→      if (skeletonId) {
  1617→        revealSection(skeletonId);
  1618→      }
  1619→    });
  1620→    
  1621→    // Also listen for modules-ready as fallback (reveal all remaining)
  1622→    window.addEventListener('ide:modules-ready', () => {
  1623→      // Reveal any skeletons that haven't been revealed yet
  1624→      ['sidebar-skeleton', 'chat-skeleton', 'editor-skeleton'].forEach(id => {
  1625→        revealSection(id);
  1626→      });
  1627→    });
  1628→    
  1629→    console.log('[SBIDE] Progressive reveal setup complete');
  1630→  }
