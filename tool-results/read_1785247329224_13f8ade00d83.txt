   278→  function loadSavedState() {
   279→    try {
   280→      const saved = localStorage.getItem('llm-manager-state');
   281→      if (saved) {
   282→        const parsed = JSON.parse(saved);
   283→        
   284→        // Restore selected provider
   285→        if (parsed.currentProviderId) {
   286→          const provider = PROVIDERS.find(p => p.id === parsed.currentProviderId);
   287→          if (provider) currentProvider = provider;
   288→        }
   289→        
   290→        // Restore API keys (basic obfuscation - not true encryption but better than plaintext)
   291→        if (parsed.providerApiKeys) {
   292→          providerApiKeys = {};
   293→          for (const [key, value] of Object.entries(parsed.providerApiKeys)) {
   294→            try {
   295→              // Attempt to decode - if it fails, store as-is (legacy format)
   296→              providerApiKeys[key] = atob(value) || value;
   297→            } catch {
   298→              providerApiKeys[key] = value; // Legacy plaintext key
   299→            }
   300→          }
   301→        }
   302→        
   303→        // Restore rate limit state
   304→        if (parsed.rateLimitState) {
   305→          rateLimitState = parsed.rateLimitState;
   306→        }
   307→      }
   308→    } catch (e) {
   309→      console.error('Failed to load LLM manager state:', e);
   310→    }
   311→  }

   339→  function setupRateLimitDetection() {
   340→    // Use observer pattern instead of monkey-patching fetch (anti-pattern)
   341→    // We hook into the API client's request/response cycle instead
   342→    
   343→    // Listen for custom events from the API module when requests complete
   344→    document.addEventListener('api:requestCompleted', (e) => {
   345→      if (e.detail?.response) {
   346→        trackRequest(e.detail.response);
   347→      }
   348→    });
   349→    
   350→    // Also patch fetch but with safety checks and cleanup awareness
   351→    if (!window._originalFetch) {
   352→      window._originalFetch = window.fetch.bind(window);
   353→      window.fetch = async function(...args) {
   354→        try {
   355→          const response = await window._originalFetch(...args);
   356→          trackRequest(response);
   357→          return response;
   358→        } catch (error) {
   359→          throw error;
   360→        }
   361→      };
   362→    }
   363→  }

   365→  function trackRequest(response) {
   366→    // Get URL from response object
   367→    const url = response.url || '';
   368→    
   369→    // Only track chat-related requests
   370→    if (!url.includes('/api/chat') && !url.includes('/chat')) return;
   371→    
   372→    const now = Date.now();
   373→    const requestId = `${now}-${Math.random().toString(36).substring(2, 11)}`;
   374→    
   375→    requestHistory.push({
   376→      id: requestId,
   377→      timestamp: now,
   378→      status: response.status,
   379→      providerId: currentProvider?.id
   380→    });
   381→    
   382→    // Clean old entries (keep last hour) and cap at MAX_ENTRIES to prevent memory leaks
   383→    const oneHourAgo = now - 3600000;
   384→    requestHistory = requestHistory.filter(r => r.timestamp > oneHourAgo);
   385→    
   386→    // Safety cap: never keep more than 1000 entries (prevents memory leak under edge cases)
   387→    const MAX_REQUEST_HISTORY = 1000;
   388→    if (requestHistory.length > MAX_REQUEST_HISTORY) {
   389→      requestHistory = requestHistory.slice(-MAX_REQUEST_HISTORY);
   390→    }
   391→    
   392→    // Check for rate limit indicators
   393→    checkRateLimitIndicators(response);
   394→    
   395→    // Update context usage estimate
   396→    updateContextUsageEstimate();
   397→  }

   399→  function checkRateLimitIndicators(response) {
   400→    const providerId = currentProvider?.id;
   401→    
   402→    // HTTP 429 = Too Many Requests (explicit rate limit)
   403→    if (response.status === 429) {
   404→      handleRateLimitHit('explicit_429');
   405→      return;
   406→    }
   407→    
   408→    // Check for rate limit headers
   409→    const remainingRequests = response.headers.get('X-RateLimit-Remaining');
   410→    const retryAfter = response.headers.get('Retry-After');
   411→    const rateLimitReset = response.headers.get('X-RateLimit-Reset');
   412→    
   413→    if (remainingRequests !== null && parseInt(remainingRequests) <= 5) {
   414→      handleRateLimitHit('low_remaining', { remaining: parseInt(remainingRequests) });
   415→    }
   416→    
   417→    if (retryAfter) {
   418→      handleRateLimitHit('retry_after', { retryAfterSeconds: parseInt(retryAfter) });
   419→    }
   420→    
   421→    // Check for common rate limit error patterns in body (async)
   422→    detectRateLimitFromBody(response);
   423→  }

   425→  async function detectRateLimitFromBody(response) {
   426→    try {
   427→      // Only inspect the body for rate-limit signals when the HTTP status
   428→      // is plausibly a rate-limit response. A 404 (not found), 400 (bad
   429→      // request), 401 (unauthorized), 500 (server error) etc. are NOT
   430→      // rate limits — even if the body happens to contain the substring
   431→      // "429" (which the previous /429/i regex matched inside URLs, hex
   432→      // colors, and Next.js chunk hashes in 404 HTML pages, causing
   433→      // false-positive rate-limit toasts on a static deployment).
   434→      const status = response.status;
   435→      const isPlausibleRateLimitStatus =
   436→        status === 429 ||
   437→        status === 503 ||
   438→        status === 420 ||  // Enhance your calm (Twitter/Cloudflare)
   439→        status === 502 ||  // Bad gateway (may be rate-related)
   440→        status === 503;   // Service unavailable
   441→      if (!isPlausibleRateLimitStatus) {
   442→        return;
   443→      }
   444→
   445→      const clone = response.clone();
   446→      const text = await clone.text();
   447→
   448→      // Patterns are made specific (word-boundaried or phrasal) so they
   449→      // don't match arbitrary substrings inside HTML/JSON error bodies.
   450→      // The bare /429/i pattern is intentionally REMOVED — if the server
   451→      // wanted to signal a rate limit it would have returned HTTP 429
   452→      // (handled above) or one of the explicit phrases below.
   453→      const rateLimitPatterns = [
   454→        /\brate[- ]?limit\b/i,
   455→        /\btoo many requests\b/i,
   456→        /\bquota exceeded\b/i,
   457→        /\bthrottl(?:ed|ing)?\b/i,
   458→        /\btry again (?:later|in \d+ (?:second|minute)s?)\b/i,
   459→        /\brequest limit (?:exceeded|reached)\b/i
   460→      ];
   461→
   462→      const isRateLimited = rateLimitPatterns.some(pattern => pattern.test(text));
   463→
   464→      if (isRateLimited) {
   465→        handleRateLimitHit('error_pattern', { message: text.substring(0, 200) });
   466→      }
   467→    } catch (e) {
   468→      // Ignore body parsing errors
   469→    }
   470→  }

   472→  function handleRateLimitHit(reason, details = {}) {
   473→    const providerId = currentProvider?.id;
   474→    
   475→    // Update rate limit state
   476→    if (!rateLimitState[providerId]) {
   477→      rateLimitState[providerId] = {
   478→        hitCount: 0,
   479→        lastHit: null,
   480→        isCurrentlyLimited: false,
   481→        resetTime: null,
   482→        reasons: []
   483→      };
   484→    }
   485→    
   486→    const state = rateLimitState[providerId];
   487→    state.hitCount++;
   488→    state.lastHit = Date.now();
   489→    state.isCurrentlyLimited = true;
   490→    state.reasons.push({ reason, details, timestamp: Date.now() });
   491→    
   492→    // Estimate reset time (default 60 seconds if not specified)
   493→    const resetDelay = details.retryAfterSeconds ? details.retryAfterSeconds * 1000 : 60000;
   494→    state.resetTime = Date.now() + resetDelay;
   495→    
   496→    // Save state
   497→    saveState();
   498→    
   499→    // Show switcher modal after multiple hits
   500→    if (state.hitCount >= 2) {
   501→      showSwitcherModal({ reason: 'rate_limited', providerId, details });
   502→    }
   503→    
   504→    // Dispatch custom event for UI updates
   505→    window.dispatchEvent(new CustomEvent('llm:rateLimit', {
   506→      detail: { providerId, reason, details, state }
   507→    }));
   508→    
   509→    console.warn(`Rate limit hit for ${providerId}:`, reason, details);
   510→  }

   512→  function checkIfRateLimitExpired() {
   513→    const now = Date.now();
   514→    let changed = false;
   515→    
   516→    Object.keys(rateLimitState).forEach(providerId => {
   517→      const state = rateLimitState[providerId];
   518→      if (state.isCurrentlyLimited && state.resetTime && now > state.resetTime) {
   519→        state.isCurrentlyLimited = false;
   520→        changed = true;
   521→      }
   522→    });
   523→    
   524→    if (changed) {
   525→      saveState();
   526→      window.dispatchEvent(new CustomEvent('llm:rateLimitReset'));
   527→    }
   528→  }

   534→  function updateContextUsageEstimate() {
   535→    const messages = window.IDEState?.get('messages') || [];
   536→    
   537→    // Calculate approximate token count (rough: 1 token ≈ 4 chars for English)
   538→    let totalChars = 0;
   539→    const messageInfo = messages.map(m => ({
   540→      role: m.role,
   541→      contentLength: m.content?.length || 0,
   542→      timestamp: m.timestamp
   543→    }));
   544→    
   545→    totalChars = messageInfo.reduce((sum, m) => sum + m.contentLength, 0);
   546→    
   547→    // Add overhead for system prompt, formatting, etc.
   548→    totalChars += 1000; 
   549→    
   550→    // Convert to approximate tokens
   551→    const estimatedTokens = Math.ceil(totalChars / 4);
   552→    
   553→    contextUsage = {
   554→      used: estimatedTokens,
   555→      total: currentProvider?.contextWindow || 128000,
   556→      percentage: Math.min(100, Math.round((estimatedTokens / (currentProvider?.contextWindow || 128000)) * 100)),
   557→      messages: messageInfo,
   558→      messageCount: messages.length
   559→    };
   560→    
   561→    // Dispatch event for UI updates
   562→    window.dispatchEvent(new CustomEvent('llm:contextUpdate', {
   563→      detail: contextUsage
   564→    }));
   565→  }

   567→  function getContextUsage() {
   568→    // Pure getter — return the last computed context usage.
   569→    //
   570→    // IMPORTANT: do NOT call updateContextUsageEstimate() here. That function
   571→    // dispatches an 'llm:contextUpdate' event, and app.js listens for that
   572→    // event by calling updateContextWindowIndicator() → which calls
   573→    // getContextUsage() → which would re-dispatch → infinite recursion
   574→    // (Maximum call stack size exceeded).
   575→    //
   576→    // The estimate is refreshed by updateContextUsageEstimate() on its own
   577→    // schedule (called from trackRequest / setProvider / message events),
   578→    // so the cached value is always reasonably fresh. If we somehow have
   579→    // never computed it (fresh page load), compute it once here WITHOUT
   580→    // dispatching the event.
   581→    if (!contextUsage) {
   582→      try {
   583→        const msgs = (window.IDEState?.get?.('messages')) || [];
   584→        const totalChars = msgs.reduce((n, m) => n + ((m?.content || '').length), 0);
   585→        const total = currentProvider?.contextWindow || 128000;
   586→        const used = Math.ceil(totalChars / 4);
   587→        contextUsage = {
   588→          used,
   589→          total,
   590→          percentage: Math.min(100, Math.round((used / total) * 100)),
   591→          messages: [],
   592→          messageCount: msgs.length
   593→        };
   594→      } catch (e) {
   595→        contextUsage = { used: 0, total: 128000, percentage: 0, messages: [], messageCount: 0 };
   596→      }
   597→    }
   598→    return contextUsage;
   599→  }

   953→  function wireUpModalEvents() {

   958→    modalElement.querySelector('.modal-close-btn')?.addEventListener('click', closeSwitcherModal);
   959→

   961→    modalElement.addEventListener('click', (e) => {
   962→      if (e.target === modalElement) closeSwitcherModal();
   963→    });
   964→    
   965→    // Escape key
   966→    document.addEventListener('keydown', handleEscapeKey);
   967→    
   968→    // Provider selection buttons
   969→    modalElement.querySelectorAll('.select-provider-btn').forEach(btn => {
   970→      btn.addEventListener('click', () => {
   971→        const providerId = btn.dataset.providerId;
   972→        const providerCard = btn.closest('.llm-provider-card');
   973→        const keyInput = providerCard.querySelector('.api-key-input');
   974→        const apiKey = keyInput?.value?.trim();
   975→        
   976→        if (setProvider(providerId, apiKey)) {
   977→          closeSwitcherModal();
   978→        }
   979→      });
   980→    });
   981→    
   982→    // API key inputs - enable/disable select button + auto-save
   983→    modalElement.querySelectorAll('.api-key-input').forEach(input => {
   984→      input.addEventListener('input', () => {
   985→        const providerId = input.dataset.providerId;
   986→        const card = input.closest('.llm-provider-card');
   987→        const selectBtn = card.querySelector('.select-provider-btn');
   988→        
   989→        if (selectBtn) {
   990→          selectBtn.disabled = !input.value.trim();
   991→        }
   992→      });
   993→      
   994→      // Auto-save key on blur (when user leaves the input)
   995→      input.addEventListener('blur', () => {
   996→        const providerId = input.dataset.providerId;
   997→        const keyValue = input.value.trim();
   998→        
   999→        if (keyValue) {
  1000→          setApiKey(providerId, keyValue);
  1001→          const card = input.closest('.llm-provider-card');
  1002→          const provider = PROVIDERS.find(p => p.id === providerId);
  1003→          showSaveConfirmation(card, provider?.name || providerId);
  1004→          console.log(`✅ API key saved for ${providerId}`);
  1005→        }
  1006→      });
  1007→      
  1008→      // Save on Enter key
  1009→      input.addEventListener('keydown', (e) => {
  1010→        if (e.key === 'Enter') {
  1011→          e.preventDefault();
  1012→          input.blur(); // Trigger save via blur event
  1013→        }
  1014→      });
  1015→    });
  1016→
  1017→    // Edit key buttons - toggle visibility of input
  1018→    modalElement.querySelectorAll('.edit-key-btn').forEach(btn => {
  1019→      btn.addEventListener('click', (e) => {
  1020→        e.stopPropagation();
  1021→        const providerId = btn.dataset.providerId;
  1022→        const card = btn.closest('.llm-provider-card');
  1023→        const savedDisplay = card.querySelector('.saved-key-display');
  1024→        const keyInput = card.querySelector('.api-key-input');
  1025→        
  1026→        if (savedDisplay && keyInput) {
  1027→          savedDisplay.classList.add('hidden');
  1028→          keyInput.classList.remove('hidden');
  1029→          keyInput.focus();
  1030→        }
  1031→      });
  1032→    });
  1033→
  1034→    // Clear key buttons - remove saved key
  1035→    modalElement.querySelectorAll('.clear-key-btn').forEach(btn => {
  1036→      btn.addEventListener('click', (e) => {
  1037→        e.stopPropagation();

  1039→        const card = btn.closest('.llm-provider-card');
  1040→        const provider = PROVIDERS.find(p => p.id === providerId);
  1041→        
  1042→        // Confirm before clearing
  1043→        if (confirm(`Remove saved API key for ${provider?.name || providerId}?`)) {
  1044→          delete providerApiKeys[providerId];
  1045→          saveState();
  1046→          
  1047→          // Refresh the modal to show updated state
  1048→          closeSwitcherModal();
  1049→          showSwitcherModal();
  1050→          
  1051→          if (window.IDEUtils) {
  1052→            IDEUtils.showToast(`API key removed for ${provider?.name || providerId}`, 'info', 3000);
  1053→          }
  1054→        }
  1055→      });
  1056→    });
  1057→
  1058→    // Rate limited badge click - show details or alternative providers
  1059→    modalElement.querySelectorAll('.status-rate-limited').forEach(badge => {
  1060→      badge.addEventListener('click', (e) => {
  1061→        e.stopPropagation();

  1117→  function formatTimeUntil(timestamp) {
  1118→    if (!timestamp) return 'unknown time';
  1119→    const diff = timestamp - Date.now();
  1120→    if (diff <= 0) return 'now';
  1121→    
  1122→    const seconds = Math.floor(diff / 1000);
  1123→    if (seconds < 60) return `${seconds}s`;
  1124→    
  1125→    const minutes = Math.floor(seconds / 60);
  1126→    if (minutes < 60) return `${minutes}m`;
  1127→    
  1128→    const hours = Math.floor(minutes / 60);
  1129→    return `${hours}h`;
  1130→  }

  1132→  /**
  1133→   * Mask API key for display - show first 4 chars and last 4, rest as dots
  1134→   */
  1135→  function maskApiKey(key) {
  1136→    if (!key || key.length <= 8) return '••••••••';
  1137→    const first = key.substring(0, 4);
  1138→    const last = key.substring(key.length - 4);
  1139→    const dots = '•'.repeat(Math.min(key.length - 8, 12));
  1140→    return `${first}${dots}${last}`;

  1162→  /**
  1163→   * Start countdown timers for rate-limited providers
  1164→   * Auto-updates badges and reverts to green when limit expires
  1165→   */
  1166→  let countdownInterval = null;
  1167→  
  1168→  function startCountdownTimers() {
  1169→    // Clear any existing interval
  1170→    if (countdownInterval) {
  1171→      clearInterval(countdownInterval);
  1172→    }
  1173→    
  1174→    // Update every second
  1175→    countdownInterval = setInterval(() => {
  1176→      if (!modalElement) {
  1177→        clearInterval(countdownInterval);
  1178→        countdownInterval = null;
  1179→        return;
  1180→      }
  1181→      
  1182→      let needsRefresh = false;
  1183→      
  1184→      // Update all countdown displays
  1185→      modalElement.querySelectorAll('.status-countdown, .rate-limit-timer').forEach(el => {
  1186→        const resetTime = parseInt(el.dataset.resetTime);
  1187→        if (!resetTime) return;
  1188→        
  1189→        const diff = resetTime - Date.now();
  1190→        
  1191→        if (diff <= 0) {
  1192→          // Rate limit has expired - mark for refresh
  1193→          el.textContent = 'now';
  1194→          needsRefresh = true;
  1195→        } else {
  1196→          // Update display
  1197→          const seconds = Math.floor(diff / 1000);
  1198→          if (seconds < 60) {
  1199→            el.textContent = `${seconds}s`;
  1200→          } else {
  1201→            const minutes = Math.floor(seconds / 60);
  1202→            el.textContent = minutes < 60 ? `${minutes}m` : `${Math.floor(minutes / 60)}h`;
  1203→          }
  1204→        }
  1205→      });
  1206→      
  1207→      // Update progress bars
  1208→      modalElement.querySelectorAll('.rate-limit-bar').forEach(bar => {
  1209→        const timer = bar.closest('.rate-limit-expanded')?.querySelector('.rate-limit-timer');
  1210→        const resetTime = timer ? parseInt(timer.dataset.resetTime) : 0;
  1211→        if (!resetTime) return;
  1212→        
  1213→        const diff = resetTime - Date.now();
  1214→        // Assume rate limit duration was ~60 seconds, calculate remaining percentage
  1215→        const totalDuration = 60000; // Default 60 second window
  1216→        const elapsed = Math.max(0, totalDuration - diff);
  1217→        const percent = Math.min(100, Math.max(0, (elapsed / totalDuration) * 100));
  1218→        
  1219→        bar.style.width = `${100 - percent}%`;
  1220→      });
  1221→      
  1222→      // Check if any rate limits have expired
  1223→      Object.keys(rateLimitState).forEach(providerId => {
  1224→        const state = rateLimitState[providerId];
  1225→        if (state.isCurrentlyLimited && state.resetTime && Date.now() > state.resetTime) {
  1226→          state.isCurrentlyLimited = false;
  1227→          needsRefresh = true;
  1228→          console.log(`✅ Rate limit expired for ${providerId}`);
  1229→        }
  1230→      });
  1231→      
  1232→      // Refresh modal if needed to show updated badges
  1233→      if (needsRefresh) {
  1234→        saveState();
  1235→        // Dispatch event for external listeners
  1236→        window.dispatchEvent(new CustomEvent('llm:rateLimitReset'));
  1237→        
  1238→        // Refresh the modal content
  1239→        const isRateLimitedContext = Object.values(rateLimitState).some(s => s.isCurrentlyLimited);
  1240→        if (modalElement && !isRateLimitedContext) {
  1241→          // Only auto-refresh if no providers are still rate limited
  1242→          // This prevents flickering when multiple providers are limited
  1243→          const stillLimited = modalElement.querySelectorAll('.status-rate-limited').length > 0;
  1244→          if (!stillLimited) {
  1245→            closeSwitcherModal();
  1246→            showSwitcherModal();
  1247→            
  1248→            if (window.IDEUtils) {
  1249→              IDEUtils.showToast('Rate limit(s) expired - providers available again', 'success', 3000);
  1250→            }
  1251→          }
  1252→        }
  1253→      }
  1254→    }, 1000);
  1255→  }
