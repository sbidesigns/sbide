     1→/**
     2→ * SBIDE - Offline Kit
     3→ * ============================================
     4→ * Self-contained module that adds offline-first capabilities to the IDE.
     5→ * Loaded as a regular <script>; exposes window.OfflineKit.
     6→ *
     7→ * Capabilities (all progressive enhancement — no-op if unsupported):
     8→ *   1. OPFS         — persistent in-browser filesystem (survives reloads,
     9→ *                     works even when IndexedDB is blocked by extensions)
    10→ *   2. FS Access    — opt-in save-to-real-disk via showDirectoryPicker()
    11→ *   3. Sync queue   — offline mutations queued in OPFS, drained on 'online'
    12→ *   4. WebLLM       — local LLM in browser via WebGPU (no install)
    13→ *   5. Ollama       — local LLM via localhost:11434 (user installs Ollama)
    14→ *   6. LLM router   — picks best available: ollama → webllm → cloud → queue
    15→ *
    16→ * Public API:
    17→ *   OfflineKit.init()
    18→ *   await OfflineKit.write(project, path, content)   // writes to all layers
    19→ *   const text = await OfflineKit.read(project, path)
    20→ *   await OfflineKit.delete(project, path)
    21→ *   const files = await OfflineKit.list(project)
    22→ *   await OfflineKit.connectFolder()                 // user gesture required
    23→ *   OfflineKit.disconnectFolder()
    24→ *   OfflineKit.hasFolder()
    25→ *   const status = OfflineKit.getStatus()            // {storage, llm, queue}
    26→ *   OfflineKit.onStatusChange(cb)
    27→ *   await OfflineKit.chat(messages, opts)            // AsyncGenerator<string>
    28→ *   OfflineKit.setOllamaEndpoint(url)

   134→  /**
   135→   * Tests if CacheStorage API is functional for this origin.
   136→   * Returns: { healthy: boolean, error: string|null }
   137→   * 
   138→   * This is critical because corrupted CacheStorage causes 
   139→   * 'Failed to execute open on CacheStorage' errors that
   140→   * CANNOT be fixed by JavaScript alone — they require
   141→   * the user to clear site data via browser settings.
   142→   */
   143→  async function checkCacheStorageHealth() {
   144→    // Basic support check
   145→    if (!('caches' in window)) {
   146→      return { healthy: false, error: 'CacheStorage API not supported in this context' };
   147→    }
   148→    
   149→    try {
   150→      // Test 1: Can we call caches.keys()?
   151→      const names = await caches.keys();
   152→      
   153→      // Test 2: Can we open a cache? (This is where corruption usually shows up)
   154→      const testCache = await caches.open('__offlinekit_health_test__');
   155→      
   156→      // Test 3: Can we put/get a value?
   157→      await testCache.put('/__health_test__', new Response('ok'));
   158→      const response = await testCache.match('/__health_test__');
   159→      const ok = response && (await response.text()) === 'ok';
   160→      
   161→      // Cleanup test cache
   162→      await testCache.delete('/__health_test__');
   163→      await caches.delete('__offlinekit_health_test__');
   164→      
   165→      if (!ok) {
   166→        return { healthy: false, error: 'CacheStorage read/write test failed' };
   167→      }
   168→      
   169→      return { healthy: true, error: null };
   170→    } catch (e) {
   171→      const msg = (e.message || e.toString()).toLowerCase();
   172→      log('CacheStorage health check FAILED:', e.message);
   173→      
   174→      // Identify specific corruption patterns
   175→      if (msg.includes('cachestorage') && (msg.includes('internal error') || msg.includes('unexpected'))) {
   176→        return { healthy: false, error: 'CacheStorage internal error — database corrupted', isCorruption: true };
   177→      } else if (msg.includes('invalidstateerror') || msg.includes('state had changed')) {
   178→        return { healthy: false, error: 'InvalidStateError — OPFS/CacheStorage state corrupted', isCorruption: true };
   179→      } else if (msg.includes('security') || msg.includes('permission')) {
   180→        return { healthy: false, error: 'Permission denied — may be in private browsing or restricted context' };
   181→      }
   182→      
   183→      return { healthy: false, error: e.message?.substring(0, 100) || 'Unknown CacheStorage error', isCorruption: true };
   184→    }
   185→  }

   225→  // Storage Quota Pre-Flight Check
   226→  // ============================================
   227→  
   228→  /**
   229→   * Checks if browser has enough storage space for the model BEFORE downloading.
   230→   * Uses navigator.storage.estimate() to get available quota.
   231→   * 
   232→   * Returns: { 
   233→   *   ok: boolean, 
   234→   *   needed: string (e.g., "500 MB"), 
   235→   *   available: string (e.g., "1.2 GB"), 
   236→   *   availableBytes: number,
   237→   *   neededBytes: number,
   238→   *   error: string|null,
   239→   *   browser: string (for tailored instructions)
   240→   * }
   241→   */
   242→  async function checkStorageQuota(modelId) {
   243→    const neededBytes = getModelSize(modelId);
   244→    const neededMB = Math.ceil(neededBytes / (1024 * 1024));
   245→    const neededStr = neededMB >= 1024 ? `${(neededMB / 1024).toFixed(1)} GB` : `${neededMB} MB`;
   246→    
   247→    // Detect browser for tailored instructions
   248→    const isOpera = navigator.userAgent.includes('OPR') || navigator.userAgent.includes('Opera');
   249→    const isChrome = navigator.userAgent.includes('Chrome') && !isOpera;
   250→    const isEdge = navigator.userAgent.includes('Edg');
   251→    const isFirefox = navigator.userAgent.includes('Firefox');
   252→    let browser = 'chrome';
   253→    if (isOpera) browser = 'opera';
   254→    else if (isEdge) browser = 'edge';
   255→    else if (isFirefox) browser = 'firefox';
   256→    
   257→    // Default result — assume OK if we can't check
   258→    const defaultResult = { 
   259→      ok: true, 
   260→      needed: neededStr, 
   261→      available: 'unknown', 
   262→      availableBytes: Infinity,
   263→      neededBytes,
   264→      error: null,
   265→      browser 
   266→    };
   267→    
   268→    // Check if Storage API is available
   269→    if (!('storage' in navigator) || !navigator.storage || !navigator.storage.estimate) {
   270→      log('Storage API not available — skipping quota check');
   271→      return defaultResult;
   272→    }
   273→    
   274→    try {
   275→      const estimate = await navigator.storage.estimate();
   276→      const quota = estimate.quota || 0; // Total allowed
   277→      const usage = estimate.usage || 0; // Currently used
   278→      const availableBytes = quota - usage;
   279→      const availableMB = Math.floor(availableBytes / (1024 * 1024));
   280→      const availableStr = availableMB >= 1024 ? `${(availableMB / 1024).toFixed(1)} GB` : `${availableMB} MB`;
   281→      
   282→      log(`Storage quota: ${availableStr} available of ${neededStr} needed (used ${Math.round(usage / (1024*1024))}MB / ${Math.round(quota / (1024*1024))}MB total)`);
   283→      
   284→      // Need at least 20% headroom beyond model size (WebLLM needs working space)
   285→      const requiredWithHeadroom = neededBytes * 1.2;
   286→      
   287→      if (availableBytes < requiredWithHeadroom) {
   288→        return {
   289→          ok: false,
   290→          needed: neededStr,
   291→          available: availableStr,
   292→          availableBytes,
   293→          neededBytes,
   294→          error: `Insufficient storage: need ~${neededStr} but only ~${availableStr} available`,
   295→          browser
   296→        };
   297→      }
   298→      
   299→      return {
   300→        ok: true,
   301→        needed: neededStr,
   302→        available: availableStr,
   303→        availableBytes,
   304→        neededBytes,
   305→        error: null,
   306→        browser
   307→      };
   308→      
   309→    } catch (e) {
   310→      log('Storage estimate failed:', e.message);
   311→      // If estimate fails, allow download to proceed (error will be caught later)
   312→      return defaultResult;
   313→    }
   314→  }

   361→  // ============================================
   362→  // Smart Pre-Flight: Auto-detect & Clean Partial Downloads
   363→  // ============================================
   364→  
   365→  /**
   366→   * Detects and auto-cleans stale/partial WebLLM downloads before attempting
   367→   * a fresh load. This prevents CacheStorage corruption errors by proactively
   368→   * removing incomplete model data from previous interrupted sessions.
   369→   * 
   370→   * Detection methods:
   371→   * 1. Check for 'offlinekit:webllm-downloading' flag in sessionStorage
   372→   *     (set during download, cleared on success — if present on load = crashed)
   373→   * 2. Check if model is marked as cached but CacheStorage health check fails
   374→   * 3. Look for OPFS entries with model name that are suspiciously small
   375→   * 
   376→   * @param {string} modelId - The model about to be loaded
   377→   * @returns {Promise<{cleaned: boolean, reason: string|null, details: string[]}>}
   378→   */
   379→  async function detectAndCleanPartialDownload(modelId) {
   380→    const result = { 
   381→      cleaned: false, 
   382→      reason: null, 
   383→      details: [] 
   384→    };
   385→    
   386→    const modelKey = modelId.replace(/[^a-zA-Z0-9_-]/g, '_').toLowerCase();
   387→    log(`Pre-flight check for model: ${modelId} (key: ${modelKey})`);
   388→    
   389→    // === DETECTION 1: Stale download-in-progress flag ===
   390→    // This flag is set when download starts, cleared on success.
   391→    // If it's present on page load, the previous session crashed mid-download.
   392→    try {
   393→      const downloadingFlag = sessionStorage.getItem('offlinekit:webllm-downloading');
   394→      if (downloadingFlag) {
   395→        const flagData = JSON.parse(downloadingFlag);
   396→        // If flag is for THIS model or is older than 5 minutes = stale
   397→        const isStale = (flagData.modelId !== modelId) || 
   398→                        (Date.now() - flagData.startTime > 300000);
   399→        
   400→        if (isStale) {
   401→          result.cleaned = true;
   402→          result.reason = `Stale download flag found (model: ${flagData.modelId}, started ${Math.round((Date.now() - flagData.startTime) / 1000)}s ago)`;
   403→          result.details.push('Detected crashed/interrupted download session');
   404→          
   405→          // Clear the stale flag
   406→          sessionStorage.removeItem('offlinekit:webllm-downloading');
   407→          result.details.push('Cleared stale download flag');
   408→        } else {
   409→          // Download was recently started for this exact model — might be a quick page refresh
   410→          // Give it 30 seconds grace period before treating as stale
   411→          if (Date.now() - flagData.startTime > 30000) {
   412→            result.cleaned = true;
   413→            result.reason = 'Recent download appears stalled';
   414→            result.details.push('Download started >30s ago but no completion detected');
   415→          }
   416→        }
   417→      }
   418→    } catch(e) {
   419→      result.details.push(`Flag check error: ${e.message?.substring(0, 40)}`);
   420→    }
   421→    
   422→    // === DETECTION 2: Model marked as cached but cache is actually corrupted/empty ===
   423→    try {
   424→      const wasCached = state.webLLMCachedModels[modelId] || 
   425→                         localStorage.getItem('offlinekit:webllm-cached') === modelId;
   426→      
   427→      if (wasCached) {
   428→        // Quick sanity check: can we even open caches?
   429→        try {
   430→          const testNames = await caches.keys();
   431→          const hasWebllmCache = testNames.some(n => 
   432→            n.includes('webllm') || n.includes('mlc') || n.includes('workbox')
   433→          );
   434→          
   435→          if (!hasWebllmCache && wasCached) {
   436→            // Model claims to be cached but no WebLLM caches exist = data was lost
   437→            result.cleaned = true;
   438→            result.reason = 'Model marked as cached but no cache storage found';
   439→            result.details.push('Cache storage missing — clearing stale cache markers');
   440→            
   441→            // Clean up stale markers
   442→            delete state.webLLMCachedModels[modelId];
   443→            saveCachedModels();
   444→            if (localStorage.getItem('offlinekit:webllm-cached') === modelId) {
   445→              localStorage.removeItem('offlinekit:webllm-cached');
   446→            }
   447→            state.webLLMCached = false;
   448→            result.details.push('Removed stale cache markers');
   449→          }
   450→        } catch(cacheErr) {
   451→          // If we can't even list caches, they might be corrupted
   452→          if (cacheErr.message?.includes('cachestorage') || 
   453→              cacheErr.message?.includes('internal error')) {
   454→            result.cleaned = true;
   455→            result.reason = 'CacheStorage appears corrupted';
   456→            result.details.push(`CacheStorage error: ${cacheErr.message?.substring(0, 60)}`);
   457→          }
   458→        }
   459→      }
   460→    } catch(e) {
   461→      result.details.push(`Cache marker check error: ${e.message?.substring(0, 40)}`);
   462→    }
   463→    
   464→    // === DETECTION 3: Previous load error that wasn't resolved ===
   465→    try {
   466→      if (state.webLLMLoadError && !result.cleaned) {
   467→        const errorMsg = (state.webLLMLoadError || '').toLowerCase();
   468→        if (errorMsg.includes('cachestorage') || errorMsg.includes('corrupted') ||
   469→            errorMsg.includes('invalidstateerror')) {
   470→          // Previous attempt failed with corruption error — clean before retrying
   471→          result.cleaned = true;
   472→          result.reason = 'Previous load failed with cache corruption error';
   473→          result.details.push(`Previous error: ${state.webLLMLoadError?.substring(0, 80)}`);
   474→        }
   475→      }
   476→    } catch(e) {
   477→      result.details.push(`Error check error: ${e.message?.substring(0, 40)}`);
   478→    }
   479→    
   480→    // === IF ANY DETECTION TRIGGERED: AUTO-CLEAN THE MODEL'S CACHE ===
   481→    if (result.cleaned) {
   482→      log(`⚡ Pre-flight detected issue: ${result.reason}`);
   483→      log('Auto-cleaning model cache before fresh download...');
   484→      
   485→      // Use existing deleteCachedModel function for surgical cleanup
   486→      try {
   487→        const deleteResult = await deleteCachedModel(modelId);
   488→        if (deleteResult.warnings && deleteResult.warnings.length > 0) {
   489→          result.details.push(...deleteResult.warnings);
   490→        }
   491→        result.details.push('✅ Auto-clean completed — ready for fresh download');
   492→      } catch(cleanErr) {
   493→        result.details.push(`Auto-clean warning: ${cleanErr.message?.substring(0, 60)}`);
   494→        // Don't fail the whole operation — still try to load
   495→      }
   496→      
   497→      // Also clear the error state since we're attempting recovery
   498→      state.webLLMLoadError = null;
   499→    } else {
   500→      log('✅ Pre-flight check passed — no issues detected');
   501→    }
   502→    
   503→    return result;
   504→  }

   548→  async function deleteCachedModel(modelId) {
   549→    const errors = []; // Collect all errors for reporting
   550→    
   551→    try {
   552→      // Remove from state
   553→      delete state.webLLMCachedModels[modelId];
   554→      saveCachedModels();
   555→      
   556→      // Clear localStorage markers
   557→      if (localStorage.getItem('offlinekit:webllm-cached') === modelId) {
   558→        localStorage.removeItem('offlinekit:webllm-cached');
   559→      }
   560→      state.webLLMCached = false;
   561→      
   562→      log('Clearing WebLLM cache for model:', modelId);
   563→      
   564→      // === METHOD 1: Clear browser Cache API (where WebLLM stores model weights) ===
   565→      try {
   566→        const cacheNames = await caches.keys();
   567→        for (const name of cacheNames) {
   568→          // WebLLM uses caches with names like 'webllm-', 'mlc-', or workbox caches
   569→          if (name.includes('webllm') || name.includes('mlc') || name.includes('workbox') || name.includes('cache')) {
   570→            await caches.delete(name);
   571→            log('Deleted CacheStorage:', name);
   572→          }
   573→        }
   574→      } catch (e) {
   575→        errors.push(`Cache API: ${e.message}`);
   576→        warn('Cache API cleanup failed:', e.message);
   577→      }
   578→      
   579→      // === METHOD 2: Try WebLLM's own cleanup ===
   580→      try {
   581→        const webllmModule = await import('https://cdn.jsdelivr.net/npm/@mlc-ai/web-llm@0.2.9/+esm');
   582→        if (webllmModule && typeof webllmModule.clearCache === 'function') {
   583→          await webllmModule.clearCache();
   584→          log('Cleared via WebLLM.clearCache()');
   585→        }
   586→      } catch (e) {
   587→        // This often fails due to CacheStorage corruption - that's ok, we tried other methods
   588→        log('WebLLM.clearCache() not available or failed:', e.message?.substring(0, 50));
   589→      }
   590→      
   591→      // === METHOD 3: Clear OPFS directory where WebLLM stores models ===
   592→      if (navigator.storage && navigator.storage.getDirectory) {
   593→        try {
   594→          const root = await navigator.storage.getDirectory();
   595→          await clearOPFSRecursively(root, ['webllm', 'cache', 'models', modelId.replace(/[^a-zA-Z0-9_-]/g, '_')]);
   596→          log('Attempted OPFS cleanup');
   597→        } catch (e) {
   598→          errors.push(`OPFS: ${e.message}`);
   599→          warn('OPFS cleanup failed:', e.message);
   600→          
   601→          // If OPFS is corrupted, try nuking everything we can access
   602→          if (e.message.includes('InvalidStateError') || e.message.includes('state had changed')) {
   603→            log('OPFS appears corrupted, attempting aggressive cleanup...');
   604→            try {
   605→              const root = await navigator.storage.getDirectory();
   606→              await nukeOPFSAggressive(root);
   607→              log('Aggressive OPFS cleanup attempted');
   608→            } catch (e2) {
   609→              errors.push(`Aggressive OPFS: ${e2.message}`);
   610→            }
   611→          }
   612→        }
   613→      }
   614→      
   615→      // === METHOD 4: Clear all IndexedDB databases ===
   616→      try {
   617→        const dbs = await indexedDB.databases();
   618→        for (const db of dbs) {
   619→          if (db.name && (db.name.includes('webllm') || db.name.includes('cache') || 
   620→              db.name.includes('model') || db.name.includes('mlc'))) {
   621→            indexedDB.deleteDatabase(db.name);
   622→            log('Deleted IndexedDB:', db.name);
   623→          }
   624→        }
   625→      } catch (e) {
   626→        errors.push(`IndexedDB: ${e.message}`);
   627→        warn('IndexedDB cleanup failed:', e.message);
   628→      }
   629→      
   630→      // === METHOD 5: Clear sessionStorage (WebLLM might use it) ===
   631→      try {
   632→        const keysToRemove = [];
   633→        for (let i = 0; i < sessionStorage.length; i++) {
   634→          const key = sessionStorage.key(i);
   635→          if (key && (key.includes('webllm') || key.includes('mlc') || key.includes('model'))) {
   636→            keysToRemove.push(key);
   637→          }
   638→        }
   639→        keysToRemove.forEach(key => sessionStorage.removeItem(key));
   640→        if (keysToRemove.length > 0) log('Cleared sessionStorage entries:', keysToRemove.length);
   641→      } catch (e) {
   642→        warn('SessionStorage cleanup failed:', e.message);
   643→      }
   644→      
   645→      notifyStatusChange();
   646→      renderBadge();
   647→      
   648→      if (errors.length > 0) {
   649→        log('Cache cleared WITH some errors:', errors);
   650→        return { success: true, warnings: errors };
   651→      }
   652→      
   653→      log('Cleared model cache for:', modelId);
   654→      return { success: true };
   655→    } catch (e) {
   656→      err('Failed to delete model cache:', e);
   657→      return { success: false, error: e.message };
   658→    }
   659→  }

   661→  /**
   662→   * Aggressive OPFS cleanup - tries to remove EVERYTHING when normal methods fail
   663→   * Use only when OPFS is in a corrupted state
   664→   */
   665→  async function nukeOPFSAggressive(rootDir) {
   666→    const removed = [];
   667→    const failed = [];
   668→    
   669→    try {
   670→      for await (const entry of rootDir.values()) {
   671→        try {
   672→          await rootDir.removeEntry(entry.name, { recursive: true });
   673→          removed.push(entry.name);
   674→        } catch (e) {
   675→          failed.push(entry.name);
   676→        }
   677→      }
   678→      log(`Aggressive OPFS: removed ${removed.length}, failed ${failed.length}`);
   679→      if (failed.length > 0) {
   680→        warn('Could not remove these OPFS entries:', failed);
   681→      }
   682→    } catch (e) {
   683→      warn('Aggressive OPFS iteration failed:', e.message);
   684→    }
   685→  }

   687→  async function clearOPFSRecursively(rootDir, dirNamesToRemove) {
   688→    for (const dirName of dirNamesToRemove) {
   689→      try {
   690→        const entry = await rootDir.removeEntry(dirName, { recursive: true });
   691→        log('Removed OPFS entry:', dirName);
   692→      } catch (e) {
   693→        // Entry might not exist, that's ok
   694→      }
   695→    }
   696→    
   697→    // Also try removing any files/dirs at root level that look like model caches
   698→    try {
   699→      for await (const entry of rootDir.values()) {
   700→        if (entry.kind === 'directory' && (
   701→          entry.name.startsWith('webllm') || 
   702→          entry.name.includes('cache') ||
   703→          entry.name.includes('model') ||
   704→          entry.name.includes('mlc')
   705→        )) {
   706→          try {
   707→            await rootDir.removeEntry(entry.name, { recursive: true });
   708→            log('Removed OPFS dir:', entry.name);
   709→          } catch (e) {}
   710→        }
   711→      }
   712→    } catch (e) {}
   713→  }

   786→  function formatTimeAgo(date) {
   787→    const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
   788→    if (seconds < 60) return 'just now';
   789→    if (seconds < 3600) return Math.floor(seconds / 60) + 'm ago';
   790→    if (seconds < 86400) return Math.floor(seconds / 3600) + 'h ago';
   791→    return Math.floor(seconds / 86400) + 'd ago';
   792→  }

   794→  function formatSize(bytes) {
   795→    if (bytes >= 1024 * 1024 * 1024) return (bytes / (1024 * 1024 * 1024)).toFixed(1) + ' GB';
   796→    if (bytes >= 1024 * 1024) return (bytes / (1024 * 1024)).toFixed(0) + ' MB';
   797→    if (bytes >= 1024) return (bytes / 1024).toFixed(0) + ' kB';
   798→    return bytes + ' B';
   799→  }

   921→  async function opfsWrite(project, path, content) {
   922→    if (!state.opfsAvailable) return false;
   923→    const projDir = await getProjectDir(project, true);
   924→    if (!projDir) return false;
   925→    try {
   926→      // Walk/create intermediate dirs
   927→      const parts = path.split('/').filter(Boolean);
   928→      let dir = projDir;
   929→      for (let i = 0; i < parts.length - 1; i++) {
   930→        dir = await dir.getDirectoryHandle(parts[i], { create: true });
   931→      }
   932→      const fileName = parts[parts.length - 1] || 'untitled';
   933→      const fileHandle = await dir.getFileHandle(fileName, { create: true });
   934→      const writable = await fileHandle.createWritable();
   935→      await writable.write(content);
   936→      await writable.close();
   937→      return true;
   938→    } catch (e) {
   939→      warn('OPFS write failed:', e);
   940→      return false;
   941→    }

   981→  async function opfsList(project) {
   982→    if (!state.opfsAvailable) return [];
   983→    const projDir = await getProjectDir(project, false);
   984→    if (!projDir) return [];
   985→    const results = [];
   986→    async function walk(dir, prefix) {
   987→      for await (const [name, handle] of dir.entries()) {
   988→        const fullPath = prefix ? `${prefix}/${name}` : name;
   989→        if (handle.kind === 'file') {
   990→          results.push({ path: fullPath, kind: 'file' });
   991→        } else {
   992→          results.push({ path: fullPath, kind: 'directory' });
   993→          await walk(handle, fullPath);
   994→        }
   995→      }
   996→    }
   997→    await walk(projDir, '');
   998→    return results;
   999→  }

  1008→  async function connectFolder() {
  1009→    if (!fsAccessSupported()) {
  1010→      warn('File System Access API not supported. Use Chrome/Edge.');
  1011→      return false;
  1012→    }
  1013→    try {
  1014→      const handle = await window.showDirectoryPicker({ mode: 'readwrite' });
  1015→      state.fsHandle = handle;
  1016→      // Persist handle to OPFS so next session can re-request permission silently
  1017→      if (state.opfsAvailable) {
  1018→        try {
  1019→          const metaDir = await state.opfsRoot.getDirectoryHandle('.offline-kit', { create: true });
  1020→          const handleFile = await metaDir.getFileHandle('fs-handle', { create: true });
  1021→          const writable = await handleFile.createWritable();
  1022→          // Note: structuredClone is the standard way, but handle serialization
  1023→          // requires the File System Access API serialization support.
  1024→          // We store a marker; actual handle re-acquisition requires user gesture
  1025→          // in some browsers. For now, just remember we had one.
  1026→          await writable.write(JSON.stringify({ connected: true, name: handle.name }));
  1027→          await writable.close();
  1028→        } catch (e) { /* non-fatal */ }
  1029→      }
  1030→      log('Connected folder:', handle.name);
  1031→      notifyStatusChange();
  1032→      return true;
  1033→    } catch (e) {
  1034→      if (e.name === 'AbortError') return false;
  1035→      warn('connectFolder failed:', e);
  1036→      return false;
  1037→    }
  1038→  }

  1047→  async function ensureFsPermission() {
  1048→    if (!state.fsHandle) return false;
  1049→    const perm = await state.fsHandle.queryPermission({ mode: 'readwrite' });
  1050→    if (perm === 'granted') return true;
  1051→    const requested = await state.fsHandle.requestPermission({ mode: 'readwrite' });
  1052→    return requested === 'granted';
  1053→  }
  1054→
  1055→  async function fsWrite(path, content) {
  1056→    if (!state.fsHandle) return false;
  1057→    if (!(await ensureFsPermission())) return false;
  1058→    try {
  1059→      const parts = path.split('/').filter(Boolean);
  1060→      let dir = state.fsHandle;
  1061→      for (let i = 0; i < parts.length - 1; i++) {
  1062→        dir = await dir.getDirectoryHandle(parts[i], { create: true });
  1063→      }
  1064→      const fileName = parts[parts.length - 1] || 'untitled';
  1065→      const fileHandle = await dir.getFileHandle(fileName, { create: true });
  1066→      const writable = await fileHandle.createWritable();
  1067→      await writable.write(content);
  1068→      await writable.close();
  1069→      return true;
  1070→    } catch (e) {
  1071→      warn('FS write failed:', e);
  1072→      return false;
  1073→    }
  1074→  }
  1075→
  1076→  async function fsRead(path) {
  1077→    if (!state.fsHandle) return null;
  1078→    if (!(await ensureFsPermission())) return null;
  1079→    try {
  1080→      const parts = path.split('/').filter(Boolean);
  1081→      let dir = state.fsHandle;
  1082→      for (let i = 0; i < parts.length - 1; i++) {
  1083→        dir = await dir.getDirectoryHandle(parts[i], { create: false });
  1084→      }
  1085→      const fileName = parts[parts.length - 1] || 'untitled';
  1086→      const fileHandle = await dir.getFileHandle(fileName, { create: false });
  1087→      const file = await fileHandle.getFile();
  1088→      return await file.text();
  1089→    } catch (e) {
  1090→      return null;
  1091→    }
  1092→  }

  1097→  async function write(project, path, content) {
  1098→    const tasks = [];
  1099→    if (state.opfsAvailable) tasks.push(opfsWrite(project, path, content));
  1100→    if (state.fsHandle) tasks.push(fsWrite(`${project}/${path}`, content));
  1101→    await Promise.all(tasks);
  1102→    // Enqueue for server sync (if a sync endpoint is set)
  1103→    enqueue({ op: 'write', project, path, content_hash: await hashContent(content), ts: Date.now() });
  1104→  }
  1105→
  1106→  async function read(project, path) {
  1107→    // Try OPFS first (faster), fall back to FS, fall back to null
  1108→    let text = await opfsRead(project, path);
  1109→    if (text !== null) return text;
  1110→    text = await fsRead(`${project}/${path}`);
  1111→    return text;
  1112→  }

  1123→  async function hashContent(content) {
  1124→    try {
  1125→      const data = new TextEncoder().encode(content);
  1126→      const buf = await crypto.subtle.digest('SHA-256', data);
  1127→      return Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2, '0')).join('');
  1128→    } catch (e) { return ''; }
  1129→  }

  1134→  async function initQueue() {
  1135→    // Load any persisted queue from OPFS
  1136→    if (!state.opfsAvailable) return;
  1137→    try {
  1138→      const metaDir = await state.opfsRoot.getDirectoryHandle('.offline-kit', { create: true });
  1139→      const handle = await metaDir.getFileHandle('sync-queue.json', { create: true });
  1140→      const file = await handle.getFile();
  1141→      const text = await file.text();
  1142→      if (text) {
  1143→        state.queue = JSON.parse(text);
  1144→        log(`Loaded ${state.queue.length} queued ops from OPFS`);
  1145→      }
  1146→    } catch (e) { /* non-fatal */ }
  1147→  }
  1148→
  1149→  async function persistQueue() {
  1150→    if (!state.opfsAvailable) return;
  1151→    try {
  1152→      const metaDir = await state.opfsRoot.getDirectoryHandle('.offline-kit', { create: true });
  1153→      const handle = await metaDir.getFileHandle('sync-queue.json', { create: true });
  1154→      const writable = await handle.createWritable();
  1155→      await writable.write(JSON.stringify(state.queue));
  1156→      await writable.close();
  1157→    } catch (e) { /* non-fatal */ }
  1158→  }

  1160→  function enqueue(op) {
  1161→    state.queue.push(op);
  1162→    persistQueue();
  1163→    notifyStatusChange();
  1164→    if (state.online) {
  1165→      drainQueue(); // best-effort, non-blocking
  1166→    }
  1167→  }
  1168→
  1169→  async function drainQueue() {
  1170→    if (state.queueDraining || !state.syncEndpoint) return;
  1171→    if (state.queue.length === 0) return;
  1172→    state.queueDraining = true;
  1173→    try {
  1174→      while (state.queue.length > 0 && state.online) {
  1175→        const op = state.queue[0];
  1176→        try {
  1177→          const res = await fetch(state.syncEndpoint, {
  1178→            method: 'POST',
  1179→            headers: { 'Content-Type': 'application/json' },
  1180→            body: JSON.stringify(op),
  1181→          });
  1182→          if (res.ok) {
  1183→            state.queue.shift();
  1184→          } else {
  1185→            break; // server error — stop, retry later
  1186→          }
  1187→        } catch (e) {
  1188→          break; // network error — stop, retry later
  1189→        }
  1190→      }
  1191→      await persistQueue();
  1192→      notifyStatusChange();
  1193→    } finally {
  1194→      state.queueDraining = false;
  1195→    }
  1196→  }

  1198→  function setupOnlineListeners() {

  1287→  async function probeOllama() {
  1288→    // Skip Ollama probe in iframe contexts (localhost won't work due to CORS)
  1289→    // Also skip if we've recently failed (cache negative result for 5 minutes)
  1290→    try {
  1291→      const lastFail = sessionStorage.getItem('offlinekit:ollama-fail-time');
  1292→      if (lastFail && (Date.now() - parseInt(lastFail, 10)) < 300000) {
  1293→        state.ollamaReachable = false;
  1294→        return; // Recently failed, skip probe
  1295→      }
  1296→      
  1297→      // Detect iframe context where localhost is inaccessible
  1298→      if (window.self !== window.top) {
  1299→        state.ollamaReachable = false;
  1300→        return; // In iframe, localhost won't work
  1301→      }
  1302→    } catch (e) {
  1303→      // sessionStorage may be blocked
  1304→    }
  1305→    
  1306→    try {
  1307→      const ctrl = new AbortController();
  1308→      // Reduced timeout: 300ms (was 800ms) for faster startup
  1309→      const t = setTimeout(() => ctrl.abort(), 300);
  1310→      const res = await fetch(`${state.ollamaEndpoint}/api/tags`, { signal: ctrl.signal });
  1311→      clearTimeout(t);
  1312→      if (res.ok) {
  1313→        const data = await res.json();
  1314→        state.ollamaReachable = true;
  1315→        state.ollamaModels = (data.models || []).map(m => m.name);
  1316→        log('Ollama reachable. Models:', state.ollamaModels);
  1317→      } else {
  1318→        state.ollamaReachable = false;
  1319→        cacheOllamaFailure();
  1320→      }
  1321→    } catch (e) {
  1322→      state.ollamaReachable = false;
  1323→      cacheOllamaFailure();
  1324→    }
  1325→    notifyStatusChange();
  1326→  }

  1339→  async function* ollamaChat(messages, opts = {}) {
  1340→    const model = opts.model || (state.ollamaModels[0]) || 'llama3.1:8b';
  1341→    const res = await fetch(`${state.ollamaEndpoint}/v1/chat/completions`, {
  1342→      method: 'POST',
  1343→      headers: { 'Content-Type': 'application/json' },
  1344→      body: JSON.stringify({
  1345→        model,
  1346→        messages,
  1347→        stream: true,
  1348→        temperature: opts.temperature ?? 0.7,
  1349→      }),
  1350→    });
  1351→    if (!res.ok) throw new Error(`Ollama error ${res.status}: ${await res.text()}`);
  1352→    const reader = res.body.getReader();
  1353→    const decoder = new TextDecoder();
  1354→    let buffer = '';
  1355→    while (true) {
  1356→      const { done, value } = await reader.read();
  1357→      if (done) break;
  1358→      buffer += decoder.decode(value, { stream: true });
  1359→      const lines = buffer.split('\n');
  1360→      buffer = lines.pop();
  1361→      for (const line of lines) {
  1362→        if (!line.startsWith('data: ')) continue;
  1363→        const data = line.slice(6).trim();
  1364→        if (data === '[DONE]') return;
  1365→        try {
  1366→          const json = JSON.parse(data);
  1367→          const delta = json.choices?.[0]?.delta?.content || '';
  1368→          if (delta) yield delta;
  1369→        } catch (e) { /* skip */ }
  1370→      }
  1371→    }
  1372→  }

  1425→  async function pickLLM() {
  1426→    // 1. Ollama if reachable
  1427→    if (state.ollamaReachable) {
  1428→      state.activeLLM = { type: 'ollama', name: 'Ollama', model: state.ollamaModels[0] || 'llama3.1' };
  1429→      return 'ollama';
  1430→    }
  1431→    // 2. WebLLM if supported and engine is loaded (or loadable)
  1432→    if (state.webLLMSupported) {
  1433→      state.activeLLM = { type: 'webllm', name: 'WebLLM', model: state.webLLMModelId };
  1434→      return 'webllm';
  1435→    }
  1436→    // 3. Cloud if configured and online
  1437→    if (state.cloudLLM && state.online) {
  1438→      state.activeLLM = { type: 'cloud', name: 'Cloud', model: state.cloudLLM.model };
  1439→      return 'cloud';
  1440→    }
  1441→    // 4. Queue for later
  1442→    state.activeLLM = { type: state.online ? 'none' : 'queued', name: state.online ? 'None' : 'Queued', model: null };
  1443→    return state.activeLLM.type;
  1444→  }

  1446→  async function* chat(messages, opts = {}) {
  1447→    const backend = await pickLLM();
  1448→    if (backend === 'ollama') {
  1449→      try { yield* ollamaChat(messages, opts); return; }
  1450→      catch (e) { warn('Ollama failed, falling through:', e.message); }
  1451→    }
  1452→    if (backend === 'webllm' || state.webLLMSupported) {
  1453→      try { yield* webLLMChat(messages, opts); return; }
  1454→      catch (e) { warn('WebLLM failed, falling through:', e.message); }
  1455→    }
  1456→    if (backend === 'cloud' || (state.cloudLLM && state.online)) {
  1457→      try { yield* cloudLLMChat(messages, opts); return; }
  1458→      catch (e) { warn('Cloud failed:', e.message); }
  1459→    }
  1460→    // 4. Queue
  1461→    yield '(queued — will respond when a backend becomes available)';
  1462→    enqueue({ op: 'chat', messages, opts, ts: Date.now() });
  1463→  }
