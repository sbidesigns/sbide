     1→/**
     2→ * SBIDE - Code Editor Component
     3→ * Handles file viewing, editing, and preview modes with tab management
     4→ */
     5→
     6→const CodeEditorComponent = (() => {
     7→  // ============================================
     8→  // State
     9→  // ============================================
    10→  
    11→  let container = null;
    12→  let tabsContainer = null;
    13→  let editorContainer = null;
    14→  
    15→  // Open tabs (array of file objects)
    16→  let openTabs = [];
    17→  let activeTabPath = null;
    18→  
    19→  // Current mode: 'view' | 'edit' | 'preview'
    20→  let currentMode = 'view';
    21→  
    22→  // Current file content
    23→  let currentContent = '';
    24→  let originalContent = ''; // For tracking changes
    25→  
    26→  // Auto-save debounce
    27→  let autoSaveTimeout = null;
    28→  const AUTO_SAVE_DELAY = 1000; // ms
    29→  
    30→  // Editor preferences (ported from v1 IDE)
    31→  let editorPrefs = { wordWrap: true, lineNumbers: true, autoSave: false, soundEffects: false };
    32→  
    33→  // Load editor prefs from state on first opportunity
    34→  function refreshEditorPrefs() {
    35→    if (window.IDEState) {
    36→      const s = IDEState.get('settings');
    37→      if (s) {
    38→        editorPrefs.wordWrap = s.wordWrap ?? true;
    39→        editorPrefs.lineNumbers = s.lineNumbers ?? true;
    40→        editorPrefs.autoSave = s.autoSave ?? false;
    41→        editorPrefs.soundEffects = s.soundEffects ?? false;
    42→      }
    43→    }
    44→  }
    45→  
    46→  // Listen for settings changes (dispatched by settings-panel.js)
    47→  document.addEventListener('ide:settings-applied', (e) => {
    48→    const prev = editorPrefs;
    49→    refreshEditorPrefs();
    50→    // Re-render edit mode if wrap or line-numbers changed and we're currently editing
    51→    if ((prev.wordWrap !== editorPrefs.wordWrap || prev.lineNumbers !== editorPrefs.lineNumbers)
    52→        && currentMode === 'edit') {
    53→      displayContent(currentContent);
    54→    }
    55→  });
    56→  
    57→  // Initial fetch
    58→  refreshEditorPrefs();
    59→  
    60→  // Callbacks
    61→  let onContentChange = null;
    62→  let onFileSave = null;
    63→
    64→  // ============================================
    65→  // Initialization
    66→  // ============================================
    67→  
    68→  /**
    69→   * Initialize code editor component
    70→   * @param {HTMLElement} element - Main container for the editor
    71→   * @param {Object} options - Configuration options
    72→   */
    73→  function init(element, options = {}) {
    74→    if (!element) {
    75→      console.error('CodeEditor: Container element required');
    76→      return;
    77→    }
    78→    
    79→    container = element;
    80→    onContentChange = options.onContentChange || null;
    81→    onFileSave = options.onFileSave || null;
    82→    
    83→    // Find or create sub-containers
    84→    tabsContainer = container.querySelector('.editor-tabs') || createTabsContainer();
    85→    editorContainer = container.querySelector('.editor-content') || createEditorContainer();
    86→    
    87→    // Setup event listeners
    88→    setupEventListeners();
    89→    
    90→    // Subscribe to state changes
    91→    if (window.IDEState) {
    92→      IDEState.subscribe((state, prevState) => {
    93→        // Handle active file changes
    94→        if (state.activeFile !== prevState.activeFile && state.activeFile) {
    95→          openFile(state.activeFile);
    96→        }
    97→        
    98→        // Handle editor panel toggle
    99→        if (state.editorPanelOpen !== prevState.editorPanelOpen) {
   100→          togglePanel(state.editorPanelOpen);
   101→        }
   102→        
   103→        // Handle file content updates from external sources
   104→        if (state.fileContent !== prevState.fileContent && state.fileContent !== currentContent) {
   105→          if (!isEditing()) {
   106→            displayContent(state.fileContent);
   107→          }
   108→        }
   109→      });
   110→    }
   111→    
   112→    // Initial render
   113→    renderTabs();
   114→    showWelcomeScreen();
   115→  }
   116→
   117→  /**
   118→   * Create tabs container if not in DOM
   119→   */
   120→  function createTabsContainer() {
   121→    const div = document.createElement('div');
   122→    div.className = 'editor-tabs';
   123→    container.prepend(div);
   124→    return div;
   125→  }
   126→
   127→  /**
   128→   * Create editor content area if not in DOM
   129→   */
   130→  function createEditorContainer() {
   131→    const div = document.createElement('div');
   132→    div.className = 'editor-content';
   133→    container.appendChild(div);
   134→    return div;
   135→  }
   136→
   137→  /**
   138→   * Setup global event listeners
   139→   */
   140→  function setupEventListeners() {
   141→    // Keyboard shortcuts
   142→    document.addEventListener('keydown', (e) => {
   143→      // Ctrl/Cmd + S to save
   144→      if ((e.ctrlKey || e.metaKey) && e.key === 's') {
   145→        e.preventDefault();
   146→        if (currentMode === 'edit') {
   147→          saveCurrentFile();
   148→        }
   149→      }
   150→      
   151→      // Ctrl/Cmd + W to close tab
   152→      if ((e.ctrlKey || e.metaKey) && e.key === 'w') {
   153→        e.preventDefault();
   154→        if (activeTabPath) {
   155→          closeTab(activeTabPath);
   156→        }
   157→      }
   158→      
   159→      // Escape to exit edit mode
   160→      if (e.key === 'Escape' && currentMode === 'edit') {
   161→        exitEditMode();
   162→      }
   163→    });
   164→    
   165→    // Tab bar click for closing tabs
   166→    if (tabsContainer) {
   167→      tabsContainer.addEventListener('click', (e) => {
   168→        const closeBtn = e.target.closest('.tab-close');
   169→        if (closeBtn) {
   170→          e.stopPropagation();
   171→          const tab = closeBtn.closest('.editor-tab');
   172→          if (tab) {
   173→            closeTab(tab.dataset.path);
   174→          }
   175→        }
   176→      });
   177→    }
   178→  }
   179→
   180→  // ============================================
   181→  // Tab Management
   182→  // ============================================
   183→  
   184→  /**
   185→   * Open a file in a new or existing tab
   186→   */
   187→  async function openFile(file) {
   188→    if (!file || !file.path) return;
   189→    
   190→    // Check if already open
   191→    const existingTab = openTabs.find(t => t.path === file.path);
   192→    if (existingTab) {
   193→      activateTab(file.path);
   194→      return;
   195→    }
   196→    
   197→    // Add to tabs
   198→    openTabs.push({ ...file });
   199→    
   200→    // Activate this tab
   201→    activateTab(file.path);
   202→    
   203→    // Load content
   204→    await loadFileContent(file);
   205→    
   206→    // Re-render tabs
   207→    renderTabs();
   208→  }
   209→
   210→  /**
   211→   * Activate a specific tab by path
   212→   */
   213→  function activateTab(path) {
   214→    activeTabPath = path;
   215→    
   216→    // Update state
   217→    const tab = openTabs.find(t => t.path === path);
   218→    if (tab && window.IDEState) {
   219→      IDEState.setActiveFile(tab);
   220→    }
   221→    
   222→    // Update visual
   223→    updateTabVisuals();
   224→    
   225→    // Load content if not loaded
   226→    if (tab && !tab.contentLoaded) {
   227→      loadFileContent(tab);
   228→    }
   229→  }
   230→
   231→  /**
   232→   * Close a tab
   233→   */
   234→  function closeTab(path) {
   235→    const index = openTabs.findIndex(t => t.path === path);
   236→    if (index === -1) return;
   237→    
   238→    // Check for unsaved changes
   239→    const tab = openTabs[index];
   240→    if (tab.hasChanges) {
   241→      const shouldClose = confirm('You have unsaved changes. Close anyway?');
   242→      if (!shouldClose) return;
   243→    }
   244→    
   245→    // Remove tab
   246→    openTabs.splice(index, 1);
   247→    
   248→    // If closed tab was active, activate adjacent tab
   249→    if (path === activeTabPath) {
   250→      if (openTabs.length > 0) {
   251→        const newIndex = Math.min(index, openTabs.length - 1);
   252→        activateTab(openTabs[newIndex].path);
   253→      } else {
   254→        activeTabPath = null;
   255→        // No tabs left -> show the empty state (#editor-no-file)
   256→        showWelcomeScreen();
   257→      }
   258→    }
   259→    
   260→    renderTabs();
   261→  }
   262→
   263→  /**
   264→   * Close all tabs except active
   265→   */
   266→  function closeOtherTabs(keepPath) {
   267→    openTabs = openTabs.filter(t => t.path === keepPath);
   268→    activeTabPath = keepPath;
   269→    renderTabs();
   270→    activateTab(keepPath);
   271→  }
   272→
   273→  /**
   274→   * Render all tabs
   275→   */
   276→  function renderTabs() {
   277→    if (!tabsContainer) return;
   278→    
   279→    tabsContainer.innerHTML = '';
   280→    
   281→    if (openTabs.length === 0) {
   282→      tabsContainer.classList.add('hidden');
   283→      return;
   284→    }
   285→    
   286→    tabsContainer.classList.remove('hidden');
   287→    
   288→    openTabs.forEach(tab => {
   289→      const tabEl = document.createElement('div');
   290→      tabEl.className = `editor-tab ${tab.path === activeTabPath ? 'active' : ''} ${tab.hasChanges ? 'has-changes' : ''}`;
   291→      tabEl.dataset.path = tab.path;
   292→      
   293→      // Icon
   294→      const icon = document.createElement('span');
   295→      icon.className = 'tab-icon';
   296→      icon.innerHTML = IDEUtils?.getFileIcon?.(tab.name) || IDEUtils?.Icons?.file || '';
   297→      tabEl.appendChild(icon);
   298→      
   299→      // Name
   300→      const name = document.createElement('span');
   301→      name.className = 'tab-name';
   302→      name.textContent = tab.name;
   303→      name.title = tab.path;
   304→      tabEl.appendChild(name);
   305→      
   306→      // Change indicator
   307→      if (tab.hasChanges) {
   308→        const dot = document.createElement('span');
   309→        dot.className = 'change-indicator';
   310→        dot.setAttribute('aria-label', 'Unsaved changes');
   311→        tabEl.appendChild(dot);
   312→      }
   313→      
   314→      // Close button
   315→      const closeBtn = document.createElement('button');
   316→      closeBtn.className = 'tab-close icon-btn';
   317→      closeBtn.setAttribute('aria-label', `Close ${tab.name}`);
   318→      closeBtn.innerHTML = IDEUtils?.Icons?.x || '';
   319→      tabEl.appendChild(closeBtn);
   320→      
   321→      // Click to activate
   322→      tabEl.addEventListener('click', (e) => {
   323→        if (!e.target.closest('.tab-close')) {
   324→          activateTab(tab.path);
   325→        }
   326→      });
   327→      
   328→      // Context menu for "close others"
   329→      tabEl.addEventListener('contextmenu', (e) => {
   330→        e.preventDefault();
   331→        closeOtherTabs(tab.path);
   332→      });
   333→      
   334→      tabsContainer.appendChild(tabEl);
   335→    });
   336→  }
   337→
   338→  /**
   339→   * Update tab visual states without full re-render
   340→   */
   341→  function updateTabVisuals() {
   342→    if (!tabsContainer) return;
   343→    
   344→    tabsContainer.querySelectorAll('.editor-tab').forEach(tabEl => {
   345→      const isActive = tabEl.dataset.path === activeTabPath;
   346→      const tabData = openTabs.find(t => t.path === tabEl.dataset.path);
   347→      
   348→      tabEl.classList.toggle('active', isActive);
   349→      tabEl.classList.toggle('has-changes', tabData?.hasChanges || false);
   350→    });
   351→  }

   353→  // ============================================
   354→  // Content Loading & Display
   355→  // ============================================
   356→  
   357→  /**
   358→   * Load file content from storage
   359→   */
   360→  async function loadFileContent(file) {
   361→    if (!file || !file.project) return;
   362→    
   363→    try {
   364→      const content = await IDEStorage.Files.readContent(file.project, file.path);
   365→      currentContent = content || '';
   366→      originalContent = currentContent;
   367→      
   368→      // Mark as loaded
   369→      const tab = openTabs.find(t => t.path === file.path);
   370→      if (tab) {
   371→        tab.contentLoaded = true;
   372→        tab.hasChanges = false;
   373→      }
   374→      
   375→      // Update state
   376→      if (window.IDEState) {
   377→        IDEState.setFileContent(currentContent);
   378→      }
   379→      
   380→      // Display based on mode
   381→      displayContent(currentContent);
   382→
   383→      // Show #editor-active now that a file is open
   384→      setEditorVisibility(true);
   385→
   386→    } catch (error) {
   387→      console.error('Failed to load file:', error);
   388→      showErrorState('Failed to load file content');
   389→    }
   390→  }
   391→
   392→  /**
   393→   * Display content in appropriate view
   394→   */
   395→  function displayContent(content) {
   396→    if (!editorContainer) return;
   397→    
   398→    currentContent = content || '';
   399→    
   400→    switch (currentMode) {
   401→      case 'view':
   402→        renderViewMode(content);
   403→        break;
   404→      case 'edit':
   405→        renderEditMode(content);
   406→        break;
   407→      case 'preview':
   408→        renderPreviewMode(content);
   409→        break;
   410→      default:
   411→        renderViewMode(content);
   412→    }
   413→  }
   414→
   415→  /**
   416→   * Show welcome/empty state.
   417→   *
   418→   * The HTML defines `#editor-no-file` as the empty state (icon + hint text).
   419→   * We toggle that element's visibility instead of rendering our own welcome
   420→   * screen, so the empty state stays consistent with the rest of the layout.
   421→   */
   422→  function showWelcomeScreen() {
   423→    // Clear any rendered content (in case we're transitioning from a file view)
   424→    if (editorContainer) {
   425→      editorContainer.innerHTML = '';
   426→    }
   427→    // Show #editor-no-file, hide #editor-active
   428→    setEditorVisibility(false);
   429→  }
   430→
   431→  /**
   432→   * Toggle visibility of #editor-no-file vs #editor-active.
   433→   * @param {boolean} fileActive - true when a file is open, false otherwise.
   434→   */
   435→  function setEditorVisibility(fileActive) {
   436→    const root = container || document;
   437→    const noFileEl = root.querySelector('#editor-no-file');
   438→    const activeEl = root.querySelector('#editor-active');
   439→    if (noFileEl) {
   440→      noFileEl.classList.toggle('hidden', fileActive);
   441→    }
   442→    if (activeEl) {
   443→      activeEl.classList.toggle('hidden', !fileActive);
   444→    }
   445→  }
   446→
   447→  /**
   448→   * Show error state
   449→   */
   450→  function showErrorState(message) {
   451→    if (!editorContainer) return;
   452→    
   453→    editorContainer.innerHTML = `
   454→      <div class="error-state">
   455→        <span class="error-icon">${IDEUtils?.Icons?.error || ''}</span>
   456→        <p>${IDEUtils?.escapeHtml(message) || message}</p>
   457→        <button class="btn btn-outline btn-sm" onclick="this.closest('.editor-content').querySelector('.retry-btn')?.click()">
   458→          Retry
   459→        </button>
   460→      </div>
   461→    `;
   462→  }

   468→  /**
   469→   * Render read-only view with syntax highlighting
   470→   */
   471→  function renderViewMode(content) {
   472→    if (!editorContainer) return;
   473→    
   474→    const activeFile = getActiveTab();
   475→    const language = activeFile ? IDEUtils.detectLanguage(activeFile.name) : 'plaintext';
   476→    
   477→    editorContainer.innerHTML = `
   478→      <div class="view-mode">
   479→        <div class="view-header">
   480→          <span class="view-mode-label">View Mode</span>
   481→          <div class="view-actions">
   482→            <button class="btn btn-sm btn-outline" data-action="edit" title="Edit (E)">
   483→              ${IDEUtils?.Icons?.edit || ''} Edit
   484→            </button>
   485→            ${canPreview(activeFile) ? `
   486→              <button class="btn btn-sm btn-outline" data-action="preview" title="Preview (P)">
   487→                ${IDEUtils?.Icons?.eye || ''} Preview
   488→              </button>
   489→            ` : ''}
   490→          </div>
   491→        </div>
   492→        <pre class="code-view"><code class="language-${language}">${IDEUtils?.escapeHtml(content) || ''}</code></pre>
   493→      </div>
   494→    `;
   495→    
   496→    // Apply syntax highlighting
   497→    if (window.hljs) {
   498→      editorContainer.querySelectorAll('pre code').forEach(block => {
   499→        hljs.highlightElement(block);
   500→      });
   501→    }
   502→    
   503→    // Wire up action buttons
   504→    wireUpViewActions();
   505→  }
   506→
   507→  /**
   508→   * Render editable mode with textarea
   509→   */
   510→  function renderEditMode(content) {
   511→    if (!editorContainer) return;
   512→    
   513→    refreshEditorPrefs();
   514→    const activeFile = getActiveTab();
   515→    const language = activeFile ? IDEUtils.detectLanguage(activeFile.name) : 'plaintext';
   516→    
   517→    const wrapAttr = editorPrefs.wordWrap ? 'wrap="soft"' : 'wrap="off"';
   518→    const gutterHtml = editorPrefs.lineNumbers
   519→      ? `<div class="code-editor-gutter" aria-hidden="true">${renderLineNumbers(content)}</div>`
   520→      : '';
   521→    const editorLayoutClass = editorPrefs.lineNumbers ? 'with-gutter' : 'no-gutter';
   522→    const overflowClass = editorPrefs.wordWrap ? 'wrap-on' : 'wrap-off';
   523→    
   524→    editorContainer.innerHTML = `
   525→      <div class="edit-mode ${editorLayoutClass} ${overflowClass}">
   526→        <div class="edit-header">
   527→          <span class="edit-mode-label">Edit Mode</span>
   528→          <span class="edit-language">${language}</span>
   529→          <div class="edit-actions">
   530→            <button class="btn btn-sm btn-primary" data-action="save" title="Save (Ctrl+S)">
   531→              ${IDEUtils?.Icons?.save || ''} Save
   532→            </button>
   533→            <button class="btn btn-sm btn-outline" data-action="cancel" title="Cancel (Esc)">
   534→              ${IDEUtils?.Icons?.x || ''} Cancel
   535→            </button>
   536→          </div>
   537→        </div>
   538→        <div class="edit-body">
   539→          ${gutterHtml}
   540→          <textarea
   541→            class="code-editor-textarea"
   542→            spellcheck="false"
   543→            autocomplete="off"
   544→            autocapitalize="off"
   545→            aria-label="Code editor"
   546→            placeholder="Start typing..."
   547→            ${wrapAttr}
   548→          >${IDEUtils?.escapeHtml(content) || ''}</textarea>
   549→        </div>
   550→        <div class="edit-status">
   551→          <span class="line-info">Line 1, Column 1</span>
   552→          <span class="char-count">${content.length} characters</span>
   553→        </div>
   554→      </div>
   555→    `;
   556→    
   557→    const textarea = editorContainer.querySelector('.code-editor-textarea');
   558→    if (textarea) {
   559→      // Auto-resize handling
   560→      textarea.addEventListener('input', () => handleEditInput(textarea));
   561→      
   562→      // Track cursor position
   563→      textarea.addEventListener('click', () => updateCursorPosition(textarea));
   564→      textarea.addEventListener('keyup', () => updateCursorPosition(textarea));
   565→      
   566→      // Sync gutter scroll with textarea
   567→      if (editorPrefs.lineNumbers) {
   568→        const gutter = editorContainer.querySelector('.code-editor-gutter');
   569→        if (gutter) {
   570→          textarea.addEventListener('scroll', () => {
   571→            gutter.scrollTop = textarea.scrollTop;
   572→          });
   573→        }
   574→      }
   575→      
   576→      // Focus textarea
   577→      textarea.focus();
   578→      
   579→      // Font-family / font-size / line-height are set in CSS on
   580→      // .edit-mode .code-editor-textarea so they stay in sync with the
   581→      // gutter (same font-size + line-height = pixel-perfect alignment).
   582→    }
   583→    
   584→    // Wire up action buttons
   585→    wireUpEditActions();
   586→  }

   588→  /**
   589→   * Render line numbers HTML for the given content
   590→   */
   591→  function renderLineNumbers(content) {
   592→    const lineCount = (content || '').split('\n').length;
   593→    let html = '';
   594→    for (let i = 1; i <= lineCount; i++) {
   595→      html += `<div class="gutter-line">${i}</div>`;
   596→    }
   597→    return html;
   598→  }
   599→  
   600→  /**
   601→   * Refresh the line-numbers gutter for the current edit content
   602→   */
   603→  function refreshGutter(content) {
   604→    if (!editorContainer) return;
   605→    const gutter = editorContainer.querySelector('.code-editor-gutter');
   606→    if (gutter) {
   607→      gutter.innerHTML = renderLineNumbers(content);
   608→    }
   609→  }

   611→  /**
   612→   * Render preview mode for HTML/Markdown
   613→   */
   614→  function renderPreviewMode(content) {
   615→    if (!editorContainer) return;
   616→    
   617→    const activeFile = getActiveTab();
   618→    const isHtml = activeFile && /\.(html?|htm)$/.test(activeFile.name.toLowerCase());
   619→    
   620→    editorContainer.innerHTML = `
   621→      <div class="preview-mode">
   622→        <div class="preview-header">
   623→          <span class="preview-mode-label">Preview</span>
   624→          <div class="preview-actions">
   625→            <button class="btn btn-sm btn-outline" data-action="view" title="Back to View">
   626→              ${IDEUtils?.Icons?.eye || ''} View Code
   627→            </button>
   628→            <button class="btn btn-sm btn-outline" data-action="open-new" title="Open in New Tab">
   629→              ${IDEUtils?.Icons?.externalLink || ''} Open New
   630→            </button>
   631→          </div>
   632→        </div>
   633→        <iframe class="preview-frame" sandbox="allow-scripts allow-same-origin"></iframe>
   634→      </div>
   635→    `;
   636→    
   637→    const iframe = editorContainer.querySelector('.preview-frame');
   638→    if (iframe) {
   639→      if (isHtml) {
   640→        // Render HTML directly
   641→        iframe.srcdoc = content;
   642→      } else {
   643→        // For markdown, render simple preview (basic conversion)
   644→        const htmlContent = markdownToHtml(content);
   645→        iframe.srcdoc = htmlContent;
   646→      }
   647→    }
   648→    
   649→    // Wire up actions
   650→    wireUpPreviewActions();
   651→  }

   654→  // Edit Handling

   656→  
   657→  /**
   658→   * Handle input in edit mode
   659→   */
   660→  function handleEditInput(textarea) {
   661→    const newContent = textarea.value;
   662→    currentContent = newContent;
   663→    
   664→    // Update character count
   665→    const charCount = editorContainer.querySelector('.char-count');
   666→    if (charCount) {
   667→      charCount.textContent = `${newContent.length} characters`;
   668→    }
   669→    
   670→    // Refresh gutter if line numbers are enabled
   671→    if (editorPrefs.lineNumbers) {
   672→      refreshGutter(newContent);
   673→    }
   674→    
   675→    // Mark tab as having changes
   676→    const hasChanges = newContent !== originalContent;
   677→    const tab = openTabs.find(t => t.path === activeTabPath);
   678→    if (tab) {
   679→      tab.hasChanges = hasChanges;
   680→    }
   681→    updateTabVisuals();
   682→    
   683→    // Notify of content change
   684→    if (onContentChange) {
   685→      onContentChange(newContent);
   686→    }
   687→    
   688→    // Debounced auto-save: actually persist the file when autoSave is enabled
   689→    clearTimeout(autoSaveTimeout);
   690→    autoSaveTimeout = setTimeout(() => {
   691→      if (editorPrefs.autoSave && hasChanges && activeTabPath) {
   692→        // Save silently (no toast, no mode switch) so the user can keep typing
   693→        silentSave().catch(err => console.error('Auto-save failed:', err));
   694→      }
   695→    }, AUTO_SAVE_DELAY);
   696→  }

   700→   * or switching out of edit mode, so the user can continue editing.

   712→

   716→    
   717→    if (IDEUtils && IDEUtils.playSound) {
   718→      IDEUtils.playSound('save');
   719→    }
   720→    
   721→    // Notify callback (e.g. version browser may want to know)
   722→    if (onFileSave) {
   723→      onFileSave(tab, currentContent);
   724→    }
   725→  }

   790→    if (!activeTabPath) return;
   791→    
   792→    const tab = openTabs.find(t => t.path === activeTabPath);
   793→    if (!tab || !tab.project) return;
   794→    
   795→    try {
   796→      await IDEStorage.Files.writeContent(tab.project, tab.path, currentContent);

   798→      // Update state
   799→      originalContent = currentContent;
   800→      tab.hasChanges = false;

   803→      
   804→      // Update last saved timestamp
   805→      if (window.IDEState) {
   806→        IDEState.setLastSaved(Date.now());
   807→      }
   808→      
   809→      // Notify callback
   810→      if (onFileSave) {
   811→        onFileSave(tab, currentContent);
   812→      }
   813→      
   814→      // Play save sound if sound effects are enabled
   815→      if (IDEUtils && IDEUtils.playSound) {
   816→        IDEUtils.playSound('save');
   817→      }
   818→      
   819→      // Show success toast
   820→      if (IDEUtils) {
   821→        IDEUtils.showToast(`Saved ${tab.name}`, 'success');
   822→      }
   823→      
   824→      // Switch back to view mode
   825→      currentMode = 'view';
   826→      displayContent(currentContent);
   827→      
   828→    } catch (error) {
   829→      console.error('Failed to save file:', error);
   830→      if (IDEUtils) {
   831→        IDEUtils.showToast(`Failed to save: ${error.message}`, 'error');
   832→      }
   833→    }
   834→  }

   897→  /**
   898→   * Simple markdown to HTML converter (basic)
   899→   */
   900→  function markdownToHtml(md) {
   901→    if (!md) return '';
   902→    
   903→    let html = md
   904→      // Headers
   905→      .replace(/^### (.*$)/gim, '<h3>$1</h3>')
   906→      .replace(/^## (.*$)/gim, '<h2>$1</h2>')
   907→      .replace(/^# (.*$)/gim, '<h1>$1</h1>')
   908→      // Bold/Italic
   909→      .replace(/\*\*\*(.*?)\*\*\*/gim, '<strong><em>$1</em></strong>')
   910→      .replace(/\*\*(.*?)\*\*/gim, '<strong>$1</strong>')
   911→      .replace(/\*(.*?)\*/gim, '<em>$1</em>')
   912→      // Code blocks
   913→      .replace(/```(\w+)?\n([\s\S]*?)```/gim, '<pre><code class="$1">$2</code></pre>')
   914→      .replace(/`(.*?)`/gim, '<code>$1</code>')
   915→      // Links
   916→      .replace(/\[([^\]]+)\]\(([^)]+)\)/gim, '<a href="$2">$1</a>')
   917→      // Line breaks
   918→      .replace(/\n/gim, '<br>');
   919→    
   920→    return `<!DOCTYPE html><html><head><style>
   921→      body { font-family: system-ui; padding: 20px; max-width: 800px; margin: 0 auto; }
   922→      pre { background: #f4f4f4; padding: 12px; border-radius: 6px; overflow-x: auto; }
   923→      code { background: #f4f4f4; padding: 2px 6px; border-radius: 3px; }
   924→      a { color: #0066cc; }
   925→    </style></head><body>${html}</body></html>`;
   926→  }

   988→  let currentMatchIndex = -1;
   989→  let currentMatches = [];
   990→  
   991→  function bindFindReplaceEvents() {
   992→    const findInput = findReplaceEl.querySelector('.fr-find-input');
   993→    const replaceInput = findReplaceEl.querySelector('.fr-replace-input');
   994→    const countEl = findReplaceEl.querySelector('.fr-count');
   995→    
   996→    // Search on input
   997→    findInput.addEventListener('input', () => performFind());
   998→    
   999→    // Keyboard navigation
  1000→    findInput.addEventListener('keydown', (e) => {
  1001→      if (e.key === 'Enter') {
  1002→        e.preventDefault();
  1003→        if (e.shiftKey) {
  1004→          navigateMatches(-1);
  1005→        } else {
  1006→          navigateMatches(1);
  1007→        }
  1008→      } else if (e.key === 'Escape') {
  1009→        e.preventDefault();
  1010→        closeFindReplace();
  1011→      }
  1012→    });
  1013→    
  1014→    // Button actions
  1015→    findReplaceEl.querySelector('[data-action="fr-prev"]')?.addEventListener('click', () => navigateMatches(-1));
  1016→    findReplaceEl.querySelector('[data-action="fr-next"]')?.addEventListener('click', () => navigateMatches(1));
  1017→    findReplaceEl.querySelector('[data-action="fr-replace"]')?.addEventListener('click', () => replaceCurrent());
  1018→    findReplaceEl.querySelector('[data-action="fr-replace-all"]')?.addEventListener('click', () => replaceAll());
  1019→    findReplaceEl.querySelector('[data-action="fr-close"]')?.addEventListener('click', closeFindReplace);
  1020→  }
  1021→  
  1022→  function performFind() {
  1023→    const textarea = editorContainer?.querySelector('.code-editor-textarea');
  1024→    const findInput = findReplaceEl?.querySelector('.fr-find-input');
  1025→    const countEl = findReplaceEl?.querySelector('.fr-count');
  1026→    
  1027→    if (!textarea || !findInput) return;
  1028→    
  1029→    const query = findInput.value;
  1030→    const text = textarea.value;
  1031→    
  1032→    currentMatches = [];
  1033→    currentMatchIndex = -1;
  1034→    
  1035→    if (query) {
  1036→      // Find all matches (case-insensitive)
  1037→      const regex = new RegExp(query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'gi');
  1038→      let match;
  1039→      while ((match = regex.exec(text)) !== null) {
  1040→        currentMatches.push({ index: match.index, length: match[0].length, text: match[0] });
  1041→      }
  1042→      
  1043→      if (currentMatches.length > 0) {
  1044→        currentMatchIndex = 0;
  1045→        highlightMatch(textarea);
  1046→      }
  1047→    }
  1048→    
  1049→    // Update count
  1050→    if (countEl) {
  1051→      countEl.textContent = currentMatches.length > 0 
  1052→        ? \`\${currentMatchIndex + 1}/\${currentMatches.length}\` 
  1053→        : '0/0';
  1054→    }
  1055→  }
  1056→  
  1057→  function navigateMatches(direction) {
  1058→    if (currentMatches.length === 0) return;
  1059→    
  1060→    const textarea = editorContainer?.querySelector('.code-editor-textarea');
  1061→    if (!textarea) return;
  1062→    
  1063→    currentMatchIndex += direction;
  1064→    if (currentMatchIndex < 0) currentMatchIndex = currentMatches.length - 1;
  1065→    if (currentMatchIndex >= currentMatches.length) currentMatchIndex = 0;
  1066→    
  1067→    highlightMatch(textarea);
  1068→    
  1069→    // Update count
  1070→    const countEl = findReplaceEl?.querySelector('.fr-count');
  1071→    if (countEl) {
  1072→      countEl.textContent = \`\${currentMatchIndex + 1}/\${currentMatches.length}\`;
  1073→    }
  1074→  }
  1075→  
  1076→  function highlightMatch(textarea) {
  1077→    const match = currentMatches[currentMatchIndex];
  1078→    if (!match) return;
  1079→    
  1080→    textarea.focus();
  1081→    textarea.setSelectionRange(match.index, match.index + match.length);
  1082→    
  1083→    // Scroll to selection
  1084→    const lineNum = textarea.value.substring(0, match.index).split('\n').length;
  1085→    const lineHeight = parseInt(getComputedStyle(textarea).lineHeight) || 20;
  1086→    textarea.scrollTop = Math.max(0, (lineNum - 10) * lineHeight);
  1087→  }
  1088→  
  1089→  function replaceCurrent() {
  1090→    const textarea = editorContainer?.querySelector('.code-editor-textarea');
  1091→    const replaceInput = findReplaceEl?.querySelector('.fr-replace-input');
  1092→    
  1093→    if (!textarea || !replaceInput || currentMatches.length === 0) return;
  1094→    
  1095→    const match = currentMatches[currentMatchIndex];
  1096→    const newText = textarea.value.substring(0, match.index) + 
  1097→                    replaceInput.value + 
  1098→                    textarea.value.substring(match.index + match.length);
  1099→    
  1100→    textarea.value = newText;
  1101→    handleEditInput(textarea); // Trigger change handler
  1102→    
  1103→    // Re-find to update matches
  1104→    performFind();
  1105→  }
  1106→  
  1107→  function replaceAll() {
  1108→    const textarea = editorContainer?.querySelector('.code-editor-textarea');
  1109→    const findInput = findReplaceEl?.querySelector('.fr-find-input');
  1110→    const replaceInput = findReplaceEl?.querySelector('.fr-replace-input');
  1111→    
  1112→    if (!textarea || !findInput) return;
  1113→    
  1114→    const query = findInput.value;
  1115→    if (!query) return;
  1116→    
  1117→    const regex = new RegExp(query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'g');
  1118→    const newText = textarea.value.replace(regex, replaceInput.value);
  1119→    
  1120→    if (newText !== textarea.value) {
  1121→      textarea.value = newText;
  1122→      handleEditInput(textarea);
  1123→      if (window.IDEUtils) IDEUtils.showToast(\`Replaced \${currentMatches.length} occurrences\`, 'success');
  1124→    }
  1125→    
  1126→    performFind(); // Re-find (should be 0 now)
  1127→  }
