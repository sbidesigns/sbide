     1→/**
     2→ * SBIDE — Connections Module
     3→ * ---------------------------------------------------------------
     4→ * Provides OAuth/PAT-style connections to HuggingFace and GitHub
     5→ * directly from a static page. Tokens are stored in OPFS via
     6→ * OfflineKit when available, falling back to localStorage with a
     7→ * light obfuscation wrapper (NOT secure storage — explicit warning
     8→ * is shown to the user in the modal).
     9→ *
    10→ * Public API (window.Connections):
    11→ *   init()                          — wire up event listeners
    12→ *   openModal()                     — show the connections modal
    13→ *   closeModal()                    — hide the modal
    14→ *   getStatus()                     — { huggingface, github } status objects
    15→ *   isConnected(provider)           — bool
    16→ *   listRepos(provider)             — array of repo summaries
    17→ *   createRepo(provider, opts)      — create a new repo / space
    18→ *   pushProject(provider, opts)     — push current IDE project to remote
    19→ *   pullProject(provider, opts)     — pull remote repo into IDE as new project
    20→ *   deleteRepo(provider, name)      — delete a remote repo (with confirm)
    21→ *   disconnect(provider)            — revoke local token (does not call provider)
    22→ *   onStatusChange(cb)              — subscribe to status changes
    23→ */

    93→  // Lightweight XOR obfuscation. NOT secure — just prevents the token
    94→  // from being trivially readable in localStorage. Real protection comes
    95→  // from OPFS when available.
    96→  const OBF_SALT = 'IDE-Conn-v1-salt';
    97→  function obfuscate(str) {
    98→    if (!str) return '';
    99→    let out = '';
   100→    for (let i = 0; i < str.length; i++) {
   101→      out += String.fromCharCode(str.charCodeAt(i) ^ OBF_SALT.charCodeAt(i % OBF_SALT.length));
   102→    }
   103→    // base64 for safe storage
   104→    return btoa(unescape(encodeURIComponent(out)));
   105→  }
   106→  function deobfuscate(b64) {
   107→    if (!b64) return '';
   108→    try {
   109→      const str = decodeURIComponent(escape(atob(b64)));
   110→      let out = '';
   111→      for (let i = 0; i < str.length; i++) {
   112→        out += String.fromCharCode(str.charCodeAt(i) ^ OBF_SALT.charCodeAt(i % OBF_SALT.length));
   113→      }
   114→      return out;
   115→    } catch (e) {
   116→      return '';
   117→    }
   118→  }

   124→  async function opfsAvailable() {
   125→    return !!(window.OfflineKit && OfflineKit.opfsAvailable);
   126→  }
   127→
   128→  async function opfsReadSecrets() {
   129→    if (!await opfsAvailable()) return null;
   130→    try {
   131→      const raw = await OfflineKit.opfsRead('.secrets', 'connections.json');
   132→      if (!raw) return {};
   133→      return JSON.parse(raw);
   134→    } catch (e) {
   135→      warn('OPFS secrets read failed', e);
   136→      return {};
   137→    }
   138→  }
   139→
   140→  async function opfsWriteSecrets(obj) {
   141→    if (!await opfsAvailable()) return false;
   142→    try {
   143→      await OfflineKit.opfsWrite('.secrets', 'connections.json', JSON.stringify(obj, null, 2));
   144→      return true;
   145→    } catch (e) {
   146→      warn('OPFS secrets write failed', e);
   147→      return false;
   148→    }
   149→  }
   150→
   151→  async function getToken(provider) {
   152→    const key = STORAGE_KEYS[provider];
   153→    // Try OPFS first
   154→    const secrets = await opfsReadSecrets();
   155→    if (secrets && secrets[key]) {
   156→      return deobfuscate(secrets[key]);
   157→    }
   158→    // Fallback localStorage
   159→    const raw = localStorage.getItem(key);
   160→    return raw ? deobfuscate(raw) : '';
   161→  }
   162→
   163→  async function setToken(provider, token) {
   164→    const key = STORAGE_KEYS[provider];
   165→    const enc = obfuscate(token || '');
   166→    // Always write to localStorage as a fallback (even if OPFS works)
   167→    localStorage.setItem(key, enc);
   168→    // And mirror to OPFS when available
   169→    const secrets = await opfsReadSecrets() || {};
   170→    secrets[key] = enc;
   171→    await opfsWriteSecrets(secrets);
   172→  }
   173→
   174→  async function clearToken(provider) {
   175→    const key = STORAGE_KEYS[provider];
   176→    localStorage.removeItem(key);
   177→    const secrets = await opfsReadSecrets() || {};
   178→    delete secrets[key];
   179→    await opfsWriteSecrets(secrets);
   180→  }

   186→  async function httpFetch(url, opts = {}) {
   187→    const res = await fetch(url, {
   188→      ...opts,
   189→      headers: {
   190→        'Accept': 'application/json',
   191→        ...(opts.headers || {})
   192→      }
   193→    });
   194→    if (!res.ok) {
   195→      let detail = '';
   196→      try {
   197→        const body = await res.json();
   198→        detail = body?.error || body?.message || JSON.stringify(body).slice(0, 200);
   199→      } catch (e) {
   200→        try { detail = (await res.text()).slice(0, 200); } catch (_) {}
   201→      }
   202→      const e = new Error(`HTTP ${res.status} ${res.statusText}${detail ? ' — ' + detail : ''}`);
   203→      e.status = res.status;
   204→      e.body = detail;
   205→      throw e;
   206→    }
   207→    const ct = res.headers.get('content-type') || '';
   208→    if (ct.includes('application/json')) return res.json();
   209→    if (ct.includes('text/')) return res.text();
   210→    return res.arrayBuffer();
   211→  }

   385→    async downloadFile(token, owner, repo, path, branch = 'main') {
   386→      const json = await httpFetch(
   387→        `https://api.github.com/repos/${owner}/${repo}/contents/${path.split('/').map(encodeURIComponent).join('/')}?ref=${branch}`,
   388→        { headers: { 'Authorization': `token ${token}` } }
   389→      );
   390→      if (json.encoding === 'base64' && json.content) {
   391→        // Handle UTF-8 properly
   392→        const binary = atob(json.content.replace(/\n/g, ''));
   393→        try {
   394→          // Try UTF-8 decode
   395→          const bytes = new Uint8Array(binary.length);
   396→          for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i);
   397→          return new TextDecoder('utf-8').decode(bytes);
   398→        } catch (e) {
   399→          return binary;
   400→        }
   401→      }
   402→      return json.content || '';
   403→    },

   405→    async uploadFile(token, owner, repo, path, content, branch = 'main', commitMsg = null, existingSha = null) {
   406→      // PUT /repos/:owner/:repo/contents/:path
   407→      // Body: { message, content (base64), branch, sha? }
   408→      const isText = typeof content === 'string';
   409→      let base64;
   410→      if (isText) {
   411→        const bytes = new TextEncoder().encode(content);
   412→        let bin = '';
   413→        for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
   414→        base64 = btoa(bin);
   415→      } else {
   416→        // Blob/arraybuffer
   417→        const buf = content instanceof ArrayBuffer ? content : await content.arrayBuffer();
   418→        const bytes = new Uint8Array(buf);
   419→        let bin = '';
   420→        for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
   421→        base64 = btoa(bin);
   422→      }
   423→      const body = {
   424→        message: commitMsg || `Update ${path}`,
   425→        content: base64,
   426→        branch
   427→      };
   428→      if (existingSha) body.sha = existingSha;
   429→      return httpFetch(
   430→        `https://api.github.com/repos/${owner}/${repo}/contents/${path.split('/').map(encodeURIComponent).join('/')}`,
   431→        {
   432→          method: 'PUT',
   433→          headers: {
   434→            'Authorization': `token ${token}`,
   435→            'Content-Type': 'application/json'
   436→          },
   437→          body: JSON.stringify(body)
   438→        }
   439→      );
   440→    },

   517→  async function connect(provider, token) {
   518→    if (!token || !token.trim()) {
   519→      throw new Error('Token is required');
   520→    }
   521→    const trimmed = token.trim();
   522→    let info;
   523→    if (provider === 'huggingface') {
   524→      info = await HF.validate(trimmed);
   525→    } else if (provider === 'github') {
   526→      info = await GH.validate(trimmed);
   527→    } else {
   528→      throw new Error(`Unknown provider: ${provider}`);
   529→    }
   530→    await setToken(provider, trimmed);
   531→    state.status[provider] = {
   532→      connected: true,
   533→      username: info.username,
   534→      name: info.name,
   535→      avatar: info.avatar
   536→    };
   537→    // Invalidate cache
   538→    state.cache[provider] = { repos: null, fetchedAt: 0 };
   539→    notifyStatusChange();
   540→    log(`Connected to ${PROVIDERS[provider].name} as ${info.username}`);
   541→    return info;
   542→  }
   543→
   544→  async function disconnect(provider) {
   545→    await clearToken(provider);
   546→    state.status[provider] = { connected: false, username: null, name: null, avatar: null };
   547→    state.cache[provider] = { repos: null, fetchedAt: 0 };
   548→    notifyStatusChange();
   549→    log(`Disconnected from ${PROVIDERS[provider].name}`);
   550→  }

   552→  async function loadStatusFromStorage() {
   553→    for (const p of Object.keys(PROVIDERS)) {
   554→      const token = await getToken(p);
   555→      if (token) {
   556→        // We have a stored token. We trust it without re-validating (faster).
   557→        // If it turns out invalid, the next API call will surface a 401 and we can
   558→        // disconnect then.
   559→        const meta = JSON.parse(localStorage.getItem(STORAGE_KEYS.githubMeta) || '{}');
   560→        if (p === 'github' && meta.username) {
   561→          state.status.github = { connected: true, username: meta.username, name: meta.name, avatar: meta.avatar };
   562→        } else {
   563→          // For HF, we don't cache user info — flag as connected but unknown username.
   564→          // The listRepos call will reveal the username.
   565→          state.status[p] = { connected: true, username: null, name: null, avatar: null };
   566→        }
   567→      }
   568→    }
   569→    notifyStatusChange();
   570→  }

   572→  async function listRepos(provider, { force = false } = {}) {
   573→    const token = await getToken(provider);
   574→    if (!token) throw new Error(`Not connected to ${PROVIDERS[provider].name}`);
   575→    const cache = state.cache[provider];
   576→    const now = Date.now();
   577→    const TTL = 60 * 1000; // 1 minute
   578→    if (!force && cache.repos && (now - cache.fetchedAt) < TTL) {
   579→      return cache.repos;
   580→    }
   581→    let repos;
   582→    if (provider === 'huggingface') {
   583→      // If we don't know the username yet, validate now to get it
   584→      if (!state.status.huggingface.username) {
   585→        const info = await HF.validate(token);
   586→        state.status.huggingface = { connected: true, ...info };
   587→        notifyStatusChange();
   588→      }
   589→      repos = await HF.listRepos(token, state.status.huggingface.username);
   590→    } else {
   591→      repos = await GH.listRepos(token);
   592→      // Cache user meta
   593→      if (!state.status.github.username) {
   594→        const info = await GH.validate(token);
   595→        state.status.github = { connected: true, ...info };
   596→        localStorage.setItem(STORAGE_KEYS.githubMeta, JSON.stringify(info));
   597→        notifyStatusChange();
   598→      }
   599→    }
   600→    state.cache[provider] = { repos, fetchedAt: now };
   601→    return repos;
   602→  }

   604→  async function createRepo(provider, opts) {
   605→    const token = await getToken(provider);
   606→    if (!token) throw new Error(`Not connected to ${PROVIDERS[provider].name}`);
   607→    let result;

   610→    } else {

   612→    }
   613→    // Invalidate cache so next listRepos call refetches
   614→    state.cache[provider] = { repos: null, fetchedAt: 0 };
   615→    return result;

   630→  /**
   631→   * Push the current IDE project's files to a remote repo.
   632→   * For HuggingFace: pushes to a model repo by default (or space/dataset if opts.type is set)
   633→   * For GitHub: pushes to a regular repo (creates if missing).
   634→   *
   635→   * opts: { repoId, type?, branch?, commitMsg? }
   636→   */
   637→  async function pushProject(provider, opts) {
   638→    const token = await getToken(provider);
   639→    if (!token) throw new Error(`Not connected to ${PROVIDERS[provider].name}`);
   640→
   641→    // Get current IDE project + files
   642→    const ideState = window.IDEState ? IDEState.getState() : null;
   643→    if (!ideState || !ideState.currentProject) {
   644→      throw new Error('No active project in the IDE. Open or create a project first.');
   645→    }
   646→    const projectName = ideState.currentProject.name;
   647→    const files = await IDEStorage.Files.getByProject(projectName);
   648→    const fileEntries = files.filter(f => f.type === 'file' && f.path);
   649→
   650→    if (fileEntries.length === 0) {
   651→      throw new Error(`Project "${projectName}" has no files to push.`);
   652→    }
   653→
   654→    let { repoId, branch = 'main', commitMsg = `Push from IDE — ${new Date().toISOString()}` } = opts;
   655→
   656→    if (provider === 'huggingface') {
   657→      const type = opts.type || 'model';
   658→      if (!repoId) {
   659→        // Default to {username}/{projectName-slug}
   660→        const username = state.status.huggingface.username;
   661→        if (!username) throw new Error('HuggingFace username unknown — re-connect to refresh.');
   662→        repoId = `${username}/${slugify(projectName)}`;
   663→      }
   664→      // Push each file
   665→      const results = [];
   666→      for (const f of fileEntries) {
   667→        try {
   668→          await HF.uploadFile(token, repoId, type, f.path, f.content || '', branch, commitMsg);
   669→          results.push({ path: f.path, ok: true });
   670→        } catch (e) {
   671→          results.push({ path: f.path, ok: false, error: e.message });
   672→        }
   673→      }
   674→      return { repoId, type, branch, pushed: results };
   675→    } else {
   676→      // GitHub
   677→      if (!repoId) throw new Error('repoId required for GitHub push (use "owner/repo")');
   678→      const [owner, repo] = repoId.split('/');
   679→      if (!owner || !repo) throw new Error('repoId must be in "owner/repo" form');
   680→      const results = [];
   681→      for (const f of fileEntries) {
   682→        try {
   683→          const existingSha = await GH.getExistingSha(token, owner, repo, f.path, branch);
   684→          await GH.uploadFile(token, owner, repo, f.path, f.content || '', branch, commitMsg, existingSha);
   685→          results.push({ path: f.path, ok: true });
   686→        } catch (e) {
   687→          results.push({ path: f.path, ok: false, error: e.message });
   688→        }
   689→      }
   690→      return { repoId, branch, pushed: results };
   691→    }
   692→  }

   694→  /**
   695→   * Pull a remote repo into the IDE as a new project.
   696→   * opts: { repoId, type?, branch?, projectName? }
   697→   */
   698→  async function pullProject(provider, opts) {
   699→    const token = await getToken(provider);
   700→    if (!token) throw new Error(`Not connected to ${PROVIDERS[provider].name}`);
   701→
   702→    let { repoId, branch = 'main', projectName } = opts;
   703→    if (!repoId) throw new Error('repoId required');
   704→
   705→    if (provider === 'huggingface') {
   706→      const type = opts.type || 'model';
   707→      const tree = await HF.listFiles(token, repoId, type, branch);
   708→      const files = (tree || []).filter(n => n.type === 'file');
   709→      if (!projectName) projectName = repoId.split('/').pop();
   710→
   711→      // Create IDE project
   712→      const existing = await IDEStorage.Projects.get(projectName);
   713→      if (existing) {
   714→        throw new Error(`Project "${projectName}" already exists. Pick a different name.`);
   715→      }
   716→      const project = await IDEStorage.Projects.create(projectName);
   717→
   718→      // Pull each file
   719→      const pulled = [];
   720→      for (const f of files) {
   721→        try {
   722→          const content = await HF.downloadFile(token, repoId, type, f.path, branch);
   723→          await IDEStorage.Files.create({
   724→            id: IDEStorage.generateId(),
   725→            project: projectName,
   726→            name: f.path.split('/').pop(),
   727→            path: f.path,
   728→            type: 'file',
   729→            content,
   730→            size: new Blob([content]).size,
   731→            updatedAt: Date.now()
   732→          });
   733→          // Mirror to OfflineKit (OPFS)
   734→          if (window.OfflineKit) {
   735→            try { await OfflineKit.write(projectName, f.path, content); } catch (_) {}
   736→          }
   737→          pulled.push({ path: f.path, ok: true });
   738→        } catch (e) {
   739→          pulled.push({ path: f.path, ok: false, error: e.message });
   740→        }
   741→      }
   742→      await IDEStorage.Projects.updateFileCount(projectName);
   743→      return { projectName, repoId, type, branch, pulled };
   744→    } else {
   745→      const [owner, repo] = repoId.split('/');
   746→      if (!owner || !repo) throw new Error('repoId must be "owner/repo"');
   747→      const files = await GH.listFiles(token, owner, repo, branch);
   748→      if (!projectName) projectName = repo;
   749→
   750→      const existing = await IDEStorage.Projects.get(projectName);
   751→      if (existing) {
   752→        throw new Error(`Project "${projectName}" already exists. Pick a different name.`);
   753→      }
   754→      const project = await IDEStorage.Projects.create(projectName);
   755→
   756→      const pulled = [];
   757→      for (const f of files) {
   758→        // Skip large files > 1MB to avoid blowing up memory
   759→        if (f.size > 1_000_000) {
   760→          pulled.push({ path: f.path, ok: false, error: 'Skipped (>1MB)' });
   761→          continue;
   762→        }
   763→        try {
   764→          const content = await GH.downloadFile(token, owner, repo, f.path, branch);
   765→          await IDEStorage.Files.create({
   766→            id: IDEStorage.generateId(),
   767→            project: projectName,
   768→            name: f.path.split('/').pop(),
   769→            path: f.path,
   770→            type: 'file',
   771→            content,
   772→            size: new Blob([content]).size,
   773→            updatedAt: Date.now()
   774→          });
   775→          if (window.OfflineKit) {
   776→            try { await OfflineKit.write(projectName, f.path, content); } catch (_) {}
   777→          }
   778→          pulled.push({ path: f.path, ok: true });
   779→        } catch (e) {
   780→          pulled.push({ path: f.path, ok: false, error: e.message });
   781→        }
   782→      }
   783→      await IDEStorage.Projects.updateFileCount(projectName);
   784→      return { projectName, repoId, branch, pulled };
   785→    }
   786→  }

   788→  function slugify(name) {
   789→    return String(name || '').toLowerCase()
   790→      .replace(/[^a-z0-9-]+/g, '-')
   791→      .replace(/^-+|-+$/g, '')
   792→      .replace(/-{2,}/g, '-')
   793→      .slice(0, 64);
   794→  }

  1431→      const expiresAt = Date.now() + (code.expires_in || 900) * 1000;
  1432→      const poll = async () => {
  1433→        if (Date.now() > expiresAt) {
  1434→          status.innerHTML = '<p class="error">Device flow expired. Try again.</p>';
  1435→          startBtn.disabled = false;
  1436→          return;
  1437→        }
  1438→        try {
  1439→          const r = await GH.pollDeviceFlow(clientId, code.device_code);
  1440→          if (r.access_token) {
  1441→            status.innerHTML = '<p class="success">Authorized! Connecting...</p>';
  1442→            await connect('github', r.access_token);
  1443→            status.classList.add('hidden');
  1444→            if (window.IDEUtils && IDEUtils.showToast) {
  1445→              IDEUtils.showToast(`Connected to GitHub as ${state.status.github.username}`, 'success');
  1446→            }
  1447→            renderConnectedView('github');
  1448→            startBtn.disabled = false;
  1449→            return;
  1450→          }
  1451→          if (r.error === 'authorization_pending') {
  1452→            // Continue polling
  1453→            setTimeout(poll, interval);
  1454→          } else if (r.error === 'slow_down') {
  1455→            setTimeout(poll, interval + 5000);
  1456→          } else if (r.error === 'expired_token') {
  1457→            status.innerHTML = '<p class="error">Code expired. Try again.</p>';
  1458→            startBtn.disabled = false;
  1459→          } else {
  1460→            status.innerHTML = `<p class="error">${escapeHtml(r.error_description || r.error || 'Unknown error')}</p>`;
  1461→            startBtn.disabled = false;
  1462→          }
  1463→        } catch (e) {
  1464→          status.innerHTML = `<p class="error">${escapeHtml(e.message)}</p>`;
  1465→          startBtn.disabled = false;
  1466→        }
  1467→      };
  1468→      setTimeout(poll, interval);
  1469→    } catch (e) {
  1470→      status.innerHTML = `<p class="error">${escapeHtml(e.message)}</p>`;
  1471→      startBtn.disabled = false;
  1472→    }
  1473→  }

  1502→  function escapeHtml(s) {
  1503→    return String(s == null ? '' : s)
  1504→      .replace(/&/g, '&amp;')
  1505→      .replace(/</g, '&lt;')
  1506→      .replace(/>/g, '&gt;')
  1507→      .replace(/"/g, '&quot;')
  1508→      .replace(/'/g, '&#39;');
  1509→  }

  1511→  function formatRelative(ts) {
  1512→    if (!ts) return '';
  1513→    const d = new Date(ts);
  1514→    if (isNaN(d.getTime())) return '';
  1515→    const now = Date.now();
  1516→    const diff = now - d.getTime();
  1517→    const mins = Math.floor(diff / 60000);
  1518→    if (mins < 1) return 'just now';
  1519→    if (mins < 60) return `${mins}m ago`;
  1520→    const hours = Math.floor(mins / 60);
  1521→    if (hours < 24) return `${hours}h ago`;
  1522→    const days = Math.floor(hours / 24);
  1523→    if (days < 30) return `${days}d ago`;
  1524→    const months = Math.floor(days / 30);
  1525→    if (months < 12) return `${months}mo ago`;
  1526→    return `${Math.floor(months / 12)}y ago`;
  1527→  }
  1528→
  1529→  function formatSize(bytes) {
  1530→    if (bytes == null) return '';
  1531→    if (bytes < 1024) return `${bytes} B`;
  1532→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  1533→    if (bytes < 1024 * 1024 * 1024) return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
  1534→    return `${(bytes / 1024 / 1024 / 1024).toFixed(1)} GB`;
  1535→  }
