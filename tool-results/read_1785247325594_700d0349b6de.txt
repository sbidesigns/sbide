    47→    inputContainer = container.querySelector('.chat-input-container');
    48→    inputElement = document.getElementById('chat-input');  // textarea
    49→    sendButton = document.getElementById('send-message-btn');  // button
    50→    
    51→    if (!messagesContainer) {
    52→      console.warn('ChatWindow: #chat-messages not found');
    53→    }
    54→    if (!inputElement) {
    55→      console.warn('ChatWindow: #chat-input not found');
    56→    }
    57→    if (!sendButton) {
    58→      console.warn('ChatWindow: #send-message-btn not found');
    59→    }
    60→    
    61→    // Setup event listeners
    62→    setupEventListeners();
    63→    
    64→    // Subscribe to state changes
    65→    if (window.IDEState) {
    66→      IDEState.subscribe((state, prevState) => {
    67→        // Handle new messages
    68→        if (state.messages !== prevState.messages) {
    69→          renderMessages(state.messages);
    70→        }
    71→        
    72→        // Handle streaming state changes
    73→        if (state.isStreaming !== prevState.isStreaming) {
    74→          setStreaming(state.isStreaming);
    75→        }
    76→        
    77→        // Handle project change - load project messages
    78→        if (state.currentProject?.name !== prevState.currentProject?.name) {
    79→          renderMessages(state.messages);
    80→        }
    81→      });
    82→      
    83→      // Initial render with existing messages
    84→      renderMessages(IDEState.get('messages') || []);
    85→    }
    86→  }

   164→  /**
   165→   * Setup event listeners
   166→   */
   167→  function setupEventListeners() {
   168→    // Send button click
   169→    if (sendButton) {
   170→      sendButton.addEventListener('click', () => sendMessage());
   171→    }
   172→    
   173→    // Input keydown (Enter to send)
   174→    if (inputElement) {
   175→      inputElement.addEventListener('keydown', (e) => {
   176→        if (e.key === 'Enter' && !e.shiftKey) {
   177→          e.preventDefault();
   178→          sendMessage();
   179→        }
   180→        
   181→        // Auto-resize textarea
   182→        autoResizeTextarea();
   183→      });
   184→      
   185→      // Auto-resize + refresh send-button disabled state on every keystroke.
   186→      // The send button's disabled state now tracks input content (not project
   187→      // existence) so the user can always click it when there's text to send.
   188→      inputElement.addEventListener('input', () => {
   189→        autoResizeTextarea();
   190→        updateInputState();
   191→      });
   192→    }
   193→  }

   198→  function autoResizeTextarea() {

   201→    inputElement.style.height = 'auto';
   202→    inputElement.style.height = Math.min(inputElement.scrollHeight, 200) + 'px';

   209→  /**
   210→   * Send a message
   211→   */
   212→  async function sendMessage() {
   213→    if (!inputElement || isStreaming) return;
   214→    
   215→    let content = inputElement.value.trim();
   216→    if (!content) return;
   217→    
   218→    // Input sanitization: remove null bytes and control characters (except newlines/tabs)
   219→    content = content.replace(/[\x00-\x08\x0B\x0C\x0E-\x1F]/g, '');
   220→    
   221→    // Validate message length (prevent abuse)
   222→    const MAX_MESSAGE_LENGTH = 100000; // ~100K chars (~25K tokens)
   223→    if (content.length > MAX_MESSAGE_LENGTH) {
   224→      content = content.substring(0, MAX_MESSAGE_LENGTH);
   225→      if (IDEUtils) IDEUtils.showToast('Message was truncated (too long)', 'warning');
   226→    }
   227→    
   228→    // Minimum content check after sanitization
   229→    if (!content.trim()) return;
   230→    
   231→    // Clear input
   232→    inputElement.value = '';
   233→    inputElement.style.height = 'auto';
   234→    
   235→    // Dispatch event for toolbar state management (e.g., auto-disable web search)
   236→    window.dispatchEvent(new CustomEvent('chat:message-sent', { detail: { content } }));
   237→    
   238→    // Create user message
   239→    const userMessage = {
   240→      id: IDEUtils?.generateId() || Date.now().toString(),
   241→      role: 'user',
   242→      content,
   243→      timestamp: Date.now()
   244→    };
   245→    
   246→    // Add to state
   247→    if (window.IDEState) {
   248→      IDEState.addMessage(userMessage);
   249→    }
   250→    
   251→    // Set streaming state (with double-check to prevent race conditions)
   252→    if (isStreaming) {
   253→      console.warn('sendMessage called while already streaming, ignoring');
   254→      return;
   255→    }
   256→    setStreaming(true);
   257→    
   258→    // Create assistant message placeholder
   259→    const assistantMessage = {
   260→      id: IDEUtils?.generateId() || (Date.now() + 1).toString(),
   261→      role: 'assistant',
   262→      content: '',
   263→      timestamp: Date.now(),
   264→      isStreaming: true,
   265→      toolCalls: [],
   266→      backend: null   // populated by onBackend callback (cloud/ollama/webllm)
   267→    };
   268→    
   269→    if (window.IDEState) {
   270→      IDEState.addMessage(assistantMessage);
   271→    }
   272→    
   273→    // Call send callback or use API directly
   274→    try {
   275→      if (onSendMessage) {
   276→        await onSendMessage(content, {
   277→          onChunk: (text) => handleStreamChunk(text),
   278→          onToolCall: (tool, args) => handleToolCall(tool, args),
   279→          onToolResult: (result, success) => handleToolResult(result, success),
   280→          onError: (error) => handleStreamError(error),
   281→          onBackend: (backend) => handleBackend(assistantMessage.id, backend),
   282→          onDone: () => handleStreamDone(assistantMessage.id)
   283→        });
   284→      } else {
   285→        // Default: use API client
   286→        await streamWithAPI(content, assistantMessage.id);
   287→      }
   288→    } catch (error) {
   289→      handleStreamError(error);
   290→    }
   291→  }

   293→  /**
   294→   * Record which LLM backend served this message and re-render
   295→   * so the header can show a "local" / "ollama" / "webllm" badge.
   296→   */
   297→  function handleBackend(messageId, backend) {
   298→    if (!window.IDEState || !backend) return;
   299→    const messages = IDEState.get('messages') || [];
   300→    const idx = messages.findIndex(m => m.id === messageId);
   301→    if (idx === -1) return;
   302→    const updated = { ...messages[idx], backend };
   303→    const next = messages.slice();
   304→    next[idx] = updated;
   305→    IDEState.set({ messages: next });
   306→    // Patch the DOM header in-place to avoid full re-render flicker
   307→    const el = document.querySelector(`.message-wrapper[data-message-id="${messageId}"] .message-role`);
   308→    if (el) {
   309→      const label = backend.type === 'ollama' ? 'AI Assistant · Ollama'
   310→                  : backend.type === 'webllm' ? 'AI Assistant · WebLLM'
   311→                  : backend.type === 'cloud'  ? 'AI Assistant'
   312→                  : backend.type === 'demo'   ? 'AI Assistant · Demo'
   313→                  : 'AI Assistant';
   314→      el.textContent = label;
   315→      el.dataset.backend = backend.type || '';
   316→    }
   317→  }

   319→  /**
   320→   * Stream using API client
   321→   */
   322→  async function streamWithAPI(content, messageId) {
   323→    const state = window.IDEState ? IDEState.getState() : {};
   324→    
   325→    currentStreamController = await IDEAPI.sendMessageStream({
   326→      message: content,
   327→      messages: state.messages?.filter(m => !m.isStreaming) || [],
   328→      projectName: state.currentProject?.name || null,
   329→      projectState: state.projectState || {},
   330→      onChunk: (text) => handleStreamChunk(text),
   331→      onToolCall: (tool, args) => handleToolCall(tool, args),
   332→      onToolResult: (result, success) => handleToolResult(result, success),
   333→      onError: (error) => handleStreamError(error),
   334→      onBackend: (backend) => handleBackend(messageId, backend),
   335→      onDone: () => handleStreamDone(messageId)
   336→    });
   337→  }
   338→
   339→  /**
   340→   * Handle incoming stream chunk
   341→   */
   342→  function handleStreamChunk(text) {
   343→    if (!text) return;
   344→    
   345→    if (window.IDEState) {
   346→      const messages = IDEState.get('messages') || [];
   347→      const lastMessage = messages[messages.length - 1];
   348→      
   349→      if (lastMessage && lastMessage.role === 'assistant') {
   350→        IDEState.updateLastMessage(lastMessage.content + text);
   351→      }
   352→    }
   353→  }
   354→
   355→  /**
   356→   * Handle tool call from AI
   357→   */
   358→  function handleToolCall(tool, args) {
   359→    // Add tool call to current message
   360→    const messages = IDEState?.get('messages') || [];
   361→    const lastMessage = messages[messages.length - 1];
   362→    
   363→    if (lastMessage && lastMessage.role === 'assistant') {
   364→      const updatedToolCalls = [...(lastMessage.toolCalls || []), {
   365→        id: IDEUtils?.generateId() || Date.now().toString(),
   366→        tool,
   367→        args,
   368→        status: 'running',
   369→        result: null
   370→      }];
   371→      
   372→      // We need to update the message with tool calls info
   373→      // This is handled by re-rendering which reads from state
   374→      if (window.IDEState) {
   375→        const updatedMessage = { ...lastMessage, toolCalls: updatedToolCalls };
   376→        const msgIndex = messages.indexOf(lastMessage);
   377→        const newMessages = [...messages];
   378→        newMessages[msgIndex] = updatedMessage;
   379→        IDEState.setMessages(newMessages);
   380→      }
   381→    }
   382→    
   383→    // Notify parent
   384→    if (onToolAction) {
   385→      onToolAction({ type: 'call', tool, args });
   386→    }
   387→  }
   388→
   389→  /**
   390→   * Handle tool result
   391→   */
   392→  function handleToolResult(result, success) {
   393→    // Update tool call status in message
   394→    const messages = IDEState?.get('messages') || [];
   395→    const lastMessage = messages[messages.length - 1];
   396→    
   397→    if (lastMessage && lastMessage.toolCalls) {
   398→      const toolCalls = [...lastMessage.toolCalls];
   399→      const lastCall = toolCalls[toolCalls.length - 1];
   400→      if (lastCall && lastCall.status === 'running') {
   401→        lastCall.status = success ? 'complete' : 'error';
   402→        lastCall.result = result;
   403→        
   404→        if (window.IDEState) {
   405→          const updatedMessage = { ...lastMessage, toolCalls };
   406→          const msgIndex = messages.indexOf(lastMessage);
   407→          const newMessages = [...messages];
   408→          newMessages[msgIndex] = updatedMessage;
   409→          IDEState.setMessages(newMessages);
   410→        }
   411→      }
   412→    }
   413→    
   414→    // Notify parent
   415→    if (onToolAction) {
   416→      onToolAction({ type: 'result', result, success });
   417→    }
   418→  }

   420→  /**
   421→   * Handle stream error
   422→   */
   423→  function handleStreamError(error) {
   424→    console.error('Stream error:', error);
   425→    
   426→    // Add error message
   427→    if (window.IDEState) {
   428→      const errorMessage = {
   429→        id: IDEUtils?.generateId() || Date.now().toString(),
   430→        role: 'assistant',
   431→        content: `**Error:** ${error.message || 'Something went wrong'}`,
   432→        timestamp: Date.now(),
   433→        isError: true
   434→      };
   435→      
   436→      // Remove streaming placeholder and add error
   437→      const messages = IDEState.get('messages').filter(m => !m.isStreaming);
   438→      IDEState.setMessages([...messages, errorMessage]);
   439→    }
   440→    
   441→    setStreaming(false);
   442→  }
   443→
   444→  /**
   445→   * Handle stream completion
   446→   */
   447→  function handleStreamDone(messageId) {
   448→    // Finalize the streaming message
   449→    if (window.IDEState) {
   450→      const messages = IDEState.get('messages') || [];
   451→      const message = messages.find(m => m.id === messageId);
   452→      
   453→      if (message) {
   454→        const finalizedMessage = { 
   455→          ...message, 
   456→          isStreaming: false,
   457→          timestamp: Date.now()
   458→        };
   459→        
   460→        const msgIndex = messages.indexOf(message);
   461→        const newMessages = [...messages];
   462→        newMessages[msgIndex] = finalizedMessage;
   463→        IDEState.setMessages(newMessages);
   464→      }
   465→    }
   466→    
   467→    setStreaming(false);
   468→  }

   496→  /**

   504→    // Update send button

   514→        sendButton.setAttribute('aria-label', 'Send message');
   515→        sendButton.onclick = () => sendMessage();
   516→        sendButton.classList.remove('btn-danger');
   517→        sendButton.classList.add('btn-primary');
   518→      }
   519→    }
   520→    
   521→    // Show/hide stop streaming button (separate button in HTML)
   522→    const stopBtn = document.getElementById('stop-streaming-btn');
   523→    if (stopBtn) {
   524→      stopBtn.classList.toggle('hidden', !isStreaming);
   525→    }
   526→    
   527→    // Enable/disable action buttons when no project selected or no messages
   528→    const hasProject = !!IDEState?.get('currentProject');
   529→    const hasMessages = (IDEState?.get('messages')?.length || 0) > 0;
   530→    
   531→    if (sendButton) {
   532→      // Send button is enabled whenever there's text in the input (or while
   533→    // streaming, so the user can click to stop). Previously this required a
   534→    // project to be selected, which made the button unclickable on a fresh
   535→    // session — Enter still worked because the textarea keydown handler did
   536→    // not check disabled state.
   537→    const hasText = !!(inputElement && inputElement.value.trim());
   538→    sendButton.disabled = !hasText && !isStreaming;
   539→    }
   540→    
   541→    // Update Clear/Export buttons
   542→    const clearBtn = document.getElementById('clear-chat-btn');
   543→    const exportBtn = document.getElementById('export-chat-btn');

   552→      if (contextResults.length > 0) {
   553→        contextIndicator.style.display = 'flex';
   554→        const countEl = contextIndicator.querySelector('.context-count');
   555→        if (countEl) countEl.textContent = contextResults.length;
   556→      } else {

   558→      }
   559→    }
   560→  }

   563→  // Message Rendering

   565→  
   566→  /**
   567→   * Render all messages
   568→   */
   569→  function renderMessages(messages) {
   570→    if (!messagesContainer) return;
   571→    
   572→    // Check if we should do full re-render or incremental
   573→    const shouldFullRender = messagesContainer.children.length === 0 || 
   574→                             messages.length <= messagesContainer.children.length;
   575→    
   576→    if (shouldFullRender) {
   577→      fullRender(messages);
   578→    } else {
   579→      incrementalRender(messages);
   580→    }
   581→    
   582→    scrollToBottom();
   583→  }
   584→
   585→  /**
   586→   * Full render of all messages
   587→   */
   588→  function fullRender(messages) {
   589→    messagesContainer.innerHTML = '';
   590→    
   591→    if (!messages || messages.length === 0) {
   592→      renderEmptyState();
   593→      return;
   594→    }
   595→    
   596→    messages.forEach((message, index) => {
   597→      const el = renderMessage(message, index);
   598→      messagesContainer.appendChild(el);
   599→    });
   600→  }
   601→
   602→  /**
   603→   * Incremental render (add only new messages)
   604→   */
   605→  function incrementalRender(messages) {
   606→    const existingCount = messagesContainer.children.length;
   607→    
   608→    for (let i = existingCount; i < messages.length; i++) {
   609→      const el = renderMessage(messages[i], i);
   610→      messagesContainer.appendChild(el);
   611→    }
   612→    
   613→    // Update last message if it's streaming
   614→    const lastMessage = messages[messages.length - 1];
   615→    if (lastMessage && lastMessage.isStreaming) {
   616→      const lastEl = messagesContainer.lastElementChild;

   780→  /**
   781→   * Format message content (markdown support)
   782→   */
   783→  function formatMessageContent(content) {
   784→    if (!content) return '';
   785→    
   786→    // Basic markdown-like formatting
   787→    let formatted = IDEUtils?.escapeHtml(content) || content;
   788→    
   789→    // Code blocks
   790→    formatted = formatted.replace(/```(\w+)?\n([\s\S]*?)```/g, (match, lang, code) => {
   791→      return `<pre class="code-block"><code class="language-${lang || 'plaintext'}">${code.trim()}</code></pre>`;
   792→    });
   793→    
   794→    // Inline code
   795→    formatted = formatted.replace(/`([^`]+)`/g, '<code>$1</code>');
   796→    
   797→    // Bold
   798→    formatted = formatted.replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>');
   799→    
   800→    // Italic
   801→    formatted = formatted.replace(/\*([^*]+)\*/g, '<em>$1</em>');
   802→    
   803→    // Line breaks
   804→    formatted = formatted.replace(/\n/g, '<br>');
   805→    
   806→    return formatted;
   807→  }

   950→  /**
   951→   * Generate contextual suggested actions based on message content
   952→   * @param {Object} message - The assistant message object
   953→   * @returns {Array} Array of suggestion objects { id, label, icon, action, prompt }
   954→   */
   955→  function generateSuggestedActions(message) {
   956→    const content = (message.content || '').toLowerCase();
   957→    const hasCode = /```|function |const |let |var |class |import |export|<div|<span|{/.test(content);
   958→    const hasFiles = message.toolCalls && message.toolCalls.some(tc => 
   959→      ['read_file', 'write_file', 'create_file', 'delete_file', 'list_files'].includes(tc.tool)
   960→    );
   961→    const hasWebSearch = message.toolCalls && message.toolCalls.some(tc => 
   962→      tc.tool === 'web_search'
   963→    );
   964→    const hasCommand = message.toolCalls && message.toolCalls.some(tc => 
   965→      tc.tool === 'execute_command'
   966→    );
   967→
   968→    const suggestions = [];
   969→
   970→    // Base suggestions for all responses
   971→    if (hasCode) {
   972→      suggestions.push({
   973→        id: 'explain-code',
   974→        label: 'Explain code',
   975→        icon: 'code',
   976→        action: 'prompt',
   977→        prompt: 'Can you explain this code in more detail?'
   978→      });
   979→      suggestions.push({
   980→        id: 'refactor',
   981→        label: 'Refactor',
   982→        icon: 'refresh',
   983→        action: 'prompt',
   984→        prompt: 'Can you suggest improvements or refactor this code?'
   985→      });
   986→    }
   987→
   988→    if (hasFiles) {
   989→      suggestions.push({
   990→        id: 'show-files',
   991→        label: 'Show related files',
   992→        icon: 'folder',
   993→        action: 'prompt',
   994→        prompt: 'What other files should I look at for this feature?'
   995→      });
   996→    }
   997→
   998→    // Check if response was about errors or issues
   999→    if (/error|bug|fix|issue|problem|fail/.test(content)) {
  1000→      suggestions.push({
  1001→        id: 'debug-more',
  1002→        label: 'Debug further',
  1003→        icon: 'search',
  1004→        action: 'prompt',
  1005→        prompt: 'Help me debug this issue step by step'
  1006→      });
  1007→    }
  1008→
  1009→    // Check if response contained explanations
  1010→    if (/how to|how do i|steps|tutorial|guide|implement/.test(content)) {
  1011→      suggestions.push({
  1012→        id: 'examples',
  1013→        label: 'Show examples',
  1014→        icon: 'lightbulb',
  1015→        action: 'prompt',
  1016→        prompt: 'Can you show me practical examples of this?'
  1017→      });
  1018→    }
  1019→
  1020→    // Always add these core suggestions (limit to 4 total)
  1021→    if (suggestions.length < 4) {
  1022→      suggestions.push({
  1023→        id: 'elaborate',
  1024→        label: 'Tell me more',
  1025→        icon: 'message-circle',
  1026→        action: 'prompt',
  1027→        prompt: 'Can you elaborate on that?'
  1028→      });
  1029→    }
  1030→
  1031→    if (suggestions.length < 4) {
  1032→      suggestions.push({
  1033→        id: 'next-steps',
  1034→        label: 'Next steps',
  1035→        icon: 'arrow-right',
  1036→        action: 'prompt',
  1037→        prompt: 'What should I do next?'
  1038→      });
  1039→    }
  1040→
  1041→    if (suggestions.length < 4 && !hasWebSearch) {
  1042→      suggestions.push({
  1043→        id: 'search',
  1044→        label: 'Search web',
  1045→        icon: 'globe',
  1046→        action: 'prompt',
  1047→        prompt: 'Search the web for more information about this topic'
  1048→      });
  1049→    }
  1050→
  1051→    // Return max 4 suggestions
  1052→    return suggestions.slice(0, 4);
  1053→  }

  1160→      return;
  1161→    }
  1162→    
  1163→    const conversation = messages
  1164→      .map(m => `[${new Date(m.timestamp).toISOString()}] ${m.role.toUpperCase()}: ${m.content}`)
  1165→      .join('\n\n');
  1166→    
  1167→    const blob = new Blob([conversation], { type: 'text/plain' });
  1168→    const url = URL.createObjectURL(blob);
  1169→    const a = document.createElement('a');
  1170→    a.href = url;
  1171→    a.download = `conversation-${Date.now()}.txt`;
  1172→    a.click();
  1173→    URL.revokeObjectURL(url);
  1174→    
  1175→    if (IDEUtils) IDEUtils.showToast('Conversation exported', 'success');
  1176→  }

  1196→  /**
  1197→   * Rollback - remove message and all after it
  1198→   */
  1199→  function rollbackMessage(messageId) {
  1200→    const messages = IDEState?.get('messages') || [];
  1201→    const index = messages.findIndex(m => m.id === messageId);
  1202→    if (index === -1) return;
  1203→    
  1204→    // Create backup before rollback
  1205→    const project = IDEState?.get('currentProject');
  1206→    if (project) {
  1207→      fetch('/api/ide/versions', {
  1208→        method: 'POST',
  1209→        headers: { 'Content-Type': 'application/json' },
  1210→        body: JSON.stringify({ 
  1211→          project: project.name,
  1212→          createArchive: true,
  1213→          suffix: '.2',
  1214→          description: 'Pre-rollback backup'
  1215→        })
  1216→      }).catch(e => console.error('Pre-rollback backup failed:', e));
  1217→    }
  1218→    
  1219→    // Keep only messages before this one
  1220→    const truncated = messages.slice(0, index);
  1221→    IDEState.setMessages(truncated);
  1222→    
  1223→    if (IDEUtils) IDEUtils.showToast('Rolled back conversation', 'success');
  1224→  }
