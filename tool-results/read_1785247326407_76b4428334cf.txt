     1→/**
     2→ * SBIDE - File Tree Component
     3→ * Renders and manages the file/folder tree in the sidebar
     4→ * Enhanced with toolbar, context menu, and full CRUD operations
     5→ */
     6→
     7→const FileTreeComponent = (() => {
     8→  // ============================================
     9→  // State
    10→  // ============================================
    11→  
    12→  let container = null;
    13→  let treeData = [];
    14→  let expandedFolders = new Set();
    15→  let selectedFile = null;
    16→  let onFileSelect = null;
    17→  let onFileAction = null;
    18→  
    19→  // Context menu state
    20→  let contextMenu = null;
    21→  let contextMenuItem = null;
    22→  
    23→  // Dialog state
    24→  let dialog = null;
    25→  let dialogCallback = null;
    26→  
    27→  // Drag state
    28→  let dragItem = null;
    29→
    30→  // Clipboard state for Copy/Cut/Paste.
    31→  // { item: <tree node>, mode: 'copy' | 'cut', projectName: <string> }
    32→  // Persisted to localStorage so clipboard survives page reloads.
    33→  const CLIPBOARD_PREF_KEY = 'ide:file-tree-clipboard';
    34→  let clipboard = loadClipboard();
    35→  
    36→  // Sort state (controlled by the HTML toolbar buttons in #panel-files header)
    37→  // sortKey: 'name' | 'date'   sortDir: 'asc' | 'desc'
    38→  let sortKey = 'name';
    39→  let sortDir = 'asc';
    40→
    41→  // ============================================
    42→  // Initialization
    43→  // ============================================
    44→  
    45→  /**
    46→   * Initialize file tree component
    47→   * @param {HTMLElement} element - Container element for the tree
    48→   * @param {Object} options - Configuration options
    49→   */
    50→  function init(element, options = {}) {
    51→    if (!element) {
    52→      console.error('FileTree: Container element required');
    53→      return;
    54→    }
    55→    
    56→    container = element;
    57→    onFileSelect = options.onFileSelect || null;
    58→    onFileAction = options.onFileAction || null;
    59→    
    60→    // Initialize with empty state (includes toolbar)
    61→    render();
    62→    
    63→    // Subscribe to state changes for file refresh
    64→    if (window.IDEState) {
    65→      IDEState.subscribe((state, prevState) => {
    66→        if (state.fileRefreshKey !== prevState.fileRefreshKey) {
    67→          loadFilesForProject(state.currentProject);
    68→        }
    69→        if (state.activeFile !== prevState.activeFile) {
    70→          setSelectedFile(state.activeFile);
    71→        }
    72→      });
    73→    }
    74→    
    75→    // Close context menu on outside click
    76→    document.addEventListener('click', (e) => {
    77→      if (contextMenu && !contextMenu.contains(e.target)) {
    78→        closeContextMenu();
    79→      }
    80→    });
    81→    
    82→    // Close dialog on escape
    83→    document.addEventListener('keydown', (e) => {
    84→      if (e.key === 'Escape') {
    85→        closeContextMenu();
    86→        closeDialog();
    87→      }
    88→    });
    89→  }
    90→
    91→  // ============================================
    92→  // Data Loading
    93→  // ============================================
    94→  
    95→  /**
    96→   * Load files for a project and render tree
    97→   */
    98→  async function loadFilesForProject(project) {
    99→    if (!project || !project.name) {
   100→      treeData = [];
   101→      render();
   102→      return;
   103→    }
   104→    
   105→    try {
   106→      const files = await IDEStorage.Files.getByProject(project.name);
   107→      treeData = buildTree(files);
   108→      
   109→      // Auto-expand root level folders
   110→      treeData.forEach(item => {
   111→        if (item.type === 'folder') {
   112→          expandedFolders.add(item.path);
   113→        }
   114→      });
   115→      
   116→      render();
   117→    } catch (error) {
   118→      console.error('Failed to load files:', error);
   119→      treeData = [];
   120→      render();
   121→    }
   122→  }
   123→
   124→  /**
   125→   * Build tree structure from flat file array
   126→   */
   127→  function buildTree(files) {
   128→    const map = new Map();
   129→    const root = [];
   130→    
   131→    // Create map entries
   132→    files.forEach(file => {
   133→      map.set(file.path, { ...file, children: [] });
   134→    });
   135→    
   136→    // Build hierarchy
   137→    files.forEach(file => {
   138→      const node = map.get(file.path);
   139→      if (!file.path || file.path === '') {
   140→        root.push(node);
   141→      } else {
   142→        const parentPath = file.path.substring(0, file.path.lastIndexOf('/'));
   143→        const parent = map.get(parentPath);
   144→        if (parent) {
   145→          parent.children.push(node);
   146→        } else {
   147→          root.push(node);
   148→        }
   149→      }
   150→    });
   151→    
   152→    // Sort: folders first, then alphabetically
   153→    sortNodes(root);
   154→    
   155→    return root.length === 1 && root[0].path === '' ? root[0].children : root;
   156→  }
   157→
   158→  /**
   159→   * Sort nodes recursively, respecting the current sortKey/sortDir.
   160→   * Folders always come before files within their parent (regardless of sort mode),
   161→   * then items within each type group are sorted by name or by updatedAt date.
   162→   */
   163→  function sortNodes(nodes) {
   164→    nodes.sort((a, b) => {
   165→      // Folders first (always)
   166→      if (a.type !== b.type) return a.type === 'folder' ? -1 : 1;
   167→      
   168→      // Same type: apply the active sort
   169→      if (sortKey === 'date') {
   170→        const ta = a.updatedAt || 0;
   171→        const tb = b.updatedAt || 0;
   172→        if (ta !== tb) {
   173→          return sortDir === 'desc' ? (tb - ta) : (ta - tb);
   174→        }
   175→        // Tie-break by name ascending (keeps sort stable)
   176→        return a.name.localeCompare(b.name);
   177→      }
   178→      
   179→      // Default: name
   180→      const cmp = a.name.localeCompare(b.name, undefined, { numeric: true, sensitivity: 'base' });
   181→      return sortDir === 'desc' ? -cmp : cmp;
   182→    });
   183→    
   184→    nodes.forEach(node => {
   185→      if (node.children && node.children.length > 0) {
   186→        sortNodes(node.children);
   187→      }
   188→    });
   189→  }
   190→  
   191→  /**
   192→   * Update the sort mode and re-render.
   193→   * @param {'name'|'date'} key
   194→   * @param {'asc'|'desc'} dir
   195→   */
   196→  function setSort(key, dir) {
   197→    sortKey = key;
   198→    sortDir = dir;
   199→    // Re-sort the in-memory tree without re-fetching from IndexedDB
   200→    sortNodes(treeData);
   201→    render();
   202→    // Notify the panel-header toggle buttons so they can sync their visual state
   203→    // (active class / aria-pressed / icon / title). sidebar.js listens for this.
   204→    document.dispatchEvent(new CustomEvent('ide:file-sort-changed', {
   205→      detail: { key: sortKey, dir: sortDir }
   206→    }));
   207→  }
   208→  
   209→  /**
   210→   * Get current sort state (used by the HTML toggle buttons to update their
   211→   * aria-pressed / icon state).
   212→   */
   213→  function getSort() {
   214→    return { key: sortKey, dir: sortDir };
   215→  }
   216→
   217→  // ============================================
   218→  // Rendering
   219→  // ============================================
   220→  
   221→  /**
   222→   * Render the complete file tree.
   223→   * The New File / New Folder / Refresh / sort-toggle buttons live in the
   224→   * panel-header in index.html (wired by sidebar.js), so we no longer render
   225→   * our own toolbar here.
   226→   */
   227→  function render() {
   228→    if (!container) return;
   229→    
   230→    // Clear container
   231→    container.innerHTML = '';
   232→    
   233→    // Show empty state if no data
   234→    if (treeData.length === 0) {
   235→      renderEmptyState();
   236→      // Even in empty state, allow dropping into root (e.g. dragging a file
   237→      // out of a subfolder into root when the tree is collapsed).
   238→      attachRootDropZone(container);
   239→      return;
   240→    }
   241→    
   242→    // Create tree container
   243→    const treeList = document.createElement('ul');
   244→    treeList.className = 'file-tree-list';
   245→    treeList.setAttribute('role', 'tree');
   246→    treeList.setAttribute('aria-label', 'Project files');
   247→    
   248→    // Create root folder wrapper - renders as a regular folder in the tree
   249→    const rootFolder = {
   250→      name: '/',  // Traditional filesystem root notation
   251→      path: '',
   252→      type: 'folder',
   253→      children: [...treeData]  // All top-level items become children of root
   254→    };
   255→    
   256→    // Render root folder at depth 0
   257→    // Its children will naturally be at depth 1, grandchildren at depth 2, etc.
   258→    const rootLi = renderTreeNode(rootFolder, 0);
   259→    treeList.appendChild(rootLi);
   260→    
   261→    container.appendChild(treeList);
   262→
   263→    // Allow dropping into empty space below the list → moves to root.
   264→    attachRootDropZone(container);
   265→  }
   266→
   267→  /**
   268→   * Make `el` accept drops that fall onto empty space (i.e. NOT on a
   269→   * .tree-row). When a drag hovers over empty space, we show a "drop into
   270→   * root" highlight; dropping moves the dragged item to the project root.
   271→   */
   272→  function attachRootDropZone(el) {
   273→    if (!el) return;
   274→
   275→    // Only attach once per container.
   276→    if (el._rootDropAttached) return;
   277→    el._rootDropAttached = true;
   278→
   279→    el.addEventListener('dragover', (e) => {
   280→      if (!dragItem) return;
   281→      // If the user is hovering over a .tree-row, let that row's own
   282→      // dragover handler take care of it — don't interfere.
   283→      const onRow = e.target.closest('.tree-row');
   284→      if (onRow) return;
   285→
   286→      e.preventDefault();
   287→      e.dataTransfer.dropEffect = 'move';
   288→      // Subtle highlight on the whole tree to signal "drop into root"
   289→      el.classList.add('root-drop-active');
   290→    });
   291→
   292→    el.addEventListener('dragleave', (e) => {
   293→      // Only clear if the pointer has actually left the container
   294→      if (!el.contains(e.relatedTarget)) {
   295→        el.classList.remove('root-drop-active');
   296→      }
   297→    });
   298→
   299→    el.addEventListener('drop', (e) => {
   300→      const onRow = e.target.closest('.tree-row');
   301→      if (onRow) {
   302→        // Let the row handler deal with this drop.
   303→        return;
   304→      }
   305→      e.preventDefault();
   306→      el.classList.remove('root-drop-active');
   307→      if (!dragItem) return;
   308→
   309→      const source = dragItem;
   310→      dragItem = null;
   311→
   312→      // Don't allow moving a root item to root (no-op)
   313→      const sourceParent = (() => {
   314→        const ls = source.path.lastIndexOf('/');
   315→        return ls > 0 ? source.path.substring(0, ls) : '';
   316→      })();
   317→      if (sourceParent === '') {
   318→        if (IDEUtils) IDEUtils.showToast(`"${source.name}" is already at root`, 'info');
   319→        return;
   320→      }
   321→
   322→      // Also fire the legacy callback
   323→      if (onFileAction) {
   324→        onFileAction({
   325→          type: 'move',
   326→          source,
   327→          target: { path: '', name: '/', type: 'folder' },
   328→          targetFolderPath: ''
   329→        });
   330→      }
   331→
   332→      confirmAndExecuteMove(source, '', { path: '', name: '/', type: 'folder' });
   333→    });
   334→  }

   391→  /**
   392→   * Render a single tree node (recursively)
   393→   */
   394→  function renderTreeNode(item, depth) {
   395→    const li = document.createElement('li');
   396→    li.className = `tree-item ${item.type}-item`;
   397→    li.setAttribute('data-path', item.path);
   398→    li.setAttribute('data-type', item.type);
   399→    li.setAttribute('role', item.type === 'folder' ? 'group' : 'treeitem');
   400→    li.style.setProperty('--depth', depth);
   401→    
   402→    const isExpanded = expandedFolders.has(item.path);
   403→    const isSelected = selectedFile && selectedFile.path === item.path;
   404→    
   405→    // Row element (clickable area)
   406→    const row = document.createElement('div');
   407→    row.className = `tree-row group ${isSelected ? 'selected' : ''}`;
   408→    row.setAttribute('aria-selected', isSelected);
   409→    row.tabIndex = 0;
   410→    
   411→    // No chevron/toggle - folder icon itself indicates expandability
   412→    // Clicking the row toggles expansion
   413→    
   414→    // Icon
   415→    const icon = document.createElement('span');
   416→    icon.className = `tree-icon ${item.type}-icon`;
   417→    icon.innerHTML = item.type === 'folder' 
   418→      ? (isExpanded ? (IDEUtils?.Icons?.folderOpen || '') : (IDEUtils?.Icons?.folder || ''))
   419→      : (IDEUtils?.getFileIcon?.(item.name) || IDEUtils?.Icons?.file || '');
   420→    row.appendChild(icon);
   421→    
   422→    // Name
   423→    const name = document.createElement('span');
   424→    name.className = 'tree-name';
   425→    name.textContent = item.name;
   426→    row.appendChild(name);
   427→    
   428→    // Actions menu (visible on hover)
   429→    const actions = document.createElement('div');
   430→    actions.className = 'tree-actions';
   431→    actions.innerHTML = `
   432→      <button class="tree-action-btn" data-action="menu" title="More actions" aria-label="Actions menu">
   433→        <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
   434→          <circle cx="12" cy="12" r="1"/><circle cx="12" cy="5" r="1"/><circle cx="12" cy="19" r="1"/>
   435→        </svg>
   436→      </button>
   437→    `;
   438→    
   439→    actions.querySelector('[data-action="menu"]').addEventListener('click', (e) => {
   440→      e.stopPropagation();
   441→      e.preventDefault();
   442→      showContextMenu(e, item);
   443→    });
   444→    
   445→    row.appendChild(actions);
   446→    
   447→    // Click handler
   448→    row.addEventListener('click', () => handleItemClick(item));
   449→    
   450→
   451→    row.addEventListener('keydown', (e) => {
   452→      if (e.key === 'Enter' || e.key === ' ') {
   453→        e.preventDefault();
   454→        handleItemClick(item);
   455→      }
   456→    });
   457→    
   458→    // Context menu
   459→    row.addEventListener('contextmenu', (e) => {
   460→      e.preventDefault();
   461→      showContextMenu(e, item);
   462→    });
   463→    
   464→    // Drag events
   465→    row.draggable = true;
   466→    row.addEventListener('dragstart', (e) => handleDragStart(e, item));
   467→    row.addEventListener('dragover', (e) => handleDragOver(e, item));
   468→    row.addEventListener('dragleave', (e) => handleDragLeave(e, item));
   469→    row.addEventListener('drop', (e) => handleDrop(e, item));
   470→    row.addEventListener('dragend', handleDragEnd);
   471→    
   472→    li.appendChild(row);
   473→    
   474→    // Render children for expanded folders
   475→    if (item.type === 'folder' && isExpanded && item.children && item.children.length > 0) {
   476→      const childContainer = document.createElement('ul');
   477→      childContainer.className = 'tree-children';
   478→      childContainer.setAttribute('role', 'group');
   479→      // Pass depth info to children container for connector line positioning
   480→      childContainer.style.setProperty('--child-depth', depth);
   481→      
   482→      item.children.forEach(child => {
   483→        const childLi = renderTreeNode(child, depth + 1);
   484→        childContainer.appendChild(childLi);
   485→      });
   486→      
   487→      li.appendChild(childContainer);
   488→    }
   489→    
   490→    return li;
   491→  }

   497→  /**
   498→   * Show context menu for an item
   499→   */
   500→  function showContextMenu(event, item) {
   501→    closeContextMenu();
   502→    
   503→    contextMenuItem = item;
   504→    
   505→    contextMenu = document.createElement('div');
   506→    contextMenu.className = 'context-menu';
   507→    contextMenu.setAttribute('role', 'menu');
   508→    
   509→    let menuItems = '';
   510→    
   511→    // For folders: add New File, New Folder options
   512→    if (item.type === 'folder') {

   514→        <div class="context-menu-item" data-action="new-file-in-folder" role="menuitem">

   525→    menuItems += `
   526→      <div class="context-menu-item" data-action="copy" role="menuitem">
   527→        <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><rect x="9" y="9" width="13" height="13" rx="2"/><path d="M5 15H4a2 2 0 01-2-2V4a2 2 0 012-2h9a2 2 0 012 2v1"/></svg>

   537→      </div>
   538→      <div class="context-menu-item ${clipboard.item ? '' : 'context-menu-disabled'}" data-action="paste-into-root" role="menuitem" ${clipboard.item ? '' : 'aria-disabled="true"'}>

   540→        Paste into root
   541→      </div>
   542→      <div class="context-menu-divider"></div>
   543→      <div class="context-menu-item" data-action="rename" role="menuitem">
   544→        <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M11 4H4a2 2 0 00-2 2v14a2 2 0 002 2h14a2 2 0 002-2v-7"/><path d="M18.5 2.5a2.121 2.121 0 013 3L12 15l-4 1 1-4 9.5-9.5z"/></svg>

   550→      </div>
   551→    `;
   552→    
   553→    contextMenu.innerHTML = menuItems;
   554→    
   555→    // Position menu
   556→    const x = event.clientX || event.pageX;

   558→    contextMenu.style.left = `${x}px`;
   559→    contextMenu.style.top = `${y}px`;
   560→    
   561→    // Adjust position if menu goes off screen
   562→    requestAnimationFrame(() => {
   563→      const rect = contextMenu.getBoundingClientRect();
   564→      if (rect.right > window.innerWidth) {
   565→        contextMenu.style.left = `${x - rect.width}px`;
   566→      }

   577→        if (menuItem.classList.contains('context-menu-disabled')) {
   578→          // Disabled items (e.g. Paste when clipboard is empty) do nothing
   579→          return;
   580→        }

   600→  /**
   601→   * Handle context menu action
   602→   */
   603→  function handleContextAction(action, item) {
   604→    switch (action) {
   605→      case 'new-file':
   606→        showCreateDialog('file');
   607→        break;
   608→      case 'new-folder':
   609→        showCreateDialog('folder');
   610→        break;
   611→      case 'new-file-in-folder':
   612→        showCreateDialog('file', item);
   613→        break;
   614→      case 'new-folder-in-folder':
   615→        showCreateDialog('folder', item);
   616→        break;
   617→      case 'copy':
   618→        clipboard = { item: { ...item }, mode: 'copy', projectName: IDEState?.get('currentProject')?.name };
   619→        saveClipboard(clipboard);
   620→        if (IDEUtils) IDEUtils.showToast(`Copied "${item.name}"`, 'info');
   621→        break;
   622→      case 'cut':
   623→        clipboard = { item: { ...item }, mode: 'cut', projectName: IDEState?.get('currentProject')?.name };
   624→        saveClipboard(clipboard);
   625→        if (IDEUtils) IDEUtils.showToast(`Cut "${item.name}"`, 'info');
   626→        break;
   627→      case 'paste':
   628→        executePaste(item, /* intoRoot */ false);
   629→        break;
   630→      case 'paste-into-root':
   631→        executePaste(item, /* intoRoot */ true);
   632→        break;
   633→      case 'rename':
   634→        startRename(item);
   635→        break;
   636→      case 'delete':
   637→        confirmDelete(item);
   638→        break;
   639→    }
   640→  }

   738→  /**
   739→   * Execute the creation
   740→   */
   741→  async function executeCreate(type, parentItem, name) {
   742→    const project = IDEState?.get('currentProject');
   743→    if (!project) return;
   744→    
   745→    // Validate filename
   746→    const validation = IDEUtils?.validateFilename?.(name);
   747→    if (!validation?.valid) {
   748→      if (IDEUtils) IDEUtils.showToast(validation?.error || 'Invalid name', 'error');
   749→      return;
   750→    }
   751→    
   752→    // Build path
   753→    let path = name;
   754→    if (parentItem && parentItem.type === 'folder') {
   755→      path = parentItem.path ? `${parentItem.path}/${name}` : name;
   756→    }
   757→    
   758→    try {
   759→      await IDEStorage.Files.create({
   760→        project: project.name,
   761→        name: name,
   762→        path: path,
   763→        type: type,
   764→        content: type === 'file' ? '' : undefined,
   765→        size: 0
   766→      });
   767→      
   768→      closeDialog();
   769→      
   770→      // Refresh tree
   771→      if (window.IDEState) {
   772→        IDEState.refreshFiles();
   773→      }
   774→      
   775→      // If created a file, open it in editor
   776→      if (type === 'file') {
   777→        setTimeout(async () => {
   778→          try {
   779→            const newFile = await IDEStorage.Files.get(project.name, path);
   780→            if (newFile && window.AppState?.codeEditor) {
   781→              AppState.codeEditor.openFile({ ...newFile, project: project.name });
   782→            }
   783→            if (selectedFile && selectedFile.path === path) return; // Already selected
   784→            // Select the new file
   785→            const fileNode = findInTree(treeData, path);
   786→            if (fileNode) selectFile(fileNode);
   787→          } catch (e) {
   788→            console.error('Failed to open new file:', e);
   789→          }
   790→        }, 100);
   791→      }
   792→      
   793→      // Expand parent if creating inside folder
   794→      if (parentItem && parentItem.type === 'folder') {
   795→        expandedFolders.add(parentItem.path);
   796→      }
   797→      
   798→      if (IDEUtils) IDEUtils.showToast(`Created ${type}: ${name}`, 'success');
   799→      
   800→    } catch (error) {
   801→      console.error(`Failed to create ${type}:`, error);
   802→      if (IDEUtils) IDEUtils.showToast(`Failed to create ${type}`, 'error');
   803→    }
   804→  }

   821→  /**
   822→   * Start inline rename for an item
   823→   */
   824→  function startRename(item) {
   825→    const row = container?.querySelector(`[data-path="${item.path}"] .tree-row`);
   826→    if (!row) return;
   827→    
   828→    const nameEl = row.querySelector('.tree-name');
   829→    if (!nameEl) return;
   830→    
   831→    const originalName = item.name;
   832→    
   833→    // Create inline input
   834→    const input = document.createElement('input');
   835→    input.type = 'text';
   836→    input.className = 'tree-rename-input';
   837→    input.value = originalName;
   838→    input.dataset.originalName = originalName;
   839→    input.dataset.itemPath = item.path;
   840→    
   841→    // Replace name element with input
   842→    nameEl.style.display = 'none';
   843→    nameEl.parentNode.insertBefore(input, nameEl.nextSibling);
   844→    input.focus();
   845→    input.select();
   846→    
   847→    // Handle rename completion
   848→    const finishRename = async () => {
   849→      const newName = input.value.trim();
   850→      
   851→      if (newName && newName !== originalName) {
   852→        await executeRename(item, newName);
   853→      }
   854→      
   855→      // Restore original display
   856→      input.remove();
   857→      nameEl.style.display = '';
   858→      nameEl.textContent = newName || originalName;
   859→    };
   860→    
   861→    input.addEventListener('blur', finishRename);
   862→    input.addEventListener('keydown', (e) => {
   863→      if (e.key === 'Enter') {
   864→        e.preventDefault();
   865→        input.blur();
   866→      } else if (e.key === 'Escape') {
   867→        input.value = originalName;
   868→        input.blur();
   869→      }
   870→    });
   871→  }

   873→  /**
   874→   * Execute rename operation
   875→   */
   876→  async function executeRename(item, newName) {
   877→    const project = IDEState?.get('currentProject');
   878→    if (!project) return;
   879→    
   880→    // Validate new name
   881→    const validation = IDEUtils?.validateFilename?.(newName);
   882→    if (!validation?.valid) {
   883→      if (IDEUtils) IDEUtils.showToast(validation?.error || 'Invalid filename', 'error');
   884→      return;
   885→    }
   886→    
   887→    try {
   888→      // Call API to rename
   889→      if (IDEAPI?.Files?.update) {
   890→        await IDEAPI.Files.update(project.name, item.path, newName);
   891→      }
   892→      
   893→      // Refresh tree
   894→      if (window.IDEState) {
   895→        IDEState.refreshFiles();
   896→      }
   897→      
   898→      if (IDEUtils) IDEUtils.showToast(`Renamed to ${newName}`, 'success');
   899→      
   900→    } catch (error) {
   901→      console.error('Failed to rename:', error);
   902→      if (IDEUtils) IDEUtils.showToast('Failed to rename', 'error');
   903→    }
   904→  }

   931→      // Call API to delete
   932→      if (IDEAPI?.Files?.delete) {

   935→        // Fallback to storage

   943→          IDEState.setActiveFile(null);

   945→      }
   946→      
   947→      // Refresh tree
   948→      if (window.IDEState) {
   949→        IDEState.refreshFiles();
   950→      }
   951→      
   952→      if (IDEUtils) IDEUtils.showToast(`Deleted ${item.type}: ${item.name}`, 'success');
   953→      
   954→    } catch (error) {
   955→      console.error('Failed to delete:', error);
   956→      if (IDEUtils) IDEUtils.showToast('Failed to delete', 'error');

  1428→  function handleDragStart(e, item) {
  1429→    dragItem = item;

  1435→  function handleDragOver(e, targetItem) {
  1436→    // Default: deny drop. Only allow when we have a dragItem and the
  1437→    // target is a valid drop location (a folder, OR a file — in which
  1438→    // case we drop into the file's parent folder, mirroring Finder/Explorer).
  1439→    if (!dragItem) return;
  1440→
  1441→    // Don't allow dropping onto self
  1442→    if (dragItem.path === targetItem.path) return;
  1443→
  1444→    // Don't allow dropping a folder into one of its own descendants
  1445→    if (dragItem.type === 'folder' && targetItem.path.startsWith(dragItem.path + '/')) {
  1446→      return;
  1447→    }
  1448→
  1449→    e.preventDefault();
  1450→    e.dataTransfer.dropEffect = 'move';
  1451→    e.target.closest('.tree-row')?.classList.add('drop-target');
  1452→  }
  1453→
  1454→  function handleDragLeave(e, targetItem) {
  1455→    e.target.closest('.tree-row')?.classList.remove('drop-target');
  1456→  }
  1457→
  1458→  function handleDrop(e, targetItem) {
  1459→    e.preventDefault();
  1460→    e.target.closest('.tree-row')?.classList.remove('drop-target');
  1461→
  1462→    if (!dragItem) return;
  1463→
  1464→    // Don't drop onto self
  1465→    if (dragItem.path === targetItem.path) {
  1466→      dragItem = null;
  1467→      return;
  1468→    }
  1469→
  1470→    // Don't drop a folder into one of its own descendants
  1471→    if (dragItem.type === 'folder' && targetItem.path.startsWith(dragItem.path + '/')) {
  1472→      if (IDEUtils) IDEUtils.showToast('Cannot move a folder into one of its own subfolders.', 'warning');
  1473→      dragItem = null;
  1474→      return;
  1475→    }
  1476→
  1477→    // Determine the destination FOLDER path.
  1478→    // - Drop onto a folder → move INTO that folder.
  1479→    // - Drop onto a file   → move into the file's PARENT folder (sibling move).
  1480→    // - Drop onto a root item with no parent → move to project root.
  1481→    let targetFolderPath;
  1482→    if (targetItem.type === 'folder') {
  1483→      targetFolderPath = targetItem.path;
  1484→    } else {
  1485→      // File: move into its parent directory
  1486→      const lastSlash = targetItem.path.lastIndexOf('/');
  1487→      targetFolderPath = lastSlash > 0 ? targetItem.path.substring(0, lastSlash) : '';
  1488→    }
  1489→
  1490→    // If the source is already in this exact folder, no-op.
  1491→    const sourceParent = (() => {
  1492→      const ls = dragItem.path.lastIndexOf('/');
  1493→      return ls > 0 ? dragItem.path.substring(0, ls) : '';
  1494→    })();
  1495→    if (sourceParent === targetFolderPath) {
  1496→      // Already there — nothing to do
  1497→      dragItem = null;
  1498→      return;
  1499→    }
  1500→
  1501→    // Capture source before clearing dragItem, then execute the move.
  1502→    const source = dragItem;
  1503→    dragItem = null;
  1504→
  1505→    // Also fire the legacy callback (in case any sidebar-level handler wants it)
  1506→    if (onFileAction) {
  1507→      onFileAction({
  1508→        type: 'move',
  1509→        source,
  1510→        target: targetItem,
  1511→        targetFolderPath
  1512→      });
  1513→    }
  1514→
  1515→    // Execute the move (with confirmation if not opted out)
  1516→    confirmAndExecuteMove(source, targetFolderPath, targetItem);
  1517→  }

  1543→  /**
  1544→   * Load the persisted clipboard {item, mode, projectName} or return null.
  1545→   * Stored as JSON so it survives page reloads. We deliberately only persist
  1546→   * the bare minimum fields needed by Files.move / Files.copy (path, name,
  1547→   * type, etc.) — content is fetched from storage at paste time.
  1548→   */
  1549→  function loadClipboard() {
  1550→    try {
  1551→      const raw = localStorage.getItem(CLIPBOARD_PREF_KEY_LOCAL);
  1552→      if (!raw) return null;
  1553→      const parsed = JSON.parse(raw);
  1554→      if (!parsed || !parsed.item || !parsed.mode || !parsed.item.path) return null;
  1555→      return parsed;
  1556→    } catch (e) { return null; }
  1557→  }

  1562→      else localStorage.removeItem(CLIPBOARD_PREF_KEY_LOCAL);
  1563→    } catch (e) { /* non-fatal */ }
  1564→  }

  1575→  async function clipboardStillExists() {

  1579→    try {

  1736→  /**
  1737→   * Resolve the destination folder path for a paste, given the right-clicked
  1738→   * item and whether the user explicitly chose "Paste into root".
  1739→   *
  1740→   *  - intoRoot=true → always '' (project root)
  1741→   *  - item is a folder → item.path (paste INTO that folder)
  1742→   *  - item is a file → item's parent folder (paste as sibling)

  1744→  function resolvePasteTarget(item, intoRoot) {
  1745→    if (intoRoot) return '';
  1746→    if (item.type === 'folder') return item.path;
  1747→    const lastSlash = item.path.lastIndexOf('/');

  1749→  }

  1751→  /**
  1752→   * Execute a paste. Behavior depends on clipboard.mode:
  1753→   *   - 'copy' → Files.copy() (duplicates source at destination; source stays)
  1754→   *   - 'cut'  → Files.move() (relocates source to destination; clears clipboard)
  1755→   *
  1756→   * Cross-project paste is supported: if the clipboard.item.projectName
  1757→   * differs from the current project, we first verify the source exists in
  1758→   * its project, then copy/move it into the current project's target folder.
  1759→   *
  1760→   * @param {Object} item - the right-clicked tree item (target)
  1761→   * @param {boolean} intoRoot - if true, paste into project root
  1762→   */

  1765→      if (IDEUtils) IDEUtils.showToast('Clipboard is empty', 'warning');
  1766→      return;
  1767→    }

  1769→    const currentProject = IDEState?.get('currentProject');

  1771→      if (IDEUtils) IDEUtils.showToast('No project selected', 'warning');
  1772→      return;
  1773→    }
  1774→
  1775→    // Verify clipboard item still exists

  1779→      clearClipboard();
  1780→      return;
  1781→    }
  1782→
  1783→    const sourcePath = clipboard.item.path;
  1784→    const targetFolderPath = resolvePasteTarget(item, intoRoot);
  1785→    const sourceName = clipboard.item.name;

  1788→    if (clipboard.item.type === 'folder') {
  1789→      const target = targetFolderPath;

  1792→        return;
  1793→      }
  1794→    }
  1795→
  1796→    try {
  1797→      if (clipboard.mode === 'cut') {
  1798→        // Move source → targetFolderPath. If source is already there, no-op.
  1799→        const sourceParent = (() => {
  1800→          const ls = sourcePath.lastIndexOf('/');
  1801→          return ls > 0 ? sourcePath.substring(0, ls) : '';
  1802→        })();

  1804→          if (IDEUtils) IDEUtils.showToast(`"${sourceName}" is already there`, 'info');
  1805→          clearClipboard();
  1806→          return;
  1807→        }
  1808→
  1809→        if (window.IDEStorage && IDEStorage.Files && IDEStorage.Files.move) {
  1810→          await IDEStorage.Files.move(currentProject.name, sourcePath, targetFolderPath);
  1811→        } else {

  1813→        }
  1814→

  1816→        const activeFile = IDEState?.get('activeFile');

  1818→          const isSelf = activeFile.path === sourcePath;
  1819→          const isDescendant = activeFile.path.startsWith(sourcePath + '/');
  1820→          if (isSelf || isDescendant) {
  1821→            const newPath = isSelf
  1822→              ? (targetFolderPath ? `${targetFolderPath}/${sourceName}` : sourceName)
  1823→              : (targetFolderPath
  1824→                  ? `${targetFolderPath}/${sourceName}/${activeFile.path.slice(sourcePath.length + 1)}`
  1825→                  : `${sourceName}/${activeFile.path.slice(sourcePath.length + 1)}`);
  1826→            IDEState.setActiveFile({ ...activeFile, path: newPath });
  1827→          }
  1828→        }
  1829→
