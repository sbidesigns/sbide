     1→/**
     2→ * SBIDE - IndexedDB Storage Layer
     3→ * Provides persistent storage for projects, files, and application state
     4→ * Uses IndexedDB for large file storage + localStorage for settings
     5→ */
     6→
     7→const IDEStorage = (() => {
     8→  const DB_NAME = 'ide-agent-platform';
     9→  const DB_VERSION = 1;
    10→  
    11→  let db = null;
    12→  
    13→  // Store names
    14→  const STORES = {
    15→    PROJECTS: 'projects',
    16→    FILES: 'files',
    17→    VERSIONS: 'versions',
    18→    MEMORY: 'memory',
    19→    MEETINGS: 'meetings',
    20→    CONTEXT: 'context'
    21→  };
    22→
    23→  /**
    24→   * Initialize the database connection
    25→   * @returns {Promise<IDBDatabase>}
    26→   */
    27→  /* === In-memory fallback (used when IndexedDB is unavailable) ===
    28→     Triggered by browser extensions, sandboxed iframes, quota issues, or
    29→     profile corruption. The fallback provides the same public API surface
    30→     (Projects/Files/Versions/Memory/Meetings/Context/exportAll/restoreAll/
    31→     clearAll) so the rest of the app keeps working. Data does NOT persist
    32→     across reloads in fallback mode. */
    33→  let fallbackMode = false;
    34→  let fallbackStores = null;
    35→
    36→  function initFallbackStores() {
    37→    const stores = {};
    38→    Object.values(STORES).forEach(name => { stores[name] = new Map(); });
    39→    return stores;
    40→  }
    41→
    42→  function fallbackGetAll(storeName) {
    43→    return Promise.resolve(Array.from(fallbackStores[storeName].values()));
    44→  }
    45→  function fallbackGet(storeName, key) {
    46→    return Promise.resolve(fallbackStores[storeName].get(key) || null);
    47→  }
    48→  function fallbackPut(storeName, data) {
    49→    // Determine the key from the store's keyPath
    50→    const keyPath = FALLBACK_KEY_PATHS[storeName];
    51→    const key = keyPath ? (Array.isArray(keyPath) ? keyPath.map(k => data[k]).join('\u0000') : data[keyPath]) : data.id;
    52→    fallbackStores[storeName].set(key, data);
    53→    return Promise.resolve(data);
    54→  }
    55→  function fallbackRemove(storeName, key) {
    56→    fallbackStores[storeName].delete(key);
    57→    return Promise.resolve();
    58→  }
    59→  function fallbackClear(storeName) {
    60→    fallbackStores[storeName].clear();
    61→    return Promise.resolve();
    62→  }
    63→  function fallbackGetByIndex(storeName, indexName, value) {
    64→    // Linear scan; fine for in-memory fallback
    65→    const keyPath = FALLBACK_INDEX_KEY_PATHS[storeName]?.[indexName];
    66→    if (!keyPath) return Promise.resolve([]);
    67→    const result = [];
    68→    for (const item of fallbackStores[storeName].values()) {
    69→      const itemVal = Array.isArray(keyPath) ? keyPath.map(k => item[k]).join('\u0000') : item[keyPath];
    70→      if (itemVal === value) result.push(item);
    71→    }
    72→    return Promise.resolve(result);
    73→  }
    74→
    75→  const FALLBACK_KEY_PATHS = {
    76→    [STORES.PROJECTS]: 'name',
    77→    [STORES.FILES]: 'id',
    78→    [STORES.VERSIONS]: 'id',
    79→    [STORES.MEMORY]: 'projectName',
    80→    [STORES.MEETINGS]: 'id',
    81→    [STORES.CONTEXT]: 'projectName'
    82→  };
    83→  const FALLBACK_INDEX_KEY_PATHS = {
    84→    [STORES.PROJECTS]: { updatedAt: 'updatedAt' },
    85→    [STORES.FILES]: { project: 'project', path: 'path', ['project,path']: ['project', 'path'] },
    86→    [STORES.VERSIONS]: { projectName: 'projectName' },
    87→    [STORES.MEETINGS]: { projectName: 'projectName' }
    88→  };
    89→
    90→  async function init() {
    91→    // If already in fallback mode, just resolve.
    92→    if (fallbackMode) return null;
    93→    if (db) return db;
    94→
    95→    // Detect if IndexedDB is even available. Some sandboxed contexts throw
    96→    // on access; others return undefined.
    97→    let indexedDBAvailable = false;
    98→    try {
    99→      indexedDBAvailable = typeof indexedDB !== 'undefined' && indexedDB !== null;
   100→    } catch (e) {
   101→      indexedDBAvailable = false;
   102→    }
   103→
   104→    if (!indexedDBAvailable) {
   105→      console.warn('IndexedDB unavailable — switching to in-memory fallback. Data will not persist.');
   106→      fallbackMode = true;
   107→      fallbackStores = initFallbackStores();
   108→      return null;
   109→    }
   110→
   111→    return new Promise((resolve, reject) => {
   112→      let request;
   113→      try {
   114→        request = indexedDB.open(DB_NAME, DB_VERSION);
   115→      } catch (e) {
   116→        console.warn('indexedDB.open threw — switching to in-memory fallback:', e);
   117→        fallbackMode = true;
   118→        fallbackStores = initFallbackStores();
   119→        resolve(null);
   120→        return;
   121→      }
   122→
   123→      request.onerror = () => {
   124→        console.warn('Failed to open IndexedDB — switching to in-memory fallback. Data will not persist.', request.error);
   125→        fallbackMode = true;
   126→        fallbackStores = initFallbackStores();
   127→        resolve(null);  // resolve (not reject) so app can continue
   128→      };
   129→
   130→      request.onsuccess = () => {
   131→        db = request.result;
   132→
   133→        db.onclose = () => { db = null; };
   134→        db.onversionchange = () => { db.close(); db = null; };
   135→
   136→        resolve(db);
   137→      };
   138→
   139→      request.onupgradeneeded = (event) => {
   140→        const database = event.target.result;
   141→
   142→        if (!database.objectStoreNames.contains(STORES.PROJECTS)) {
   143→          const projectStore = database.createObjectStore(STORES.PROJECTS, { keyPath: 'name' });
   144→          projectStore.createIndex('updatedAt', 'updatedAt', { unique: false });
   145→        }
   146→
   147→        if (!database.objectStoreNames.contains(STORES.FILES)) {
   148→          const fileStore = database.createObjectStore(STORES.FILES, { keyPath: 'id' });
   149→          fileStore.createIndex('project', 'project', { unique: false });
   150→          fileStore.createIndex('path', 'path', { unique: false });
   151→          fileStore.createIndex(['project', 'path'], ['project', 'path'], { unique: true });
   152→        }
   153→
   154→        if (!database.objectStoreNames.contains(STORES.VERSIONS)) {
   155→          const versionStore = database.createObjectStore(STORES.VERSIONS, { keyPath: 'id' });
   156→          versionStore.createIndex('projectName', 'projectName', { unique: false });
   157→        }
   158→
   159→        if (!database.objectStoreNames.contains(STORES.MEMORY)) {
   160→          database.createObjectStore(STORES.MEMORY, { keyPath: 'projectName' });
   161→        }
   162→
   163→        if (!database.objectStoreNames.contains(STORES.MEETINGS)) {
   164→          const meetingStore = database.createObjectStore(STORES.MEETINGS, { keyPath: 'id' });
   165→          meetingStore.createIndex('projectName', 'projectName', { unique: false });
   166→        }
   167→
   168→        if (!database.objectStoreNames.contains(STORES.CONTEXT)) {
   169→          database.createObjectStore(STORES.CONTEXT, { keyPath: 'projectName' });
   170→        }
   171→      };
   172→    });
   173→  }
   174→
   175→  /**
   176→   * Generic transaction helper
   177→   */
   178→  function withTransaction(storeName, mode, callback) {
   179→    return new Promise(async (resolve, reject) => {
   180→      try {
   181→        if (fallbackMode) {
   182→          // In-memory fallback: synthesize a minimal store-like object
   183→          // with get/put/delete/getAll/clear/index methods that operate
   184→          // on the fallback Map. This lets the existing callback code work
   185→          // unchanged.
   186→          const fakeStore = {
   187→            get: (key) => {
   188→              const req = { onsuccess: null, onerror: null, result: fallbackStores[storeName].get(key) || null };
   189→              // Fire callbacks asynchronously to mimic IDB semantics
   190→              Promise.resolve().then(() => req.onsuccess && req.onsuccess());
   191→              return req;
   192→            },
   193→            getAll: () => {
   194→              const req = { onsuccess: null, onerror: null, result: Array.from(fallbackStores[storeName].values()) };
   195→              Promise.resolve().then(() => req.onsuccess && req.onsuccess());
   196→              return req;
   197→            },
   198→            put: (data) => {
   199→              const keyPath = FALLBACK_KEY_PATHS[storeName];
   200→              const key = keyPath ? (Array.isArray(keyPath) ? keyPath.map(k => data[k]).join('\u0000') : data[keyPath]) : data.id;
   201→              fallbackStores[storeName].set(key, data);
   202→              const req = { onsuccess: null, onerror: null, result: key };
   203→              Promise.resolve().then(() => req.onsuccess && req.onsuccess());
   204→              return req;
   205→            },
   206→            delete: (key) => {
   207→              fallbackStores[storeName].delete(key);
   208→              const req = { onsuccess: null, onerror: null };
   209→              Promise.resolve().then(() => req.onsuccess && req.onsuccess());
   210→              return req;
   211→            },
   212→            clear: () => {
   213→              fallbackStores[storeName].clear();
   214→              const req = { onsuccess: null, onerror: null };
   215→              Promise.resolve().then(() => req.onsuccess && req.onsuccess());
   216→              return req;
   217→            },
   218→            index: (indexName) => {
   219→              const keyPath = FALLBACK_INDEX_KEY_PATHS[storeName]?.[indexName];
   220→              return {
   221→                getAll: (value) => {
   222→                  const result = [];
   223→                  if (keyPath) {
   224→                    for (const item of fallbackStores[storeName].values()) {
   225→                      const itemVal = Array.isArray(keyPath) ? keyPath.map(k => item[k]).join('\u0000') : item[keyPath];
   226→                      if (itemVal === value) result.push(item);
   227→                    }
   228→                  }
   229→                  const req = { onsuccess: null, onerror: null, result };
   230→                  Promise.resolve().then(() => req.onsuccess && req.onsuccess());
   231→                  return req;
   232→                }
   233→              };
   234→            }
   235→          };
   236→          const result = await callback(fakeStore);
   237→          // In fallback mode there's no transaction lifecycle — resolve immediately.
   238→          resolve(result);
   239→          return;
   240→        }
   241→
   242→        const database = await init();
   243→        if (!database) {
   244→          // Should not happen (fallbackMode should be true), but guard anyway.
   245→          resolve(null);
   246→          return;
   247→        }
   248→        const transaction = database.transaction(storeName, mode);
   249→        const store = transaction.objectStore(storeName);
   250→
   251→        const result = await callback(store);
   252→
   253→        transaction.oncomplete = () => resolve(result);
   254→        transaction.onerror = () => reject(transaction.error);
   255→      } catch (error) {
   256→        reject(error);
   257→      }
   258→    });
   259→  }

   261→  /**
   262→   * Generate a unique ID
   263→   */
   264→  function generateId() {
   265→    return Date.now().toString(36) + Math.random().toString(36).substr(2, 9);
   266→  }
   267→
   268→  /**
   269→   * Get all items from a store
   270→   */
   271→  async function getAll(storeName) {
   272→    return withTransaction(storeName, 'readonly', (store) => {
   273→      return new Promise((resolve, reject) => {
   274→        const request = store.getAll();
   275→        request.onsuccess = () => resolve(request.result);
   276→        request.onerror = () => reject(request.error);
   277→      });
   278→    });
   279→  }
   280→
   281→  /**
   282→   * Get a single item by key
   283→   */
   284→  async function get(storeName, key) {
   285→    return withTransaction(storeName, 'readonly', (store) => {
   286→      return new Promise((resolve, reject) => {
   287→        const request = store.get(key);
   288→        request.onsuccess = () => resolve(request.result);
   289→        request.onerror = () => reject(request.error);
   290→      });
   291→    });
   292→  }
   293→
   294→  /**
   295→   * Put or update an item
   296→   */
   297→  async function put(storeName, data) {
   298→    return withTransaction(storeName, 'readwrite', (store) => {
   299→      return new Promise((resolve, reject) => {
   300→        const request = store.put(data);
   301→        request.onsuccess = () => resolve(request.result);
   302→        request.onerror = () => reject(request.error);
   303→      });
   304→    });
   305→  }
   306→
   307→  /**
   308→   * Delete an item by key
   309→   */
   310→  async function remove(storeName, key) {
   311→    return withTransaction(storeName, 'readwrite', (store) => {
   312→      return new Promise((resolve, reject) => {
   313→        const request = store.delete(key);
   314→        request.onsuccess = () => resolve();
   315→        request.onerror = () => reject(request.error);
   316→      });
   317→    });
   318→  }
   319→
   320→  /**
   321→   * Clear all items in a store
   322→   */
   323→  async function clear(storeName) {
   324→    return withTransaction(storeName, 'readwrite', (store) => {
   325→      return new Promise((resolve, reject) => {
   326→        const request = store.clear();
   327→        request.onsuccess = () => resolve();
   328→        request.onerror = () => reject(request.error);
   329→      });
   330→    });
   331→  }
   332→
   333→  /**
   334→   * Get items by index
   335→   */
   336→  async function getByIndex(storeName, indexName, value) {
   337→    return withTransaction(storeName, 'readonly', (store) => {
   338→      return new Promise((resolve, reject) => {
   339→        const index = store.index(indexName);
   340→        const request = index.getAll(value);
   341→        request.onsuccess = () => resolve(request.result);
   342→        request.onerror = () => reject(request.error);
   343→      });
   344→    });
   345→  }

   510→    /**
   511→     * Delete a file or folder (and children)
   512→     */
   513→    async delete(projectName, path) {
   514→      const file = await this.get(projectName, path);
   515→      if (!file) return;
   516→
   517→      // If it's a folder, delete all children
   518→      if (file.type === 'folder' && file.children) {
   519→        for (const childPath of file.children) {
   520→          await this.delete(projectName, childPath);
   521→        }
   522→      }
   523→
   524→      await remove(STORES.FILES, file.id);
   525→
   526→      // Mirror delete to OfflineKit (OPFS + filesystem + sync queue)
   527→      if (window.OfflineKit) {
   528→        try { await OfflineKit.remove(projectName, path); }
   529→        catch (e) { /* non-fatal */ }
   530→      }
   531→    },
   532→
   533→    /**
   534→     * Move a file or folder (and all descendants) into a target folder.
   535→     *
   536→     * @param {string} projectName
   537→     * @param {string} sourcePath  - path of the file/folder being moved
   538→     * @param {string} targetFolderPath - path of the destination FOLDER
   539→     *        (use '' or null to move to project root)
   540→     * @returns {Object} the moved source (with updated path)
   541→     * @throws if source equals target, source is an ancestor of target,
   542→     *         or a same-named item already exists at the destination.
   543→     */
   544→    async move(projectName, sourcePath, targetFolderPath) {
   545→      const srcSlash = sourcePath.endsWith('/') ? sourcePath.slice(0, -1) : sourcePath;
   546→      const tgtSlash = (targetFolderPath || '').endsWith('/')
   547→        ? targetFolderPath.slice(0, -1) : (targetFolderPath || '');
   548→
   549→      // Validate: can't move into itself
   550→      if (srcSlash === tgtSlash) {
   551→        throw new Error('Cannot move a file/folder into itself.');
   552→      }
   553→      // Validate: can't move a folder into one of its own descendants
   554→      if (tgtSlash.startsWith(srcSlash + '/')) {
   555→        throw new Error('Cannot move a folder into one of its own subfolders.');
   556→      }
   557→
   558→      const source = await this.get(projectName, srcSlash);
   559→      if (!source) throw new Error(`Source "${srcSlash}" not found.`);
   560→
   561→      // Compute new path
   562→      const newName = source.name;
   563→      const newPath = tgtSlash ? `${tgtSlash}/${newName}` : newName;
   564→
   565→      // Reject if an item with the same path already exists at the destination
   566→      const existing = await this.get(projectName, newPath);
   567→      if (existing) {
   568→        throw new Error(`An item named "${newName}" already exists in the destination folder.`);
   569→      }
   570→
   571→      // Collect ALL files for this project so we can find descendants of source
   572→      // (descendants are any items whose path starts with `srcSlash + '/'`).
   573→      const allFiles = await this.getByProject(projectName);
   574→      const descendants = allFiles.filter(f =>
   575→        f.path === srcSlash || f.path.startsWith(srcSlash + '/')
   576→      );
   577→
   578→      // Rewrite each descendant's path by replacing the srcSlash prefix
   579→      // with newPath. Use a transaction-like loop — best-effort atomic.
   580→      const updates = [];
   581→      for (const f of descendants) {
   582→        const suffix = f.path === srcSlash ? '' : f.path.slice(srcSlash.length); // includes leading '/'
   583→        const movedPath = newPath + suffix;
   584→        updates.push({ file: f, newPath: movedPath });
   585→      }
   586→
   587→      // Apply: update path on each, then re-put
   588→      for (const { file, newPath: movedPath } of updates) {
   589→        const updated = {
   590→          ...file,
   591→          path: movedPath,
   592→          // name stays the same (only the directory chain changes)
   593→          updatedAt: Date.now()
   594→        };
   595→        await put(STORES.FILES, updated);
   596→      }
   597→
   598→      // Mirror to OfflineKit (best-effort): delete at old path, write at new path.
   599→      // We can't easily enumerate OfflineKit's per-file content here, so we just
   600→      // delete the old source path; the next content write will create the new path.
   601→      if (window.OfflineKit) {
   602→        try { await OfflineKit.remove(projectName, srcSlash); }
   603→        catch (e) { /* non-fatal */ }
   604→      }
   605→
   606→      // Return the moved source (with its new path)
   607→      return { ...source, path: newPath };
   608→    },
   609→
   610→    /**
   611→     * Copy a file or folder (and all descendants) into a target folder.
   612→     * If an item with the same name exists at the destination, " (copy)"
   613→     * (then " (copy 2)", " (copy 3)"…) is appended to the name.
   614→     *
   615→     * @param {string} projectName
   616→     * @param {string} sourcePath  - path of the file/folder being copied
   617→     * @param {string} targetFolderPath - destination FOLDER ('' = root)
   618→     * @returns {Object} the copied root item (with its new path)
   619→     */
   620→    async copy(projectName, sourcePath, targetFolderPath) {
   621→      const srcSlash = sourcePath.endsWith('/') ? sourcePath.slice(0, -1) : sourcePath;
   622→      const tgtSlash = (targetFolderPath || '').endsWith('/')
   623→        ? targetFolderPath.slice(0, -1) : (targetFolderPath || '');
   624→
   625→      const source = await this.get(projectName, srcSlash);
   626→      if (!source) throw new Error(`Source "${srcSlash}" not found.`);
   627→
   628→      // Find a non-colliding name at the destination
   629→      let baseName = source.name;
   630→      let candidateName = baseName;
   631→      let counter = 0;
   632→      // eslint-disable-next-line no-constant-condition
   633→      while (true) {
   634→        const candidatePath = tgtSlash ? `${tgtSlash}/${candidateName}` : candidateName;
   635→        const clash = await this.get(projectName, candidatePath);
   636→        if (!clash) break;
   637→        counter += 1;
   638→        candidateName = counter === 1
   639→          ? `${baseName} (copy)`
   640→          : `${baseName} (copy ${counter})`;
   641→      }
   642→
   643→      const newPath = tgtSlash ? `${tgtSlash}/${candidateName}` : candidateName;
   644→
   645→      // Collect descendants of source (including source itself)
   646→      const allFiles = await this.getByProject(projectName);
   647→      const descendants = allFiles.filter(f =>
   648→        f.path === srcSlash || f.path.startsWith(srcSlash + '/')
   649→      );
   650→
   651→      // Create copies with rewritten paths
   652→      for (const f of descendants) {
   653→        const suffix = f.path === srcSlash ? '' : f.path.slice(srcSlash.length);
   654→        const copiedPath = newPath + suffix;
   655→        const copy = {
   656→          ...f,
   657→          id: generateId(),
   658→          path: copiedPath,
   659→          // Keep original name for descendants (only root copy gets the "(copy)" suffix)
   660→          name: f.path === srcSlash ? candidateName : f.name,
   661→          updatedAt: Date.now()
   662→        };
   663→        await put(STORES.FILES, copy);
   664→        // Mirror to OfflineKit if the file has content
   665→        if (window.OfflineKit && typeof f.content === 'string') {
   666→          try { await OfflineKit.write(projectName, copiedPath, f.content); }
   667→          catch (e) { /* non-fatal */ }
   668→        }
   669→      }
   670→
   671→      return { ...source, id: undefined, name: candidateName, path: newPath };
   672→    },

   674→    /**
   675→     * Write/update file content

   677→    async writeContent(projectName, path, content) {
   678→      const result = await this.update(projectName, path, {
   679→        content,
   680→        size: new Blob([content]).size
   681→      });
   682→      // Mirror to OfflineKit (OPFS + filesystem + sync queue) if available.
   683→      // Best-effort: don't let it block or fail the primary write.
   684→      if (window.OfflineKit) {
   685→        try { await OfflineKit.write(projectName, path, content); }
   686→        catch (e) { /* non-fatal — primary write already succeeded */ }
   687→      }
   688→      return result;
   689→    },

   697→      // Fallback: if primary store is missing the file (e.g. fresh session
   698→      // after IndexedDB was cleared but OPFS still has it), try OfflineKit.
   699→      if (window.OfflineKit) {
   700→        try {
   701→          const text = await OfflineKit.read(projectName, path);
   702→          if (text !== null) return text;
   703→        } catch (e) { /* non-fatal */ }
   704→      }
   705→      return null;

   721→      
   722→      for (const [filePath, content] of Object.entries(files)) {
   723→        const fullPath = basePath ? `${basePath}/${filePath}` : filePath;
   724→        const parts = fullPath.split('/');
   725→        const fileName = parts.pop();
   726→        const dirPath = parts.join('/');
   727→        
   728→        // Ensure parent directories exist
   729→        if (parts.length > 0) {
   730→          await this.ensureDirectory(projectName, parts);
   731→        }
   732→        
   733→        // Create the file
   734→        const file = await this.create({
   735→          project: projectName,
   736→          name: fileName,
   737→          path: fullPath,
   738→          type: 'file',
   739→          content: typeof content === 'string' ? content : JSON.stringify(content),
   740→          size: typeof content === 'string' ? content.length : JSON.stringify(content).length
   741→        });
   742→        
   743→        created.push(file);
   744→      }
   745→      
   746→      return created;

  1003→  /**
  1004→   * Build file tree from flat array
  1005→   */
  1006→  function buildFileTree(files) {
  1007→    const root = [];
  1008→    const map = new Map();
  1009→    
  1010→    // First pass: create map of all items
  1011→    for (const file of files) {
  1012→      map.set(file.path, { ...file, children: [] });
  1013→    }
  1014→    
  1015→    // Second pass: build hierarchy
  1016→    for (const file of files) {
  1017→      const node = map.get(file.path);
  1018→      if (!file.path || file.path === '') {
  1019→        root.push(node);
  1020→      } else {
  1021→        const parentPath = file.path.substring(0, file.path.lastIndexOf('/'));
  1022→        const parent = map.get(parentPath);
  1023→        if (parent) {
  1024→          parent.children.push(node);
  1025→        } else {
  1026→          root.push(node);
  1027→        }
  1028→      }
  1029→    }
  1030→    
  1031→    // Sort: folders first, then alphabetically
  1032→    const sortNodes = (nodes) => {
  1033→      nodes.sort((a, b) => {
  1034→        if (a.type !== b.type) {
  1035→          return a.type === 'folder' ? -1 : 1;
  1036→        }
  1037→        return a.name.localeCompare(b.name);
  1038→      });
  1039→      for (const node of nodes) {
  1040→        if (node.children) {
  1041→          sortNodes(node.children);
  1042→        }
  1043→      }
  1044→    };
  1045→    
  1046→    sortNodes(root);
  1047→    
  1048→    // Return only children of root (if root is empty string)
  1049→    return root.length === 1 && root[0].path === '' ? root[0].children : root;
  1050→  }
