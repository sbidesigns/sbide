   367→  /**
   368→   * Toggle sidebar visibility
   369→   * HTML structure: #sidebar contains #sidebar-collapsed and #sidebar-expanded
   370→   */
   371→  function toggleSidebar(forceOpen) {
   372→    isOpen = typeof forceOpen === 'boolean' ? forceOpen : !isOpen;
   373→    
   374→    // Update main sidebar element
   375→    if (sidebarElement) {
   376→      sidebarElement.classList.toggle('open', isOpen);
   377→      sidebarElement.classList.toggle('closed', !isOpen);
   378→    }
   379→    
   380→    // Show/hide collapsed vs expanded sections
   381→    const collapsedDiv = document.getElementById('sidebar-collapsed');
   382→    const expandedDiv = document.getElementById('sidebar-expanded');
   383→    
   384→    if (collapsedDiv) {
   385→      // Collapsed div shows when sidebar is closed OR when in rail mode
   386→      const showCollapsed = !isOpen || isCollapsed;
   387→      collapsedDiv.classList.toggle('hidden', isOpen && !isCollapsed);
   388→    }
   389→    
   390→    if (expandedDiv) {
   391→      // Expanded div shows when sidebar is open AND not in rail mode
   392→      expandedDiv.classList.toggle('hidden', !isOpen || isCollapsed);
   393→    }
   394→    
   395→    // Update mobile menu button
   396→    const mobileBtn = document.getElementById('mobile-menu-btn');
   397→    if (mobileBtn) {
   398→      mobileBtn.setAttribute('aria-expanded', String(isOpen));
   399→      // Swap icons
   400→      const menuIcon = mobileBtn.querySelector('.icon-menu');
   401→      const xIcon = mobileBtn.querySelector('.icon-x');
   402→      if (menuIcon && xIcon) {
   403→        menuIcon.classList.toggle('hidden', isOpen);
   404→        xIcon.classList.toggle('hidden', !isOpen);
   405→      }
   406→    }
   407→    
   408→    // Sync overlay
   409→    syncOverlay();
   410→    
   411→    // Update state
   412→    if (window.IDEState) {
   413→      IDEState.setSidebarOpen(isOpen);
   414→    }
   415→  }

   417→  /**
   418→   * Sync mobile overlay visibility with sidebar open state.
   419→   * (Restored from v1: only show overlay on mobile when sidebar is open.)
   420→   */
   421→  function syncOverlay() {
   422→    const mobileOverlay = document.getElementById('mobile-overlay');
   423→    if (mobileOverlay) {
   424→      // Only show overlay on mobile when sidebar is open
   425→      const isMobile = window.innerWidth <= 767;
   426→      mobileOverlay.classList.toggle('hidden', !isOpen || !isMobile);
   427→    }

   430→  /**
   431→   * Set collapsed (rail-only) mode.
   432→   *
   433→   * Animation timing: the `.sidebar` element has `transition: width var(--transition-slow)`
   434→   * (300ms). To make sidebar→rail animate the same way rail→sidebar does, we keep
   435→   * #sidebar-expanded visible (clipped by .sidebar's overflow:hidden) while the width
   436→   * shrinks, then swap to the rail after the transition completes. Without this delay,
   437→   * the CSS rule `.sidebar.collapsed #sidebar-expanded { display: none }` would fire
   438→   * instantly and the user would see the content vanish before the empty space collapses.
   439→   */
   440→  let collapseAnimTimeout = null;
   441→  const COLLAPSE_ANIM_MS = 300; // matches --transition-slow
   442→
   443→  function setCollapsed(collapsed) {
   444→    isCollapsed = collapsed;
   445→
   446→    // Cancel any pending content-swap from a previous toggle
   447→    if (collapseAnimTimeout) {
   448→      clearTimeout(collapseAnimTimeout);
   449→      collapseAnimTimeout = null;
   450→    }
   451→
   452→    if (sidebarElement) {
   453→      sidebarElement.classList.toggle('collapsed', collapsed);
   454→    }
   455→
   456→    // Restored from v1: When un-collapsing, ensure sidebar is open (not closed)
   457→    if (!collapsed && !isOpen) {
   458→      isOpen = true;
   459→      if (sidebarElement) {
   460→        sidebarElement.classList.add('open');
   461→        sidebarElement.classList.remove('closed');
   462→      }
   463→    }
   464→
   465→    const collapsedDiv = document.getElementById('sidebar-collapsed');
   466→    const expandedDiv = document.getElementById('sidebar-expanded');
   467→
   468→    if (collapsed) {
   469→      // Collapsing sidebar → rail:
   470→      //  - Keep #sidebar-expanded visible during the width animation. It will be
   471→      //    clipped naturally by .sidebar's `overflow: hidden` as the width shrinks.
   472→      //  - Keep #sidebar-collapsed hidden via .hidden (!important) so it doesn't
   473→      //    steal layout space from #sidebar-expanded mid-animation (which would
   474→      //    cause expanded content to reflow/jump).
   475→      //  - After 300ms (width transition done), swap: hide expanded, show rail.
   476→      if (expandedDiv) expandedDiv.classList.remove('hidden');
   477→      if (collapsedDiv) collapsedDiv.classList.add('hidden');
   478→
   479→      collapseAnimTimeout = setTimeout(() => {
   480→        if (expandedDiv) expandedDiv.classList.add('hidden');
   481→        if (collapsedDiv) collapsedDiv.classList.remove('hidden');
   482→        collapseAnimTimeout = null;
   483→      }, COLLAPSE_ANIM_MS);
   484→    } else {
   485→      // Expanding rail → sidebar:
   486→      //  - Show #sidebar-expanded immediately so it reveals progressively as the
   487→      //    width grows (this is the direction the user already sees as "animated").
   488→      //  - #sidebar-collapsed is auto-hidden by the CSS rule
   489→      //    `.sidebar:not(.collapsed):not(.closed) #sidebar-collapsed { display: none }`
   490→      //    the instant .collapsed is removed, which is fine for expand.
   491→      if (expandedDiv) expandedDiv.classList.remove('hidden');
   492→    }
   493→
   494→    // Keep overlay in sync (Restored from v1)
   495→    syncOverlay();
   496→
   497→    // Persist state (Restored from v1)
   498→    if (window.IDEState) {
   499→      IDEState.setSidebarOpen(isOpen);
   500→    }
   501→  }

   506→  function toggleCollapse(forceCollapsed) {
   507→    setCollapsed(typeof forceCollapsed === 'boolean' ? forceCollapsed : !isCollapsed);
   508→  }

   528→  /**
   529→   * Switch between sidebar tabs
   530→   * HTML uses .tab-btn for tab buttons and #panel-* for panels
   531→   */
   532→  function switchTab(tabName, updateState = true) {
   533→    if (!tabName) return;
   534→    
   535→    activeTab = tabName;
   536→    
   537→    // Update tab UI - both .tab-btn (expanded) and .sidebar-nav-btn (collapsed)
   538→    const tabs = container.querySelectorAll('.tab-btn');
   539→    tabs.forEach(tab => {
   540→      const isActive = tab.dataset.tab === tabName;
   541→      tab.classList.toggle('active', isActive);
   542→      tab.setAttribute('aria-selected', String(isActive));
   543→    });
   544→    
   545→    // Also update nav buttons in collapsed sidebar
   546→    const navBtns = container.querySelectorAll('.sidebar-nav-btn');
   547→    navBtns.forEach(btn => {
   548→      btn.classList.toggle('active', btn.dataset.tab === tabName);
   549→    });
   550→    
   551→    // Show/hide panels using correct IDs: panel-files, panel-search, etc.
   552→    const panelMap = {
   553→      'files': 'panel-files',
   554→      'search': 'panel-search',
   555→      'memory': 'panel-memory',
   556→      'versions': 'panel-versions',
   557→      'meetings': 'panel-meetings'
   558→    };
   559→    
   560→    Object.entries(panelMap).forEach(([key, panelId]) => {
   561→      const el = document.getElementById(panelId);
   562→      if (el) {
   563→        el.classList.toggle('active', key === tabName);
   564→      }
   565→    });
   566→    
   567→    // Update state
   568→    if (updateState && window.IDEState) {
   569→      IDEState.setActiveTab(tabName);
   570→    }
   571→  }

   577→  /**
   578→   * Create a new project
   579→   */
   580→  async function createProject(displayName) {
   581→    const validation = IDEUtils?.validateProjectName(displayName);
   582→    if (!validation?.valid) {
   583→      if (IDEUtils) IDEUtils.showToast(validation?.error || 'Invalid project name', 'error');
   584→      return;
   585→    }
   586→    
   587→    // Use slugified name internally, keep display name for UI
   588→    const slug = validation.slug || IDEUtils?.slugify(displayName) || displayName;
   589→    
   590→    try {
   591→      // Create in storage with slug as the key
   592→      const project = await IDEStorage.Projects.create(slug);
   593→      
   594→      // Store display name separately if different from slug
   595→      if (slug !== displayName.trim()) {
   596→        project.displayName = displayName.trim();
   597→        await IDEStorage.Projects.update(slug, { displayName: displayName.trim() });
   598→      }
   599→      
   600→      // Set as current project
   601→      if (window.IDEState) {
   602→        await IDEState.setCurrentProject(project);
   603→      }
   604→      
   605→      // Load project data
   606→      await loadProjectData(project);
   607→      
   608→      // Hide modal
   609→      hideModal(document.getElementById('new-project-modal'));
   610→      
   611→      // Clear input
   612→      const input = document.getElementById('new-project-name');
   613→      if (input) input.value = '';
   614→      
   615→      // Show success message with both names if different
   616→      const showName = slug !== displayName.trim() 
   617→        ? `${displayName.trim()} (${slug})` 
   618→        : slug;
   619→      if (IDEUtils) IDEUtils.showToast(`Project "${showName}" created!`, 'success');
   620→      
   621→      // Switch to files tab
   622→      switchTab('files');
   623→      
   624→    } catch (error) {
   625→      console.error('Failed to create project:', error);
   626→      if (IDEUtils) IDEUtils.showToast(`Failed to create project: ${error.message}`, 'error');
   627→    }
   628→  }

   631→   * Import project from ZIP
   632→   */
   633→  async function importProject(fileInput, nameOverride) {
   634→    if (!fileInput || !fileInput.files.length) return;
   635→    
   636→    const file = fileInput.files[0];
   637→    
   638→    // Validate file type (only allow ZIP files)
   639→    if (!file.name.toLowerCase().endsWith('.zip') && file.type !== 'application/zip' && file.type !== 'application/x-zip-compressed') {
   640→      if (IDEUtils) IDEUtils.showToast('Please select a valid .zip file', 'error');
   641→      return;
   642→    }
   643→    
   644→    // Validate file size (max 50MB to prevent browser crash/memory issues)
   645→    const MAX_FILE_SIZE = 50 * 1024 * 1024; // 50MB
   646→    if (file.size > MAX_FILE_SIZE) {
   647→      const sizeMB = (file.size / (1024 * 1024)).toFixed(1);
   648→      if (IDEUtils) IDEUtils.showToast(`File too large (${sizeMB}MB). Maximum size is 50MB.`, 'error');
   649→      return;
   650→    }
   651→    
   652→    // Validate minimum file size (not empty)
   653→    if (file.size === 0) {
   654→      if (IDEUtils) IDEUtils.showToast('Selected file is empty', 'error');
   655→      return;
   656→    }
   657→    
   658→    const name = nameOverride || file.name.replace(/\.zip$/i, '');
   659→    
   660→    try {
   661→      // Validate name first
   662→      const validation = IDEUtils?.validateProjectName(name);
   663→      if (!validation?.valid) {
   664→        if (IDEUtils) IDEUtils.showToast(validation?.error || 'Invalid project name', 'error');
   665→        return;
   666→      }
   667→      
   668→      // Create project
   669→      const project = await IDEStorage.Projects.create(name);
   670→      
   671→      // Read ZIP file
   672→      if (window.JSZip) {
   673→        let zip;
   674→        try {
   675→          zip = await JSZip.loadAsync(file);
   676→        } catch (zipError) {
   677→          console.error('Invalid ZIP file:', zipError);
   678→          if (IDEUtils) IDEUtils.showToast('Invalid or corrupted ZIP file', 'error');
   679→          return;
   680→        }
   681→        
   682→        const files = {};
   683→        const MAX_FILES = 1000; // Prevent ZIP bomb
   684→        let totalSize = 0;
   685→        
   686→        // Extract all files with safety limits
   687→        for (const [relativePath, zipEntry] of Object.entries(zip.files)) {
   688→          // Safety: limit number of files
   689→          if (Object.keys(files).length >= MAX_FILES) {
   690→            console.warn(`ZIP contains more than ${MAX_FILES} files, truncating`);
   691→            if (IDEUtils) IDEUtils.showToast(`Too many files in ZIP (limit: ${MAX_FILES}). Some files were skipped.`, 'warning');
   692→            break;
   693→          }
   694→          
   695→          if (!zipEntry.dir) {
   696→            // Safety: check individual file size
   697→            if (zipEntry._data?.uncompressedSize > 10 * 1024 * 1024) { // 10MB per file limit
   698→              console.warn(`Skipping large file: ${relativePath}`);
   699→              continue;
   700→            }
   701→            
   702→            try {
   703→              const content = await zipEntry.async('string');
   704→              // Safety: sanitize path to prevent path traversal
   705→              const safePath = relativePath.replace(/^\.+\//, '').replace(/^\//, '');
   706→              if (safePath) {
   707→                files[safePath] = content;
   708→                totalSize += content.length;
   709→              }
   710→            } catch (readError) {
   711→              console.warn(`Failed to read ${relativePath}:`, readError);
   712→            }
   713→          }
   714→        }
   715→        
   716→        // Safety: check total extracted size
   717→        if (totalSize > 100 * 1024 * 1024) { // 100MB total
   718→          if (IDEUtils) IDEUtils.showToast('Extracted content exceeds size limit. Import may be incomplete.', 'warning');
   719→        }
   720→        
   721→        // Import files to storage
   722→        await IDEStorage.Files.importFiles(project.name, files);
   723→        
   724→        // Update project file count
   725→        await IDEStorage.Projects.updateFileCount(project.name);
   726→        
   727→        // Set as current project
   728→        if (window.IDEState) {
   729→          await IDEState.setCurrentProject(project);
   730→        }
   731→        
   732→        // Load data
   733→        await loadProjectData(project);
   734→        
   735→        // Hide modal and reset
   736→        hideModal(document.getElementById('import-modal'));
   737→        fileInput.value = '';
   738→        const nameInput = document.getElementById('import-name-input');
   739→        if (nameInput) nameInput.value = '';
   740→        
   741→        if (IDEUtils) IDEUtils.showToast(`Imported ${Object.keys(files).length} files`, 'success');
   742→        switchTab('files');
   743→      }
   744→      
   745→    } catch (error) {
   746→      console.error('Failed to import:', error);
   747→      if (IDEUtils) IDEUtils.showToast(`Import failed: ${error.message}`, 'error');
   748→    }
   749→  }

  1159→  async function performSearch() {
  1160→    const input = document.getElementById('search-input');
  1161→    if (!input) return;
  1162→    
  1163→    const query = input.value.trim();
  1164→    if (!query) return;
  1165→    
  1166→    const project = IDEState?.get('currentProject');
  1167→    if (!project) {
  1168→      if (IDEUtils) IDEUtils.showToast('Select a project first', 'warning');
  1169→      return;
  1170→    }
  1171→    
  1172→    // Show loading state
  1173→    if (window.IDEState) {
  1174→      IDEState.setSearching(true);
  1175→      IDEState.setSearchQuery(query);
  1176→    }
  1177→    
  1178→    try {
  1179→      // Search in local files
  1180→      const files = await IDEStorage.Files.getByProject(project.name);
  1181→      const results = [];
  1182→      
  1183→      const lowerQuery = query.toLowerCase();
  1184→      files.forEach(file => {
  1185→        if (file.type === 'file' && file.content) {
  1186→          const lines = file.content.split('\n');
  1187→          lines.forEach((line, index) => {
  1188→            if (line.toLowerCase().includes(lowerQuery)) {
  1189→              results.push({
  1190→                file: file.path,
  1191→                line: index + 1,
  1192→                content: line.trim(),
  1193→                match: line.toLowerCase().indexOf(lowerQuery)
  1194→              });
  1195→            }
  1196→          });
  1197→        }
  1198→      });
  1199→      
  1200→      // Update state with results
  1201→      if (window.IDEState) {
  1202→        IDEState.setSearchResults(results);
  1203→        IDEState.setSearching(false);
  1204→      }
  1205→      
  1206→      // Render results
  1207→      renderSearchResults(results, query);
  1208→      
  1209→      // Switch to search tab
  1210→      switchTab('search');
  1211→      
  1212→    } catch (error) {
  1213→      console.error('Search failed:', error);
  1214→      if (window.IDEState) {
  1215→        IDEState.setSearching(false);
  1216→      }
  1217→    }
  1218→  }

  1220→  function renderSearchResults(results, query) {
  1221→    if (!searchPanel) return;
  1222→    
  1223→    const content = searchPanel.querySelector('.results-container');
  1224→    if (!content) return;
  1225→    
  1226→    if (!results?.length) {
  1227→      content.innerHTML = `
  1228→        <div class="no-results">
  1229→          <span class="no-results-icon">${IDEUtils?.Icons?.search || ''}</span>
  1230→          <p>No results found for "${IDEUtils?.escapeHtml(query) || query}"</p>
  1231→        </div>
  1232→      `;
  1233→      return;
  1234→    }
  1235→    
  1236→    let html = `<p class="results-count">${results.length} results found</p><ul class="search-results">`;
  1237→    
  1238→    // Group by file
  1239→    const grouped = {};
  1240→    results.forEach(r => {
  1241→      if (!grouped[r.file]) grouped[r.file] = [];
  1242→      grouped[r.file].push(r);
  1243→    });
  1244→

  1345→  // Sidebar Resize Handle (adjustable expanded width)
  1346→  // ============================================
  1347→
  1348→  /**
  1349→   * Setup the drag handle on the right edge of the expanded sidebar.
  1350→   * Dragging resizes --sidebar-width-expanded on the sidebar element,
  1351→   * which both .sidebar and .sidebar-expanded read from. The custom
  1352→   * width is persisted to localStorage and restored on next load.
  1353→   */
  1354→  function setupSidebarResizeHandle() {
  1355→    const handle = sidebarElement?.querySelector('.sidebar-resize-handle');
  1356→    if (!handle) return;
  1357→
  1358→    const STORAGE_KEY = 'ide:sidebar-width';
  1359→    const MIN_REM = 12;   // 192px
  1360→    const MAX_REM = 32;   // 512px
  1361→    const DEFAULT_REM = 18;
  1362→
  1363→    // Restore saved width
  1364→    try {
  1365→      const saved = parseFloat(localStorage.getItem(STORAGE_KEY));
  1366→      if (!isNaN(saved) && saved >= MIN_REM && saved <= MAX_REM) {
  1367→        sidebarElement.style.setProperty('--sidebar-width-expanded', `${saved}rem`);
  1368→      }
  1369→    } catch (e) { /* ignore */ }
  1370→
  1371→    let dragging = false;
  1372→    let startX = 0;
  1373→    let startWidthRem = 0;
  1374→
  1375→    function onPointerDown(e) {
  1376→      // Only resize when sidebar is expanded (not collapsed/closed)
  1377→      if (sidebarElement.classList.contains('collapsed') ||
  1378→          sidebarElement.classList.contains('closed')) return;
  1379→      dragging = true;
  1380→      startX = e.clientX || (e.touches && e.touches[0]?.clientX) || 0;
  1381→      const current = getComputedStyle(sidebarElement).getPropertyValue('--sidebar-width-expanded').trim();
  1382→      const m = String(current).match(/([\d.]+)rem/);
  1383→      startWidthRem = m ? parseFloat(m[1]) : DEFAULT_REM;
  1384→      handle.classList.add('dragging');
  1385→      sidebarElement.classList.add('resizing');
  1386→      document.body.style.cursor = 'col-resize';
  1387→      document.body.style.userSelect = 'none';
  1388→      e.preventDefault();
  1389→    }
  1390→
  1391→    function onPointerMove(e) {
  1392→      if (!dragging) return;
  1393→      const x = e.clientX || (e.touches && e.touches[0]?.clientX) || 0;
  1394→      const deltaPx = x - startX;
  1395→      // Convert px to rem using the root font-size
  1396→      const rootFontSize = parseFloat(getComputedStyle(document.documentElement).fontSize) || 16;
  1397→      const deltaRem = deltaPx / rootFontSize;
  1398→      let newWidthRem = startWidthRem + deltaRem;
  1399→      newWidthRem = Math.max(MIN_REM, Math.min(MAX_REM, newWidthRem));
  1400→      sidebarElement.style.setProperty('--sidebar-width-expanded', `${newWidthRem}rem`);
  1401→    }
  1402→
  1403→    function onPointerUp() {
  1404→      if (!dragging) return;
  1405→      dragging = false;
  1406→      handle.classList.remove('dragging');
  1407→      sidebarElement.classList.remove('resizing');
  1408→      document.body.style.cursor = '';
  1409→      document.body.style.userSelect = '';
  1410→      // Persist
  1411→      const current = getComputedStyle(sidebarElement).getPropertyValue('--sidebar-width-expanded').trim();
  1412→      const m = String(current).match(/([\d.]+)rem/);
  1413→      if (m) {
  1414→        try { localStorage.setItem(STORAGE_KEY, m[1]); } catch (e) { /* ignore */ }
  1415→      }
  1416→    }
  1417→
  1418→    handle.addEventListener('mousedown', onPointerDown);
  1419→    handle.addEventListener('touchstart', onPointerDown, { passive: false });
  1420→    document.addEventListener('mousemove', onPointerMove);
  1421→    document.addEventListener('touchmove', onPointerMove, { passive: false });
  1422→    document.addEventListener('mouseup', onPointerUp);
  1423→    document.addEventListener('touchend', onPointerUp);
  1424→
  1425→    // Keyboard support: arrow keys adjust width by 1rem
  1426→    handle.addEventListener('keydown', (e) => {
  1427→      if (sidebarElement.classList.contains('collapsed') ||
  1428→          sidebarElement.classList.contains('closed')) return;
  1429→      const current = getComputedStyle(sidebarElement).getPropertyValue('--sidebar-width-expanded').trim();
  1430→      const m = String(current).match(/([\d.]+)rem/);
  1431→      let w = m ? parseFloat(m[1]) : DEFAULT_REM;
  1432→      if (e.key === 'ArrowLeft')  { w = Math.max(MIN_REM, w - 1); e.preventDefault(); }
  1433→      else if (e.key === 'ArrowRight') { w = Math.min(MAX_REM, w + 1); e.preventDefault(); }
  1434→      else if (e.key === 'Enter' || e.key === ' ') { w = DEFAULT_REM; e.preventDefault(); }
  1435→      else return;
  1436→      sidebarElement.style.setProperty('--sidebar-width-expanded', `${w}rem`);
  1437→      try { localStorage.setItem(STORAGE_KEY, String(w)); } catch (e2) { /* ignore */ }
  1438→    });
  1439→  }
