     1→/**
     2→ * SBIDE - Message Bubble Component
     3→ * Renders individual chat messages with markdown, code blocks, and action buttons
     4→ * Ported from React MessageBubble.tsx with full feature parity
     5→ */
     6→
     7→const MessageBubbleComponent = (() => {
     8→  // ============================================
     9→  // State
    10→  // ============================================
    11→  
    12→  let container = null;
    13→  let message = null;
    14→  let isStreaming = false;
    15→  
    16→  // Callbacks
    17→  let onRegenerate = null;
    18→  let onRollback = null;
    19→  let onEditAndRegenerate = null;
    20→  
    21→  // Dialog state
    22→  let activeDialog = null;
    23→
    24→  // ============================================
    25→  // Initialization
    26→  // ============================================
    27→  
    28→  /**
    29→   * Initialize message bubble component
    30→   * @param {HTMLElement} element - Container element for the bubble
    31→   * @param {Object} options - Configuration options
    32→   */
    33→  function init(element, options = {}) {
    34→    if (!element) {
    35→      console.error('MessageBubble: Container element required');
    36→      return;
    37→    }
    38→    
    39→    container = element;
    40→    onRegenerate = options.onRegenerate || null;
    41→    onRollback = options.onRollback || null;
    42→    onEditAndRegenerate = options.onEditAndRegenerate || null;
    43→  }
    44→
    45→  // ============================================
    46→  // Rendering
    47→  // ============================================
    48→  
    49→  /**
    50→   * Render a message bubble
    51→   * @param {Object} msg - Message object { id, role, content, timestamp, ... }
    52→   * @param {Object} renderOptions - Additional options
    53→   */
    54→  function render(msg, renderOptions = {}) {
    55→    if (!container) return;
    56→    
    57→    message = msg;
    58→    isStreaming = renderOptions.isStreaming || false;
    59→    onRegenerate = renderOptions.onRegenerate || onRegenerate;
    60→    onRollback = renderOptions.onRollback || onRollback;
    61→    onEditAndRegenerate = renderOptions.onEditAndRegenerate || onEditAndRegenerate;
    62→    
    63→    const isUser = msg.role === 'user';
    64→    const isSystem = msg.role === 'system';
    65→    
    66→    // System message (centered)
    67→    if (isSystem) {
    68→      container.innerHTML = `
    69→        <div class="message-bubble message-system">
    70→          <span class="message-system-content">${escapeHtml(msg.content)}</span>
    71→        </div>
    72→      `;
    73→      return;
    74→    }
    75→    
    76→    // Regular message (user or assistant)
    77→    const formattedTime = formatTime(msg.timestamp);
    78→    const avatarHtml = renderAvatar(isUser);
    79→    const contentHtml = renderContent(msg, isUser);
    80→    const metaHtml = renderMetadata(msg, isUser, formattedTime);
    81→    const actionsHtml = !isStreaming ? renderActions(msg, isUser) : '';
    82→    const sourcesHtml = !isUser && msg.sources && msg.sources.length > 0 ? renderSources(msg.sources) : '';
    83→    const streamingProgressHtml = isStreaming && !isUser && msg.content ? renderStreamingProgress(msg.content) : '';
    84→    
    85→    container.innerHTML = `
    86→      <div class="message-bubble ${isUser ? 'message-user' : 'message-assistant'}" data-message-id="${msg.id}">
    87→        <div class="message-avatar">${avatarHtml}</div>
    88→        <div class="message-body">
    89→          <div class="message-content ${isUser ? 'user-content' : 'assistant-content'}">
    90→            ${contentHtml}
    91→            ${isStreaming && isUser ? '<span class="streaming-cursor"></span>' : ''}
    92→          </div>
    93→          ${streamingProgressHtml}
    94→          ${metaHtml}
    95→          ${actionsHtml}
    96→          ${sourcesHtml}
    97→        </div>
    98→      </div>
    99→      
   100→      <!-- Dialogs will be appended here when needed -->
   101→      <div class="message-dialogs"></div>
   102→    `;
   103→    
   104→    // Wire up event listeners
   105→    wireUpActions(container, msg, isUser);
   106→    
   107→    // Apply syntax highlighting to code blocks
   108→    applySyntaxHighlighting(container);
   109→  }

   132→  /**
   133→   * Render message content with markdown support
   134→   */
   135→  function renderContent(msg, isUser) {
   136→    const content = msg.content || '';
   137→    
   138→    // Show typing indicator for empty streaming messages
   139→    if (isStreaming && !content) {
   140→      return `
   141→        <div class="typing-indicator">
   142→          <span class="typing-dot"></span>
   143→          <span class="typing-dot"></span>
   144→          <span class="typing-dot"></span>
   145→        </div>
   146→      `;
   147→    }
   148→    
   149→    // Parse and render markdown
   150→    return parseMarkdown(content, isUser);
   151→  }
   152→  
   153→  /**
   154→   * Parse markdown content to HTML
   155→   */
   156→  function parseMarkdown(content, isUser) {
   157→    if (!content) return '';
   158→    
   159→    let html = escapeHtml(content);
   160→    
   161→    // Code blocks (```language\ncode\n```)
   162→    html = html.replace(/```(\w*)\n([\s\S]*?)```/g, (match, language, code) => {
   163→      return renderCodeBlock(code.trim(), language.trim(), isUser);
   164→    });
   165→    
   166→    // Inline code (`code`)
   167→    html = html.replace(/`([^`]+)`/g, '<code class="inline-code">$1</code>');
   168→    
   169→    // Bold (**text**)
   170→    html = html.replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>');
   171→    
   172→    // Italic (*text*)
   173→    html = html.replace(/\*([^*]+)\*/g, '<em>$1</em>');
   174→    
   175→    // Headers
   176→    html = html.replace(/^### (.+)$/gm, '<h3>$1</h3>');
   177→    html = html.replace(/^## (.+)$/gm, '<h2>$1</h2>');
   178→    html = html.replace(/^# (.+)$/gm, '<h1>$1</h1>');
   179→    
   180→    // Links [text](url)
   181→    html = html.replace(/\[([^\]]+)\]\(([^)]+)\)/g, '<a href="$2" target="_blank" rel="noopener noreferrer">$1</a>');
   182→    
   183→    // Line breaks
   184→    html = html.replace(/\n/g, '<br>');
   185→    
   186→    // Paragraphs (double newline)
   187→    html = html.replace(/<br><br>/g, '</p><p>');
   188→    
   189→    return `<div class="markdown-content">${html}</div>`;
   190→  }

   202→        <div class="code-block-header">
   203→          <div class="code-block-info">
   204→            <span class="code-language">${language || 'text'}</span>
   205→            ${isLarge ? `<span class="code-line-count">${lineCount} lines</span>` : ''}
   206→          </div>
   207→          <button class="code-copy-btn icon-btn xs" title="Copy code" data-code="${escapeAttr(code)}">
   208→            <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
   209→              <rect x="9" y="9" width="13" height="13" rx="2" ry="2"/>
   210→              <path d="M5 15H4a2 2 0 01-2-2V4a2 2 0 012-2h9a2 2 0 012 2v1"/>
   211→            </svg>
   212→            <span>Copy</span>
   213→          </button>
   214→        </div>
   215→        <div class="code-block-body ${isLarge ? 'collapsed' : ''}">
   216→          <pre class="code-pre"><code class="language-${language}">${code}</code></pre>
   217→          ${isLarge ? `
   218→            <div class="code-expand-overlay">

   220→                <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">

   227→        </div>
   228→        ${isLarge ? `
   229→          <div class="code-block-footer hidden">
   230→            <button class="code-collapse-btn btn btn-ghost btn-xs">
   231→              <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">

   233→              </svg>
   234→              Collapse
   235→            </button>
   236→          </div>
   237→        ` : ''}
   238→      </div>
   239→    `;
   240→  }

   242→  /**
   243→   * Render metadata (timestamp, badges)
   244→   */
   245→  function renderMetadata(msg, isUser, formattedTime) {
   246→    const badges = [];
   247→    
   248→    // Files badge
   249→    if ((msg.filesCreated?.length || msg.filesModified?.length)) {
   250→      badges.push(`
   251→        <span class="message-badge badge-files">
   252→          <svg width="10" height="10" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
   253→            <path d="M13 2H6a2 2 0 00-2 2v16a2 2 0 002 2h12a2 2 0 002-2V9z"/>
   254→            <polyline points="13 2 13 9 20 9"/>
   255→          </svg>
   256→          ${(msg.filesCreated?.length || 0) + (msg.filesModified?.length || 0)} files
   257→        </span>
   258→      `);
   259→    }
   260→    
   261→    // Anchors badge
   262→    if (msg.memoryAnchors && msg.memoryAnchors.length > 0) {
   263→      badges.push(`
   264→        <span class="message-badge badge-anchors">
   265→          <svg width="10" height="10" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
   266→            <circle cx="12" cy="5" r="3"/>
   267→            <line x1="12" y1="22" x2="12" y2="8"/>
   268→            <path d="M5 12H2a10 10 0 0020 0h-3"/>
   269→          </svg>
   270→          ${msg.memoryAnchors.length} anchors
   271→        </span>
   272→      `);
   273→    }
   274→    
   275→    // Confidence badge
   276→    if (msg.confidence !== undefined) {
   277→      const confidenceClass = msg.confidence > 0.8 ? 'confidence-high' : msg.confidence > 0.5 ? 'confidence-medium' : 'confidence-low';
   278→      badges.push(`<span class="message-badge ${confidenceClass}">${Math.round(msg.confidence * 100)}% confident</span>`);
   279→    }
   280→    
   281→    return `
   282→      <div class="message-meta ${isUser ? 'meta-user' : 'meta-assistant'}">
   283→        <span class="message-time" title="${new Date(msg.timestamp).toLocaleString()}">
   284→          <svg width="10" height="10" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
   285→            <circle cx="12" cy="12" r="10"/>
   286→            <polyline points="12 6 12 12 16 14"/>
   287→          </svg>
   288→          ${formattedTime}
   289→        </span>
   290→        ${badges.join('')}
   291→      </div>
   292→    `;
   293→  }

   412→  /**
   413→   * Wire up action button events
   414→   */
   415→  function wireUpActions(bubbleEl, msg, isUser) {
   416→    // Copy action
   417→    bubbleEl.querySelector('.btn-action-copy')?.addEventListener('click', () => handleCopy(msg));
   418→    
   419→    // Edit action (user only)
   420→    bubbleEl.querySelector('.btn-action-edit')?.addEventListener('click', () => handleEdit(msg));
   421→    
   422→    // Delete action
   423→    bubbleEl.querySelector('.btn-action-delete')?.addEventListener('click', () => handleDelete(msg));
   424→    
   425→    // Rollback action (AI only)
   426→    bubbleEl.querySelector('.btn-action-rollback')?.addEventListener('click', () => handleRollback(msg));
   427→    
   428→    // Regenerate action (AI only)
   429→    bubbleEl.querySelector('.btn-action-regenerate')?.addEventListener('click', () => handleRegenerate(msg));
   430→    
   431→    // Code block copy buttons
   432→    bubbleEl.querySelectorAll('.code-copy-btn').forEach(btn => {
   433→      btn.addEventListener('click', () => handleCodeCopy(btn));
   434→    });
   435→    
   436→    // Code block expand/collapse
   437→    bubbleEl.querySelector('.code-expand-btn')?.addEventListener('click', (e) => {
   438→      const block = e.target.closest('.code-block');
   439→      expandCodeBlock(block);
   440→    });
   441→    
   442→    bubbleEl.querySelector('.code-collapse-btn')?.addEventListener('click', (e) => {
   443→      const block = e.target.closest('.code-block');
   444→      collapseCodeBlock(block);
   445→    });
   446→  }

   448→  /**
   449→   * Handle copy message
   450→   */
   451→  async function handleCopy(msg) {
   452→    try {
   453→      await navigator.clipboard.writeText(msg.content || '');
   454→      if (IDEUtils) IDEUtils.showToast('Copied to clipboard', 'success');
   455→      
   456→      // Update button state briefly
   457→      const btn = container.querySelector('.btn-action-copy');
   458→      if (btn) {
   459→        btn.classList.add('copied');
   460→        setTimeout(() => btn.classList.remove('copied'), 2000);
   461→      }
   462→    } catch (err) {
   463→      console.error('Failed to copy:', err);
   464→      if (IDEUtils) IDEUtils.showToast('Failed to copy', 'error');
   465→    }
   466→  }
   467→  
   468→  /**
   469→   * Handle copy code block
   470→   */
   471→  async function handleCodeCopy(btn) {
   472→    const code = btn.dataset.code;
   473→    if (!code) return;
   474→    
   475→    try {
   476→      await navigator.clipboard.writeText(code);
   477→      btn.classList.add('copied');
   478→      btn.innerHTML = `
   479→        <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
   480→          <polyline points="20 6 9 17 4 12"/>
   481→        </svg>
   482→        <span>Copied!</span>
   483→      `;
   484→      setTimeout(() => {
   485→        btn.classList.remove('copied');
   486→        btn.innerHTML = `
   487→          <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
   488→            <rect x="9" y="9" width="13" height="13" rx="2" ry="2"/>
   489→            <path d="M5 15H4a2 2 0 01-2-2V4a2 2 0 012-2h9a2 2 0 012 2v1"/>
   490→          </svg>
   491→          <span>Copy</span>
   492→        `;
   493→      }, 2000);
   494→    } catch (err) {
   495→      console.error('Failed to copy code:', err);
   496→    }
   497→  }

   499→  /**
   500→   * Expand large code block
   501→   */
   502→  function expandCodeBlock(block) {
   503→    const body = block?.querySelector('.code-block-body');
   504→    const overlay = block?.querySelector('.code-expand-overlay');

   508→    if (overlay) overlay.style.display = 'none';
   509→    if (footer) footer.classList.remove('hidden');
   510→  }

   515→  function collapseCodeBlock(block) {

   752→  /**
   753→   * Wire up common dialog events
   754→   */
   755→  function wireUpDialogEvents(dialogEl, { onCancel, onConfirm }) {
   756→    dialogEl.querySelector('.cancel-dialog-btn, .dialog-close-btn')?.addEventListener('click', onCancel);
   757→    dialogEl.querySelector('.confirm-edit-btn, .confirm-delete-btn, .confirm-rollback-btn, .confirm-regenerate-btn')?.addEventListener('click', onConfirm);
   758→    
   759→    dialogEl.addEventListener('click', (e) => {
   760→      if (e.target === dialogEl) onCancel();
   761→    });
   762→    
   763→    dialogEl.addEventListener('keydown', (e) => {
   764→      if (e.key === 'Escape') onCancel();
   765→    });
   766→  }

   798→   * Format timestamp to time string
   799→   */
   800→  function formatTime(timestamp) {
   801→    if (!timestamp) return '';
   802→    return new Date(timestamp).toLocaleTimeString([], {
   803→      hour: '2-digit',
   804→      minute: '2-digit'
   805→    });
   806→  }

   808→  /**
   809→   * Escape HTML entities
   810→   */
   811→  function escapeHtml(text) {
   812→    if (!text) return '';
   813→    const div = document.createElement('div');
   814→    div.textContent = text;
   815→    return div.innerHTML;
   816→  }
   817→  
   818→  /**
   819→   * Escape HTML attribute value
   820→   */
   821→  function escapeAttr(text) {
   822→    if (!text) return '';
   823→    return text
   824→      .replace(/&/g, '&amp;')
   825→      .replace(/"/g, '&quot;')
   826→      .replace(/'/g, '&#39;')
   827→      .replace(/</g, '&lt;')
   828→      .replace(/>/g, '&gt;');
   829→  }
