   367→  /**
   368→   * Wire up all modal event listeners
   369→   */
   370→  function wireUpModalEvents(modalEl) {
   371→    // Close button
   372→    modalEl.querySelector('.settings-close-btn').addEventListener('click', close);
   373→    
   374→    // Click outside to close
   375→    modalEl.addEventListener('click', (e) => {
   376→      if (e.target === modalEl) close();
   377→    });
   378→    
   379→    // Escape key
   380→    modalEl.addEventListener('keydown', (e) => {
   381→      if (e.key === 'Escape') close();
   382→    });
   383→    
   384→    // Theme selection
   385→    modalEl.querySelectorAll('.theme-option').forEach(btn => {
   386→      btn.addEventListener('click', () => {
   387→        selectTheme(btn.dataset.themeId);
   388→      });
   389→    });
   390→    
   391→    // Font scale selection
   392→    modalEl.querySelectorAll('.font-scale-option').forEach(btn => {
   393→      btn.addEventListener('click', () => {
   394→        setFontScale(parseFloat(btn.dataset.fontScale));
   395→      });
   396→    });
   397→    
   398→    // Toggle switches
   399→    modalEl.querySelectorAll('.toggle-input').forEach(toggle => {
   400→      toggle.addEventListener('change', () => {
   401→        updateToggleSetting(toggle.id, toggle.checked);
   402→      });
   403→    });
   404→    
   405→    // System prompt textarea
   406→    const promptTextarea = modalEl.querySelector('#system-prompt-textarea');
   407→    if (promptTextarea) {
   408→      const charCount = modalEl.querySelector('#prompt-char-count');
   409→      
   410→      // Update character count on input
   411→      promptTextarea.addEventListener('input', () => {
   412→        if (charCount) charCount.textContent = promptTextarea.value.length;
   413→        modal.dataset.systemPrompt = promptTextarea.value;
   414→      });
   415→      
   416→      // Initialize character count
   417→      if (charCount) charCount.textContent = promptTextarea.value.length;
   418→      
   419→      // Reset button
   420→      modalEl.querySelector('.reset-prompt-btn')?.addEventListener('click', () => {
   421→        promptTextarea.value = getDefaultSystemPrompt();
   422→        if (charCount) charCount.textContent = promptTextarea.value.length;
   423→        modal.dataset.systemPrompt = promptTextarea.value;
   424→        if (IDEUtils) IDEUtils.showToast('System prompt reset to default', 'info');
   425→      });
   426→      
   427→      // Expand button
   428→      modalEl.querySelector('.expand-prompt-btn')?.addEventListener('click', () => {
   429→        openExpandedPromptEditor(promptTextarea.value, (newValue) => {
   430→          promptTextarea.value = newValue;
   431→          if (charCount) charCount.textContent = promptTextarea.value.length;
   432→          modal.dataset.systemPrompt = newValue;
   433→        });
   434→      });
   435→      
   436→      // Preset buttons
   437→      modalEl.querySelectorAll('.preset-btn').forEach(btn => {
   438→        btn.addEventListener('click', () => {
   439→          const preset = getPresetPrompt(btn.dataset.prompt);
   440→          if (preset) {
   441→            promptTextarea.value = preset;
   442→            if (charCount) charCount.textContent = promptTextarea.value.length;
   443→            modal.dataset.systemPrompt = preset;
   444→            if (IDEUtils) IDEUtils.showToast(`Applied "${btn.textContent.trim()}" template`, 'success');
   445→          }
   446→        });
   447→      });
   448→    }
   449→    
   450→    // Save button
   451→    modalEl.querySelector('.save-settings-btn').addEventListener('click', () => {
   452→      saveSettings();
   453→      close();
   454→      if (IDEUtils) IDEUtils.showToast('Settings saved', 'success');
   455→    });
   456→    
   457→    // Reset button
   458→    modalEl.querySelector('.reset-settings-btn').addEventListener('click', () => {
   459→      if (confirm('Reset all settings to defaults?')) {
   460→        resetToDefaults();
   461→        renderModal(); // Re-render with defaults
   462→        if (IDEUtils) IDEUtils.showToast('Settings reset to defaults', 'info');
   463→      }
   464→    });
   465→  }

   490→  /**
   491→   * Select a theme
   492→   */
   493→  function selectTheme(themeId) {
   494→    // Update UI
   495→    modal.querySelectorAll('.theme-option').forEach(btn => {
   496→      const isActive = btn.dataset.themeId === themeId;
   497→      btn.classList.toggle('active', isActive);
   498→      btn.setAttribute('aria-checked', isActive);
   499→      
   500→      // Update checkmark
   501→      let check = btn.querySelector('.theme-check');
   502→      if (isActive && !check) {
   503→        check = document.createElement('span');
   504→        check.className = 'theme-check';
   505→        check.textContent = '✓';
   506→        btn.appendChild(check);
   507→      } else if (!isActive && check) {
   508→        check.remove();
   509→      }
   510→    });
   511→    
   512→    // Apply theme through the central IDEThemes module so that:
   513→    //   - the data-theme attribute is set
   514→    //   - the theme-X CSS class is added (used by per-theme overrides in styles.css)
   515→    //   - the meta theme-color is updated for mobile browsers
   516→    //   - the choice is persisted to IDEState
   517→    //   - the 'themechange' event fires for other components
   518→    if (window.IDEThemes) {
   519→      IDEThemes.applyTheme(themeId);
   520→    } else {
   521→      // Fallback: direct attribute only (legacy path)
   522→      document.documentElement.setAttribute('data-theme', themeId);
   523→    }
   524→    
   525→    // Store pending change
   526→    modal.dataset.pendingTheme = themeId;
   527→  }

   564→  function saveSettings() {
   565→    const prev = getSettings();
   566→    const settings = {
   567→      theme: modal?.dataset.pendingTheme || prev.theme,
   568→      fontScale: parseFloat(modal?.dataset.pendingFontScale || prev.fontScale),
   569→      autoCreateZipBackup: modal?.dataset['setting-zip-backup'] === null ? prev.autoCreateZipBackup
   570→                          : modal?.dataset['setting-zip-backup'] === 'true',
   571→      showNotifications: modal?.dataset['setting-notifications'] === null ? prev.showNotifications
   572→                          : modal?.dataset['setting-notifications'] === 'true',
   573→      autoCheckpoint: modal?.dataset['setting-auto-checkpoint'] === null ? prev.autoCheckpoint
   574→                          : modal?.dataset['setting-auto-checkpoint'] === 'true',
   575→      wordWrap: modal?.dataset['setting-word-wrap'] === null ? prev.wordWrap
   576→                  : modal?.dataset['setting-word-wrap'] === 'true',
   577→      lineNumbers: modal?.dataset['setting-line-numbers'] === null ? prev.lineNumbers
   578→                    : modal?.dataset['setting-line-numbers'] === 'true',
   579→      autoSave: modal?.dataset['setting-auto-save'] === null ? prev.autoSave
   580→                  : modal?.dataset['setting-auto-save'] === 'true',
   581→      soundEffects: modal?.dataset['setting-sound-effects'] === null ? prev.soundEffects
   582→                      : modal?.dataset['setting-sound-effects'] === 'true',
   583→      systemPrompt: modal?.dataset.systemPrompt !== undefined 
   584→                    ? modal.dataset.systemPrompt 
   585→                    : prev.systemPrompt || ''
   586→    };
   587→    
   588→    // Persist to storage
   589→    try {
   590→      localStorage.setItem('ide-settings', JSON.stringify(settings));
   591→    } catch (e) {
   592→      console.error('Failed to save settings:', e);
   593→    }
   594→    
   595→    // Update state
   596→    if (window.IDEState) {
   597→      IDEState.set({ settings });
   598→    }
   599→    
   600→    // Apply settings
   601→    applySettings(settings);
   602→    
   603→    // Notify other components that editor-affecting settings may have changed
   604→    document.dispatchEvent(new CustomEvent('ide:settings-applied', { detail: { settings, prev } }));
   605→  }
