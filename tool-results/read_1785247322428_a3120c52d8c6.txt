   138→  /**
   139→   * Detect programming language from filename/extension
   140→   */
   141→  function detectLanguage(filename) {
   142→    if (!filename) return 'plaintext';
   143→    const lowerName = filename.toLowerCase();
   144→    
   145→    // Check special filenames first
   146→    if (lowerName === 'dockerfile') return 'dockerfile';
   147→    if (lowerName === 'makefile') return 'makefile';
   148→    if (lowerName === '.gitignore' || lowerName === '.env') return 'ini';
   149→    if (lowerName === 'readme.md' || lowerName === 'changelog.md') return 'markdown';
   150→    
   151→    // Get extension
   152→    const lastDotIndex = lowerName.lastIndexOf('.');
   153→    if (lastDotIndex === -1) return 'plaintext';
   154→    
   155→    const ext = lowerName.substring(lastDotIndex);
   156→    return LanguageMap[ext] || 'plaintext';
   157→  }

   159→  /**
   160→   * Get file icon based on extension/type
   161→   */
   162→  function getFileIcon(filename, type) {
   163→    if (type === 'folder') return Icons.folder;
   164→    
   165→    if (!filename) return Icons.file;
   166→    const ext = filename.substring(filename.lastIndexOf('.')).toLowerCase();
   167→    
   168→    switch (ext) {
   169→      case '.js':
   170→      case '.ts':
   171→      case '.jsx':
   172→      case '.tsx':
   173→        return Icons.fileCode;
   174→      case '.json':
   175→        return Icons.fileJson;
   176→      case '.png':
   177→      case '.jpg':
   178→      case '.jpeg':
   179→      case '.gif':
   180→      case '.svg':
   181→      case '.webp':
   182→        return Icons.fileImage;
   183→      case '.md':
   184→      case '.markdown':
   185→        return Icons.fileMarkdown;
   186→      default:
   187→        return Icons.file;
   188→    }
   189→  }

   195→  /**
   196→   * Format bytes to human-readable size
   197→   */
   198→  function formatBytes(bytes, decimals = 2) {
   199→    if (!bytes || bytes === 0) return '0 B';
   200→    
   201→    const k = 1024;
   202→    const sizes = ['B', 'KB', 'MB', 'GB'];
   203→    const i = Math.floor(Math.log(bytes) / Math.log(k));
   204→    
   205→    return parseFloat((bytes / Math.pow(k, i)).toFixed(decimals)) + ' ' + sizes[i];
   206→  }

   208→  /**
   209→   * Format timestamp to relative time
   210→   */
   211→  function formatRelativeTime(timestamp) {
   212→    if (!timestamp) return '';
   213→    
   214→    const now = Date.now();
   215→    const diff = now - timestamp;
   216→    const seconds = Math.floor(diff / 1000);
   217→    const minutes = Math.floor(seconds / 60);
   218→    const hours = Math.floor(minutes / 60);
   219→    const days = Math.floor(hours / 24);
   220→    
   221→    if (seconds < 60) return 'just now';
   222→    if (minutes < 60) return `${minutes}m ago`;
   223→    if (hours < 24) return `${hours}h ago`;
   224→    if (days < 7) return `${days}d ago`;
   225→    
   226→    return new Date(timestamp).toLocaleDateString();
   227→  }

   229→  /**
   230→   * Format date to locale string
   231→   */
   232→  function formatDate(timestamp, options = {}) {
   233→    if (!timestamp) return '';
   234→    const defaults = { 
   235→      month: 'short', 
   236→      day: 'numeric',
   237→      hour: '2-digit',
   238→      minute: '2-digit'
   239→    };
   240→    return new Date(timestamp).toLocaleDateString(undefined, { ...defaults, ...options });
   241→  }

   243→  /**
   244→   * Truncate text with ellipsis
   245→   */
   246→  function truncate(text, maxLength = 50) {
   247→    if (!text) return '';
   248→    if (text.length <= maxLength) return text;

   273→  /**
   274→   * Debounce function execution
   275→   */
   276→  function debounce(fn, delay = 300) {
   277→    let timeoutId;
   278→    return function (...args) {
   279→      clearTimeout(timeoutId);
   280→      timeoutId = setTimeout(() => fn.apply(this, args), delay);
   281→    };
   282→  }

   284→  /**
   285→   * Throttle function execution
   286→   */
   287→  function throttle(fn, limit = 100) {
   288→    let inThrottle;
   289→    return function (...args) {
   290→      if (!inThrottle) {
   291→        fn.apply(this, args);
   292→        inThrottle = true;
   293→        setTimeout(() => inThrottle = false, limit);
   294→      }
   295→    };
   296→  }

   316→  /**
   317→   * Create element with optional attributes and children
   318→   */
   319→  function createElement(tag, attrs = {}, children = []) {
   320→    const el = document.createElement(tag);
   321→    
   322→    Object.entries(attrs).forEach(([key, value]) => {
   323→      if (key === 'className') {
   324→        el.className = value;
   325→      } else if (key.startsWith('on') && typeof value === 'function') {
   326→        el.addEventListener(key.slice(2).toLowerCase(), value);
   327→      } else if (key === 'innerHTML') {
   328→        el.innerHTML = value;
   329→      } else if (key === 'textContent') {
   330→        el.textContent = value;
   331→      } else {
   332→        el.setAttribute(key, value);
   333→      }
   334→    });
   335→    
   336→    children.forEach(child => {
   337→      if (typeof child === 'string') {
   338→        el.insertAdjacentHTML('beforeend', child);
   339→      } else if (child instanceof Node) {
   340→        el.appendChild(child);
   341→      }
   342→    });
   343→    
   344→    return el;
   345→  }

   388→  let audioCtx = null;

   390→  /**
   391→   * Lazily create a shared Web Audio context (must be created on user gesture).
   392→   */
   393→  function getAudioCtx() {
   394→    if (audioCtx) return audioCtx;
   395→    try {
   396→      const Ctor = window.AudioContext || window.webkitAudioContext;
   397→      if (!Ctor) return null;
   398→      audioCtx = new Ctor();
   399→    } catch (e) {
   400→      console.warn('Web Audio API unavailable:', e);
   401→      return null;
   402→    }
   403→    return audioCtx;
   404→  }

   406→  /**
   407→   * Returns true if sound effects are currently enabled in settings.
   408→   */
   409→  function soundEffectsEnabled() {
   410→    if (window.IDEState) {
   411→      const s = IDEState.get('settings');
   412→      if (s && typeof s.soundEffects === 'boolean') return s.soundEffects;
   413→    }
   414→    // Fallback: check localStorage directly
   415→    try {
   416→      const stored = JSON.parse(localStorage.getItem('ide-settings') || '{}');
   417→      return !!stored.soundEffects;
   418→    } catch {
   419→      return false;
   420→    }
   421→  }

   423→  /**
   424→   * Play a short synthesized sound cue. No asset files needed.
   425→   * @param {'save'|'error'|'success'|'info'|'click'} type
   426→   */
   427→  function playSound(type = 'info') {
   428→    if (!soundEffectsEnabled()) return;
   429→    const ctx = getAudioCtx();
   430→    if (!ctx) return;
   431→    
   432→    // Some browsers suspend audio until a user gesture resumes it
   433→    if (ctx.state === 'suspended') {
   434→      ctx.resume().catch(() => {});
   435→    }
   436→    
   437→    const now = ctx.currentTime;
   438→    const osc = ctx.createOscillator();
   439→    const gain = ctx.createGain();
   440→    osc.connect(gain);
   441→    gain.connect(ctx.destination);
   442→    
   443→    // Type-specific envelope and frequency
   444→    let freq = 440;
   445→    let endFreq = 660;
   446→    let dur = 0.08;
   447→    let peak = 0.06;
   448→    switch (type) {
   449→      case 'save':
   450→      case 'success':
   451→        freq = 523.25; endFreq = 783.99; dur = 0.12; peak = 0.07; // C5 -> G5
   452→        osc.type = 'sine';
   453→        break;
   454→      case 'error':
   455→        freq = 311.13; endFreq = 207.65; dur = 0.18; peak = 0.08; // Eb4 -> Ab3
   456→        osc.type = 'square';
   457→        break;
   458→      case 'click':
   459→        freq = 660; endFreq = 660; dur = 0.04; peak = 0.04;
   460→        osc.type = 'triangle';
   461→        break;
   462→      case 'info':
   463→      default:
   464→        freq = 440; endFreq = 523.25; dur = 0.08; peak = 0.05; // A4 -> C5
   465→        osc.type = 'sine';
   466→        break;
   467→    }
   468→    
   469→    osc.frequency.setValueAtTime(freq, now);
   470→    osc.frequency.exponentialRampToValueAtTime(Math.max(20, endFreq), now + dur);
   471→    gain.gain.setValueAtTime(0.0001, now);
   472→    gain.gain.exponentialRampToValueAtTime(peak, now + 0.01);
   473→    gain.gain.exponentialRampToValueAtTime(0.0001, now + dur);
   474→    
   475→    osc.start(now);
   476→    osc.stop(now + dur + 0.02);
   477→  }

   483→  /**
   484→   * Show toast notification
   485→   */
   486→  function showToast(message, type = 'info', duration = 4000) {
   487→    const container = document.getElementById('toast-container');
   488→    if (!container) return;
   489→

   498→      `<span>${escapeHtml(message)}</span>`,
   499→      `<button class="icon-btn toast-close" aria-label="Dismiss">${Icons.x}</button>`
   500→    ]);

   504→    // Trigger animation

   506→    
   507→    // Close button handler
   508→    toast.querySelector('.toast-close').addEventListener('click', () => removeToast(toast));
   509→    
   510→    // Auto-remove after duration
   511→    setTimeout(() => removeToast(toast), duration);
   512→  }

   526→  async function copyToClipboard(text) {
   527→    try {
   528→      await navigator.clipboard.writeText(text);
   529→      showToast('Copied to clipboard!', 'success');
   530→      return true;
   531→    } catch (error) {
   532→      console.error('Failed to copy:', error);
   533→      // Fallback for older browsers
   534→      const textarea = document.createElement('textarea');
   535→      textarea.value = text;
   536→      textarea.style.position = 'fixed';
   537→      textarea.style.opacity = '0';
   538→      document.body.appendChild(textarea);
   539→      textarea.select();
   540→      document.execCommand('copy');
   541→      document.body.removeChild(textarea);
   542→      showToast('Copied!', 'success');
   543→      return true;
   544→    }
   545→  }

   554→  function downloadFile(content, filename, mimeType = 'text/plain') {

   577→  /**
   578→   * Convert any string to a URL-safe kebab-case slug
   579→   * "My Awesome Project!" -> "my-awesome-project"
   580→   */
   581→  function slugify(text) {
   582→    if (!text) return '';
   583→    
   584→    return text
   585→      .trim()
   586→      .toLowerCase()
   587→      // Replace spaces and special chars with hyphens
   588→      .replace(/[\s_]+/g, '-')                    // spaces/underscores → hyphens
   589→      .replace(/[^a-z0-9-]/g, '')                  // remove non-alphanumeric (except hyphen)
   590→      .replace(/-+/g, '-')                          // collapse multiple hyphens
   591→      .replace(/^-|-$/g, '')                        // trim leading/trailing hyphens
   592→      .substring(0, 48);                            // max length
   593→  }

   595→  /**
   596→   * Validate project name format
   597→   * Now accepts ANY name - auto-slugifies for internal use
   598→   */
   599→  function validateProjectName(name) {
   600→    if (!name || name.trim().length === 0) {
   601→      return { valid: false, error: 'Project name is required' };
   602→    }
   603→    if (name.trim().length < 1) {
   604→      return { valid: false, error: 'Please enter a name' };
   605→    }
   606→    if (name.length > 60) {
   607→      return { valid: false, error: 'Name must be under 60 characters' };
   608→    }
   609→    // Accept anything - we'll slugify it internally
   610→    const slug = slugify(name);
   611→    if (slug.length < 1) {
   612→      return { valid: false, error: 'Name must contain at least one letter or number' };
   613→    }
   614→    return { valid: true, error: null, slug };
   615→  }
   616→
   617→  /**
   618→   * Validate filename
   619→   */
   620→  function validateFilename(name) {
   621→    if (!name || name.trim().length === 0) {
   622→      return { valid: false, error: 'Filename is required' };
   623→    }
   624→    if (/[<>:"/\\|?*]/.test(name)) {
   625→      return { valid: false, error: 'Filename contains invalid characters' };
   626→    }
   627→    if (name === '.' || name === '..') {
   628→      return { valid: false, error: 'Invalid filename' };
   629→    }
   630→    return { valid: true, error: null };
   631→  }
